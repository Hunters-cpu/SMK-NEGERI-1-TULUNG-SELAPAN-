import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { Activity } from '../types';
import { MediaUploadField } from '../components/MediaUploadField';
import { SocialActions } from '../components/SocialActions';
import { ScrollEmboss } from '../components/ScrollEmboss';
import { QRCodeModal, QRCodeData } from '../components/QRCodeModal';
import { 
  Camera, 
  Calendar, 
  MapPin, 
  Users, 
  Plus, 
  Search, 
  Filter, 
  Trash2, 
  Edit3, 
  Video as VideoIcon, 
  Image as ImageIcon, 
  Play, 
  X, 
  Eye,
  Sparkles,
  Save,
  Tag,
  FolderPlus,
  QrCode
} from 'lucide-react';

export const ActivitiesView: React.FC = () => {
  const { 
    activities, 
    activityCategories,
    addActivityCategory,
    deleteActivityCategory,
    authSession, 
    addActivity, 
    updateActivity, 
    deleteActivity,
    toggleActivityLike,
    addActivityComment,
    incrementActivityShare,
    requestDeleteConfirm 
  } = useApp();

  const [searchTerm, setSearchTerm] = useState<string>('');
  const [selectedCategory, setSelectedCategory] = useState<string>('Semua');
  const [isModalOpen, setIsModalOpen] = useState<boolean>(false);
  const [isCategoryModalOpen, setIsCategoryModalOpen] = useState<boolean>(false);
  const [newCategoryInput, setNewCategoryInput] = useState<string>('');
  const [editingActivity, setEditingActivity] = useState<Activity | null>(null);
  const [viewDetailActivity, setViewDetailActivity] = useState<Activity | null>(null);
  const [qrModalData, setQrModalData] = useState<QRCodeData | null>(null);

  // Form state for add/edit
  const [formState, setFormState] = useState({
    title: '',
    date: new Intl.DateTimeFormat('id-ID', { day: 'numeric', month: 'long', year: 'numeric' }).format(new Date()),
    category: activityCategories[0] || 'Kelas 10',
    mediaType: 'image' as 'image' | 'video',
    mediaUrl: '',
    description: '',
    location: '',
    peserta: ''
  });

  const categories = ['Semua', ...activityCategories];

  const filteredActivities = activities.filter(act => {
    const matchSearch = act.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
                        act.description.toLowerCase().includes(searchTerm.toLowerCase()) ||
                        (act.location && act.location.toLowerCase().includes(searchTerm.toLowerCase()));
    const matchCategory = selectedCategory === 'Semua' || act.category === selectedCategory;
    return matchSearch && matchCategory;
  });

  const handleOpenAdd = () => {
    setEditingActivity(null);
    setFormState({
      title: '',
      date: new Intl.DateTimeFormat('id-ID', { day: 'numeric', month: 'long', year: 'numeric' }).format(new Date()),
      category: activityCategories[0] || 'Kelas 10',
      mediaType: 'image',
      mediaUrl: '',
      description: '',
      location: '',
      peserta: ''
    });
    setIsModalOpen(true);
  };

  const handleOpenEdit = (act: Activity) => {
    setEditingActivity(act);
    setFormState({
      title: act.title,
      date: act.date,
      category: act.category,
      mediaType: act.mediaType,
      mediaUrl: act.mediaUrl || '',
      description: act.description,
      location: act.location || '',
      peserta: act.peserta || ''
    });
    setIsModalOpen(true);
  };

  const handleSave = (e: React.FormEvent) => {
    e.preventDefault();
    if (!formState.title.trim()) return;

    if (editingActivity) {
      updateActivity(editingActivity.id, {
        title: formState.title.trim(),
        date: formState.date.trim(),
        category: formState.category,
        mediaType: formState.mediaType,
        mediaUrl: formState.mediaUrl.trim() || undefined,
        description: formState.description.trim(),
        location: formState.location.trim() || undefined,
        peserta: formState.peserta.trim() || undefined
      });
    } else {
      addActivity({
        title: formState.title.trim(),
        date: formState.date.trim(),
        category: formState.category,
        mediaType: formState.mediaType,
        mediaUrl: formState.mediaUrl.trim() || undefined,
        description: formState.description.trim(),
        location: formState.location.trim() || undefined,
        peserta: formState.peserta.trim() || undefined
      });
    }
    setIsModalOpen(false);
  };

  const handleAddCategory = (e: React.FormEvent) => {
    e.preventDefault();
    if (!newCategoryInput.trim()) return;
    addActivityCategory(newCategoryInput.trim());
    setNewCategoryInput('');
  };

  return (
    <div className="space-y-8 pb-24 text-[#2D2D2A]">
      {/* Top Banner Header */}
      <div className="bg-white border border-[#E0E0D8] rounded-[32px] p-6 sm:p-8 shadow-sm flex flex-col md:flex-row md:items-center justify-between gap-6">
        <div className="space-y-2">
          <div className="inline-flex items-center gap-2 px-3 py-1 bg-[#F5F5F0] text-[#5A5A40] text-xs font-bold uppercase tracking-widest rounded-full border border-[#E0E0D8]">
            <Camera className="w-3.5 h-3.5 text-[#5A5A40]" />
            <span>Dokumentasi Resmi & Arsip Sekolah</span>
          </div>
          <h2 className="font-serif text-2xl sm:text-3xl font-bold text-[#2D2D2A] tracking-tight">
            Arsip Kegiatan Sekolah
          </h2>
          <p className="text-[#8E8E85] text-sm max-w-xl">
            Seluruh rekam jejak agenda tahunan, prestasi perlombaan, karnaval budaya, serta momen kebersamaan warga sekolah tersimpan di sini.
          </p>
        </div>

        {authSession.role === 'super_admin' && (
          <div className="flex flex-wrap items-center gap-2.5">
            <button
              onClick={() => setIsCategoryModalOpen(true)}
              className="px-4 py-3 rounded-2xl bg-white border border-[#E0E0D8] hover:bg-[#F5F5F0] text-[#5A5A40] font-bold text-xs uppercase tracking-wider flex items-center justify-center gap-2 shadow-xs transition-all cursor-pointer"
            >
              <Tag className="w-4 h-4" />
              <span>Kelola Kategori</span>
            </button>
            <button
              id="btn-add-activity"
              onClick={handleOpenAdd}
              className="px-5 py-3 rounded-2xl bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-xs uppercase tracking-wider flex items-center justify-center gap-2 shadow-sm transition-all cursor-pointer"
            >
              <Plus className="w-4 h-4" />
              <span>Tambah Kegiatan Baru</span>
            </button>
          </div>
        )}
      </div>

      {/* Search & Filter Bar */}
      <div className="flex flex-col sm:flex-row items-center justify-between gap-4">
        {/* Search Input */}
        <div className="relative w-full sm:w-80">
          <Search className="w-4 h-4 absolute left-3.5 top-3 text-[#8E8E85]" />
          <input
            type="text"
            value={searchTerm}
            onChange={e => setSearchTerm(e.target.value)}
            placeholder="Cari nama kegiatan / lokasi..."
            className="w-full pl-10 pr-4 py-2.5 bg-white border border-[#E0E0D8] rounded-2xl text-sm text-[#2D2D2A] placeholder-[#8E8E85] focus:outline-none focus:border-[#5A5A40]"
          />
        </div>

        {/* Category Chips Scroll */}
        <div className="flex items-center gap-2 overflow-x-auto w-full sm:w-auto pb-1 sm:pb-0 scrollbar-none">
          {categories.map(cat => (
            <button
              key={cat}
              onClick={() => setSelectedCategory(cat)}
              className={`px-4 py-2 rounded-xl text-xs font-bold whitespace-nowrap transition-all cursor-pointer ${
                selectedCategory === cat
                  ? 'bg-[#5A5A40] text-white shadow-sm'
                  : 'bg-white text-[#8E8E85] hover:text-[#2D2D2A] border border-[#E0E0D8]'
              }`}
            >
              {cat}
            </button>
          ))}
        </div>
      </div>

      {/* Activities Grid */}
      {filteredActivities.length === 0 ? (
        <div className="bg-white border border-[#E0E0D8] rounded-[32px] p-12 text-center text-[#8E8E85]">
          <Camera className="w-12 h-12 mx-auto text-[#8E8E85] mb-3" />
          <p className="font-serif text-base font-bold text-[#2D2D2A]">Tidak ada kegiatan yang sesuai pencarian.</p>
          <p className="text-xs text-[#8E8E85] mt-1">Coba gunakan kata kunci lain atau ubah filter kategori.</p>
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6 sm:gap-7">
          {filteredActivities.map((act, idx) => (
            <ScrollEmboss key={act.id} delay={idx * 40} interactiveHover className="h-full">
              <div
                className="h-full bg-white border border-[#E0E0D8] rounded-[28px] overflow-hidden shadow-sm hover:shadow-md transition-all flex flex-col justify-between group"
              >
                {/* Media Preview Box - INTINYA JANGAN TAMPILKAN JIKA TIDAK DITAMBAHKAN */}
                {act.mediaUrl ? (
                  <div className="relative aspect-[16/10] w-full bg-[#F5F5F0] overflow-hidden">
                    {act.mediaType === 'video' ? (
                      <div className="relative w-full h-full">
                        <video
                          src={act.mediaUrl}
                          autoPlay
                          muted
                          loop
                          playsInline
                          className="w-full h-full object-cover"
                        />
                        <div className="absolute top-3 left-3 px-2.5 py-1 bg-[#8C4A32] text-white text-[10px] font-bold uppercase tracking-wider rounded-lg shadow flex items-center gap-1.5 backdrop-blur-sm">
                          <VideoIcon className="w-3 h-3" />
                          <span>Video Otomatis</span>
                        </div>
                      </div>
                    ) : (
                      <img
                        src={act.mediaUrl}
                        alt={act.title}
                        className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
                        loading="lazy"
                      />
                    )}

                    {/* Category Badge overlay */}
                    <div className="absolute bottom-3 left-3 px-3 py-1 bg-white/90 text-[#5A5A40] text-xs font-bold rounded-xl border border-[#E0E0D8] backdrop-blur-sm shadow-sm">
                      {act.category}
                    </div>

                    {/* View Details & QR Buttons overlay */}
                    <div className="absolute top-3 right-3 flex items-center gap-1.5">
                      <button
                        type="button"
                        onClick={(e) => {
                          e.stopPropagation();
                          setQrModalData({
                            type: 'activity',
                            id: act.id,
                            title: act.title,
                            subtitle: `${act.date} • ${act.category}${act.location ? ` • ${act.location}` : ''}`,
                            category: act.category
                          });
                        }}
                        className="p-2.5 rounded-xl bg-white/90 hover:bg-[#5A5A40] text-[#5A5A40] hover:text-white backdrop-blur-sm border border-[#E0E0D8] shadow transition-all active:scale-90 cursor-pointer"
                        title="Buat & Pindai Kode QR Kenangan Kegiatan Ini"
                      >
                        <QrCode className="w-4 h-4" />
                      </button>
                      <button
                        type="button"
                        onClick={() => setViewDetailActivity(act)}
                        className="p-2.5 rounded-xl bg-white/90 hover:bg-white text-[#5A5A40] backdrop-blur-sm border border-[#E0E0D8] shadow transition-transform active:scale-90 cursor-pointer"
                        title="Lihat Rincian Lengkap"
                      >
                        <Eye className="w-4 h-4" />
                      </button>
                    </div>
                  </div>
                ) : (
                  /* No Media Header */
                  <div className="p-5 pb-0 flex items-center justify-between">
                    <span className="px-3 py-1 bg-[#F5F5F0] text-[#5A5A40] text-xs font-bold rounded-xl border border-[#E0E0D8]">
                      {act.category}
                    </span>
                    <div className="flex items-center gap-1.5">
                      <button
                        type="button"
                        onClick={(e) => {
                          e.stopPropagation();
                          setQrModalData({
                            type: 'activity',
                            id: act.id,
                            title: act.title,
                            subtitle: `${act.date} • ${act.category}${act.location ? ` • ${act.location}` : ''}`,
                            category: act.category
                          });
                        }}
                        className="p-2 rounded-xl bg-[#F5F5F0] hover:bg-[#5A5A40] text-[#5A5A40] hover:text-white transition-colors cursor-pointer"
                        title="Buat & Pindai Kode QR Kenangan Kegiatan Ini"
                      >
                        <QrCode className="w-4 h-4" />
                      </button>
                      <button
                        type="button"
                        onClick={() => setViewDetailActivity(act)}
                        className="p-2 rounded-xl bg-[#F5F5F0] hover:bg-[#E0E0D8] text-[#5A5A40] transition-colors cursor-pointer"
                        title="Lihat Rincian"
                      >
                        <Eye className="w-4 h-4" />
                      </button>
                    </div>
                  </div>
                )}

                {/* Card Content Body */}
                <div className="p-5 sm:p-6 space-y-3 flex-1 flex flex-col justify-between">
                  <div className="space-y-2">
                    <div className="flex items-center gap-2 text-xs text-[#8E8E85]">
                      <Calendar className="w-3.5 h-3.5 text-[#5A5A40]" />
                      <span>{act.date}</span>
                    </div>

                    <h3 className="font-serif text-base sm:text-lg font-bold text-[#2D2D2A] group-hover:text-[#5A5A40] transition-colors line-clamp-2">
                      {act.title}
                    </h3>

                    <p className="text-xs sm:text-sm text-[#8E8E85] line-clamp-3 leading-relaxed">
                      {act.description}
                    </p>
                  </div>

                  {/* Meta details if available */}
                  {(act.location || act.peserta) && (
                    <div className="pt-3 border-t border-[#E0E0D8] space-y-1.5 text-xs text-[#8E8E85]">
                      {act.location && (
                        <div className="flex items-center gap-2 line-clamp-1">
                          <MapPin className="w-3.5 h-3.5 text-[#8C4A32] flex-shrink-0" />
                          <span>{act.location}</span>
                        </div>
                      )}
                      {act.peserta && (
                        <div className="flex items-center gap-2 line-clamp-1">
                          <Users className="w-3.5 h-3.5 text-[#5A5A40] flex-shrink-0" />
                          <span>{act.peserta}</span>
                        </div>
                      )}
                    </div>
                  )}

                  {/* Social Interactions: Like, Comment, Share */}
                  <SocialActions
                    itemId={act.id}
                    itemTitle={act.title}
                    itemCategory={act.category}
                    likes={act.likes || 0}
                    likedBy={act.likedBy || []}
                    comments={act.comments || []}
                    shares={act.shares || 0}
                    onToggleLike={() => toggleActivityLike(act.id)}
                    onAddComment={(content, author) => addActivityComment(act.id, content, author)}
                    onShare={() => incrementActivityShare(act.id)}
                  />

                  {/* Admin Actions Toolbar */}
                  {authSession.role === 'super_admin' && (
                    <div className="pt-2 border-t border-[#E0E0D8]/60 flex items-center justify-end gap-2">
                      <button
                        type="button"
                        onClick={() => handleOpenEdit(act)}
                        className="px-3 py-1.5 rounded-xl bg-[#F5F5F0] hover:bg-[#E0E0D8] text-[#5A5A40] text-xs font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
                      >
                        <Edit3 className="w-3.5 h-3.5" />
                        <span>Edit</span>
                      </button>
                      <button
                        type="button"
                        onClick={() => {
                          requestDeleteConfirm({
                            itemName: act.title,
                            message: 'Yakin ingin menghapus kegiatan ini? Tindakan ini tidak bisa dibatalkan.',
                            onConfirm: () => deleteActivity(act.id)
                          });
                        }}
                        className="px-3 py-1.5 rounded-xl bg-[#8C4A32]/10 hover:bg-[#8C4A32]/20 text-[#8C4A32] text-xs font-bold flex items-center gap-1.5 transition-colors cursor-pointer"
                      >
                        <Trash2 className="w-3.5 h-3.5" />
                        <span>Hapus</span>
                      </button>
                    </div>
                  )}
                </div>
              </div>
            </ScrollEmboss>
          ))}
        </div>
      )}

      {/* 🔹 CATEGORY MANAGEMENT MODAL (ADMIN UTAMA) */}
      {isCategoryModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-md">
          <div className="relative w-full max-w-md bg-white border border-[#E0E0D8] rounded-[32px] shadow-2xl p-6 text-[#2D2D2A]">
            <div className="flex items-center justify-between pb-3 border-b border-[#E0E0D8] mb-4">
              <h3 className="font-serif text-lg font-bold text-[#5A5A40]">Kelola Kategori Kegiatan</h3>
              <button onClick={() => setIsCategoryModalOpen(false)} className="text-[#8E8E85] hover:text-[#2D2D2A] p-1.5 rounded-xl">
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleAddCategory} className="space-y-3 mb-4">
              <label className="block text-xs font-bold text-[#2D2D2A]">Tambah Kategori Baru</label>
              <div className="flex items-center gap-2">
                <input
                  type="text"
                  required
                  value={newCategoryInput}
                  onChange={e => setNewCategoryInput(e.target.value)}
                  placeholder="Contoh: Acara Alumni / Wisuda"
                  className="flex-1 px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-xs sm:text-sm focus:outline-none focus:border-[#5A5A40]"
                />
                <button
                  type="submit"
                  className="px-4 py-2 bg-[#5A5A40] hover:bg-[#484833] text-white rounded-xl text-xs font-bold cursor-pointer"
                >
                  Tambah
                </button>
              </div>
            </form>

            <div className="space-y-2 max-h-60 overflow-y-auto pr-1">
              <label className="block text-xs font-bold text-[#8E8E85]">Daftar Kategori Saat Ini</label>
              {activityCategories.map(cat => (
                <div key={cat} className="flex items-center justify-between p-2.5 bg-[#F8F8F5] rounded-xl border border-[#E0E0D8] text-xs font-medium">
                  <span>{cat}</span>
                  {activityCategories.length > 1 && (
                    <button
                      onClick={() => deleteActivityCategory(cat)}
                      className="text-[#8C4A32] hover:text-red-700 p-1 cursor-pointer"
                      title="Hapus Kategori"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  )}
                </div>
              ))}
            </div>

            <div className="pt-4 border-t border-[#E0E0D8] text-right mt-4">
              <button
                onClick={() => setIsCategoryModalOpen(false)}
                className="px-4 py-2 bg-[#5A5A40] text-white text-xs font-bold rounded-xl cursor-pointer"
              >
                Selesai
              </button>
            </div>
          </div>
        </div>
      )}

      {/* 🔹 DETAIL VIEW MODAL */}
      {viewDetailActivity && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-md">
          <div className="relative w-full max-w-3xl bg-white border border-[#E0E0D8] rounded-[32px] shadow-2xl overflow-hidden max-h-[92vh] flex flex-col text-[#2D2D2A]">
            {/* Modal Media Box (Only if mediaUrl exists) */}
            {viewDetailActivity.mediaUrl && (
              <div className="relative aspect-[16/9] w-full bg-neutral-900 flex-shrink-0">
                {viewDetailActivity.mediaType === 'video' ? (
                  <video
                    src={viewDetailActivity.mediaUrl}
                    controls
                    autoPlay
                    playsInline
                    className="w-full h-full object-contain"
                  />
                ) : (
                  <img
                    src={viewDetailActivity.mediaUrl}
                    alt={viewDetailActivity.title}
                    className="w-full h-full object-contain"
                  />
                )}

                <button
                  onClick={() => setViewDetailActivity(null)}
                  className="absolute top-4 right-4 p-2 rounded-full bg-black/50 hover:bg-black/70 text-white backdrop-blur-md border border-white/20 cursor-pointer"
                >
                  <X className="w-5 h-5" />
                </button>
              </div>
            )}

            {/* Modal Content */}
            <div className="p-6 sm:p-8 overflow-y-auto space-y-4">
              {!viewDetailActivity.mediaUrl && (
                <div className="flex justify-end">
                  <button
                    onClick={() => setViewDetailActivity(null)}
                    className="p-1.5 rounded-xl hover:bg-[#F5F5F0] text-[#8E8E85] hover:text-[#2D2D2A]"
                  >
                    <X className="w-5 h-5" />
                  </button>
                </div>
              )}

              <div className="flex flex-wrap items-center justify-between gap-2">
                <div className="flex flex-wrap items-center gap-2">
                  <span className="px-3.5 py-1 bg-[#5A5A40] text-white font-bold text-xs rounded-xl">
                    {viewDetailActivity.category}
                  </span>
                  <span className="px-3.5 py-1 bg-[#F5F5F0] text-[#5A5A40] font-semibold text-xs rounded-xl flex items-center gap-1.5 border border-[#E0E0D8]">
                    <Calendar className="w-3.5 h-3.5 text-[#5A5A40]" />
                    {viewDetailActivity.date}
                  </span>
                </div>

                <button
                  type="button"
                  onClick={() => setQrModalData({
                    type: 'activity',
                    id: viewDetailActivity.id,
                    title: viewDetailActivity.title,
                    subtitle: `${viewDetailActivity.date} • ${viewDetailActivity.category}${viewDetailActivity.location ? ` • ${viewDetailActivity.location}` : ''}`,
                    category: viewDetailActivity.category
                  })}
                  className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-xl bg-[#5A5A40] hover:bg-[#484833] text-white text-xs font-bold shadow-xs cursor-pointer active:scale-95 transition-all"
                  title="Buat & Unduh Kode QR Kenangan Kegiatan Ini"
                >
                  <QrCode className="w-3.5 h-3.5" />
                  <span>Pindai / Bagikan QR</span>
                </button>
              </div>

              <h2 className="font-serif text-xl sm:text-2xl font-bold text-[#2D2D2A]">
                {viewDetailActivity.title}
              </h2>

              <p className="text-sm sm:text-base text-[#2D2D2A] leading-relaxed whitespace-pre-line bg-[#F8F8F5] p-5 rounded-2xl border border-[#E0E0D8]">
                {viewDetailActivity.description}
              </p>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3 pt-2">
                {viewDetailActivity.location && (
                  <div className="p-3.5 rounded-2xl bg-[#F8F8F5] border border-[#E0E0D8] text-xs">
                    <span className="text-[#8E8E85] block font-semibold mb-0.5">Lokasi Pelaksanaan:</span>
                    <span className="text-[#2D2D2A] font-bold">{viewDetailActivity.location}</span>
                  </div>
                )}
                {viewDetailActivity.peserta && (
                  <div className="p-3.5 rounded-2xl bg-[#F8F8F5] border border-[#E0E0D8] text-xs">
                    <span className="text-[#8E8E85] block font-semibold mb-0.5">Peserta Kegiatan:</span>
                    <span className="text-[#2D2D2A] font-bold">{viewDetailActivity.peserta}</span>
                  </div>
                )}
              </div>

              {/* Social Interactions in Detail Modal */}
              <SocialActions
                itemId={viewDetailActivity.id}
                itemTitle={viewDetailActivity.title}
                itemCategory={viewDetailActivity.category}
                likes={viewDetailActivity.likes || 0}
                likedBy={viewDetailActivity.likedBy || []}
                comments={viewDetailActivity.comments || []}
                onToggleLike={() => toggleActivityLike(viewDetailActivity.id)}
                onAddComment={(content, author) => addActivityComment(viewDetailActivity.id, content, author)}
              />
            </div>
          </div>
        </div>
      )}

      {/* 🔹 ADD / EDIT ACTIVITY MODAL */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-md">
          <div className="relative w-full max-w-2xl bg-white border border-[#E0E0D8] rounded-[32px] shadow-2xl p-6 sm:p-8 max-h-[90vh] overflow-y-auto text-[#2D2D2A]">
            <div className="flex items-center justify-between pb-4 border-b border-[#E0E0D8] mb-5">
              <h3 className="font-serif text-xl font-bold text-[#5A5A40]">
                {editingActivity ? 'Edit Data Kegiatan Sekolah' : 'Tambah Kegiatan Sekolah Baru'}
              </h3>
              <button onClick={() => setIsModalOpen(false)} className="text-[#8E8E85] hover:text-[#2D2D2A] p-1.5 rounded-xl hover:bg-[#F5F5F0]">
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSave} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Judul Kegiatan *</label>
                <input
                  type="text"
                  required
                  value={formState.title}
                  onChange={e => setFormState({ ...formState, title: e.target.value })}
                  placeholder="Contoh: Peringatan Hari Pahlawan & Teater Kolosal"
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                />
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Tanggal Pelaksanaan *</label>
                  <input
                    type="text"
                    required
                    value={formState.date}
                    onChange={e => setFormState({ ...formState, date: e.target.value })}
                    placeholder="Contoh: 10 November 2024"
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Kategori Kegiatan *</label>
                  <select
                    value={formState.category}
                    onChange={e => setFormState({ ...formState, category: e.target.value })}
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  >
                    {categories.filter(c => c !== 'Semua').map(c => (
                      <option key={c} value={c}>{c}</option>
                    ))}
                  </select>
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Tipe Media (Opsional)</label>
                  <select
                    value={formState.mediaType}
                    onChange={e => setFormState({ ...formState, mediaType: e.target.value as 'image' | 'video' })}
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  >
                    <option value="image">📸 Foto / Dokumentasi Gambar</option>
                    <option value="video">🎥 Video (MP4 / WebM)</option>
                  </select>
                </div>

                <div className="sm:col-span-2">
                  <MediaUploadField
                    id="activity-media-upload"
                    label={formState.mediaType === 'video' ? 'Upload Video Kegiatan (Opsional)' : 'Upload Foto Kegiatan (Opsional)'}
                    mediaType={formState.mediaType}
                    value={formState.mediaUrl}
                    onChange={url => setFormState({ ...formState, mediaUrl: url })}
                    required={false}
                    helperText="Upload foto/video kegiatan bersifat opsional. Jika tidak diisi, media tidak akan ditampilkan."
                  />
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Lokasi Kegiatan</label>
                  <input
                    type="text"
                    value={formState.location}
                    onChange={e => setFormState({ ...formState, location: e.target.value })}
                    placeholder="Contoh: Lapangan Utama / Aula"
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Peserta Terlibat</label>
                  <input
                    type="text"
                    value={formState.peserta}
                    onChange={e => setFormState({ ...formState, peserta: e.target.value })}
                    placeholder="Contoh: Seluruh Siswa Kelas X & Guru"
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Deskripsi Lengkap Kegiatan *</label>
                <textarea
                  rows={4}
                  required
                  value={formState.description}
                  onChange={e => setFormState({ ...formState, description: e.target.value })}
                  placeholder="Ceritakan jalannya kegiatan, susunan acara, dan pencapaian yang diraih..."
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                />
              </div>

              <div className="flex items-center justify-end gap-3 pt-4 border-t border-[#E0E0D8]">
                <button
                  type="button"
                  onClick={() => setIsModalOpen(false)}
                  className="px-4 py-2 text-sm font-bold bg-[#E0E0D8] hover:bg-[#D0D0C8] text-[#2D2D2A] rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="inline-flex items-center gap-2 px-6 py-3 text-xs font-bold uppercase tracking-wider bg-emerald-600 hover:bg-emerald-700 active:bg-emerald-800 text-white rounded-xl shadow-md cursor-pointer hover:scale-[1.02] active:scale-95 transition-all"
                >
                  <Save className="w-4 h-4" />
                  <span>SIMPAN KEGIATAN</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
      {/* 📱 QR CODE MODAL */}
      <QRCodeModal
        isOpen={!!qrModalData}
        onClose={() => setQrModalData(null)}
        data={qrModalData}
      />
    </div>
  );
};
