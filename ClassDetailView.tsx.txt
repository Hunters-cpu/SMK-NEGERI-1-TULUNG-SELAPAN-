import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { ClassRoom, Student, ClassMemoryPhoto, PerangkatKelas, ClassScheduleItem } from '../types';
import { MediaUploadField } from '../components/MediaUploadField';
import { MultiMediaUploadField, MultiMediaItem } from '../components/MultiMediaUploadField';
import { ScrollEmboss } from '../components/ScrollEmboss';
import { SocialActions } from '../components/SocialActions';
import { TikTokIcon } from '../components/TikTokIcon';
import { QRCodeModal } from '../components/QRCodeModal';
import { 
  School, 
  Users, 
  Camera, 
  Award, 
  KeyRound, 
  UserCheck, 
  ShieldCheck, 
  ArrowLeft, 
  Plus, 
  Edit3, 
  Trash2, 
  Search, 
  CheckCircle2, 
  AlertCircle, 
  Lock, 
  User, 
  Phone, 
  Instagram, 
  Sparkles, 
  RefreshCw, 
  Calendar, 
  X, 
  Eye, 
  Save,
  GraduationCap,
  ExternalLink,
  BookOpen,
  Clock,
  QrCode
} from 'lucide-react';

interface ClassDetailViewProps {
  classRoom: ClassRoom;
  onBack: () => void;
}

type ClassSubTab = 'gallery' | 'officers' | 'students' | 'schedules' | 'student_auth';

export const ClassDetailView: React.FC<ClassDetailViewProps> = ({ classRoom, onBack }) => {
  const { 
    authSession, 
    students, 
    updateClass,
    addClassPhoto, 
    addClassPhotosBatch,
    deleteClassPhoto, 
    toggleClassPhotoPrivacy,
    toggleClassPhotoLike,
    addClassPhotoComment,
    incrementClassPhotoShare,
    updateClassPerangkat, 
    addStudent, 
    updateStudent, 
    deleteStudent, 
    updateStudentSelf,
    loginStudent,
    logout,
    resetStudentSession,
    requestDeleteConfirm,
    deviceId,
    classSchedules,
    addClassSchedule,
    updateClassSchedule,
    deleteClassSchedule,
    showToast
  } = useApp();

  const [activeTab, setActiveTab] = useState<ClassSubTab>('gallery');
  const [searchTerm, setSearchTerm] = useState<string>('');
  const [lightboxPhoto, setLightboxPhoto] = useState<ClassMemoryPhoto | null>(null);
  const [isQrModalOpen, setIsQrModalOpen] = useState<boolean>(false);

  // TikTok Edit State (for Super Admin & Class Admin)
  const [isEditingTikTok, setIsEditingTikTok] = useState<boolean>(false);
  const [tiktokInput, setTiktokInput] = useState<string>(classRoom.tiktokUrl || '');

  // Check if current user is authorized to manage this class
  const isSuperAdmin = authSession.role === 'super_admin';
  const isThisClassAdmin = authSession.role === 'class_admin' && authSession.classId === classRoom.id;
  const canManageClass = isSuperAdmin || isThisClassAdmin;

  // Class Students & Stats
  const classStudents = students.filter(s => s.classId === classRoom.id);
  const totalStudents = classStudents.length;
  const maleCount = classStudents.filter(s => s.gender === 'L').length;
  const femaleCount = classStudents.filter(s => s.gender === 'P').length;

  const filteredStudents = classStudents.filter(s =>
    s.name.toLowerCase().includes(searchTerm.toLowerCase()) ||
    s.nisn.includes(searchTerm)
  );

  // Is logged in as a student belonging to this class?
  const currentStudentUser =
    authSession.role === 'student' && authSession.studentId
      ? students.find(s => s.id === authSession.studentId)
      : null;

  // State for Add Photo Modal & Multi-Media Upload
  const [isPhotoModalOpen, setIsPhotoModalOpen] = useState<boolean>(false);
  const [multiMediaList, setMultiMediaList] = useState<MultiMediaItem[]>([]);
  const [photoForm, setPhotoForm] = useState({
    title: '',
    date: new Intl.DateTimeFormat('id-ID', { day: 'numeric', month: 'long', year: 'numeric' }).format(new Date()),
    imageUrl: '',
    description: ''
  });

  // State for Edit Perangkat
  const [isEditingPerangkat, setIsEditingPerangkat] = useState<boolean>(false);
  const [perangkatForm, setPerangkatForm] = useState<PerangkatKelas>({ ...classRoom.perangkat });

  // State for Add/Edit Student Modal
  const [isStudentModalOpen, setIsStudentModalOpen] = useState<boolean>(false);
  const [editingStudent, setEditingStudent] = useState<Student | null>(null);
  const [studentForm, setStudentForm] = useState({
    nisn: '',
    name: '',
    gender: 'L' as 'L' | 'P',
    photoUrl: '',
    quote: '',
    citaCita: '',
    pesanKesan: '',
    instagram: '',
    phone: '',
    passwordHash: '123456'
  });

  // State for Student In-Class Login Form
  const [inClassLoginStudentId, setInClassLoginStudentId] = useState<string>('');
  const [inClassLoginPassword, setInClassLoginPassword] = useState<string>('');
  const [inClassLoginError, setInClassLoginError] = useState<string>('');

  // State for Student Self Profile Editor
  const [selfProfileForm, setSelfProfileForm] = useState({
    quote: currentStudentUser?.quote || '',
    citaCita: currentStudentUser?.citaCita || '',
    pesanKesan: currentStudentUser?.pesanKesan || '',
    instagram: currentStudentUser?.instagram || '',
    phone: currentStudentUser?.phone || '',
    photoUrl: currentStudentUser?.photoUrl || '',
    newPassword: ''
  });

  // Handlers
  const handleSavePhoto = (e: React.FormEvent) => {
    e.preventDefault();

    const uploaderName = isSuperAdmin ? 'Admin Utama' : `Admin Kelas ${classRoom.name}`;

    // If multi-media items are selected
    if (multiMediaList.length > 0) {
      const itemsToUpload = multiMediaList.map(item => ({
        title: item.title?.trim() || photoForm.title.trim() || 'Foto Kenangan Kelas',
        date: photoForm.date.trim(),
        imageUrl: item.url,
        description: photoForm.description.trim() || undefined,
        uploadedBy: uploaderName
      }));

      addClassPhotosBatch(classRoom.id, itemsToUpload);
      setMultiMediaList([]);
      setPhotoForm({
        title: '',
        date: new Intl.DateTimeFormat('id-ID', { day: 'numeric', month: 'long', year: 'numeric' }).format(new Date()),
        imageUrl: '',
        description: ''
      });
      setIsPhotoModalOpen(false);
      return;
    }

    // Single item fallback
    if (!photoForm.title.trim() || !photoForm.imageUrl.trim()) return;

    addClassPhoto(classRoom.id, {
      title: photoForm.title.trim(),
      date: photoForm.date.trim(),
      imageUrl: photoForm.imageUrl.trim(),
      description: photoForm.description.trim() || undefined,
      uploadedBy: uploaderName
    });
    setPhotoForm({
      title: '',
      date: new Intl.DateTimeFormat('id-ID', { day: 'numeric', month: 'long', year: 'numeric' }).format(new Date()),
      imageUrl: '',
      description: ''
    });
    setIsPhotoModalOpen(false);
  };

  const handleSavePerangkat = (e: React.FormEvent) => {
    e.preventDefault();
    updateClassPerangkat(classRoom.id, perangkatForm);
    setIsEditingPerangkat(false);
  };

  const handleOpenAddStudent = () => {
    setEditingStudent(null);
    setStudentForm({
      nisn: `006${Math.floor(1000000 + Math.random() * 9000000)}`,
      name: '',
      gender: 'L',
      photoUrl: '',
      quote: '',
      citaCita: '',
      pesanKesan: '',
      instagram: '',
      phone: '',
      passwordHash: '123456'
    });
    setIsStudentModalOpen(true);
  };

  const handleOpenEditStudent = (std: Student) => {
    setEditingStudent(std);
    setStudentForm({
      nisn: std.nisn,
      name: std.name,
      gender: std.gender,
      photoUrl: std.photoUrl || '',
      quote: std.quote || '',
      citaCita: std.citaCita || '',
      pesanKesan: std.pesanKesan || '',
      instagram: std.instagram || '',
      phone: std.phone || '',
      passwordHash: std.passwordHash || '123456'
    });
    setIsStudentModalOpen(true);
  };

  const handleSaveStudent = (e: React.FormEvent) => {
    e.preventDefault();
    if (!studentForm.name.trim()) return;

    if (editingStudent) {
      updateStudent(editingStudent.id, {
        nisn: studentForm.nisn.trim(),
        name: studentForm.name.trim(),
        gender: studentForm.gender,
        photoUrl: studentForm.photoUrl.trim() || undefined,
        quote: studentForm.quote.trim() || undefined,
        citaCita: studentForm.citaCita.trim() || undefined,
        pesanKesan: studentForm.pesanKesan.trim() || undefined,
        instagram: studentForm.instagram.trim() || undefined,
        phone: studentForm.phone.trim() || undefined,
        passwordHash: studentForm.passwordHash.trim() || '123456'
      });
    } else {
      addStudent({
        nisn: studentForm.nisn.trim() || `006${Date.now().toString().slice(-7)}`,
        name: studentForm.name.trim(),
        gender: studentForm.gender,
        photoUrl: studentForm.photoUrl.trim() || undefined,
        quote: studentForm.quote.trim() || undefined,
        citaCita: studentForm.citaCita.trim() || undefined,
        pesanKesan: studentForm.pesanKesan.trim() || undefined,
        instagram: studentForm.instagram.trim() || undefined,
        phone: studentForm.phone.trim() || undefined,
        classId: classRoom.id,
        passwordHash: studentForm.passwordHash.trim() || '123456'
      });
    }
    setIsStudentModalOpen(false);
  };

  const handleInClassLogin = (e: React.FormEvent) => {
    e.preventDefault();
    setInClassLoginError('');

    if (!inClassLoginStudentId) {
      setInClassLoginError('Silakan pilih nama Anda terlebih dahulu.');
      return;
    }
    if (!inClassLoginPassword) {
      setInClassLoginError('Silakan masukkan kata sandi Anda.');
      return;
    }

    const res = loginStudent(inClassLoginStudentId, inClassLoginPassword);
    if (!res.success && res.message) {
      setInClassLoginError(res.message);
    }
  };

  const handleSaveSelfProfile = (e: React.FormEvent) => {
    e.preventDefault();
    if (!currentStudentUser) return;

    updateStudentSelf(currentStudentUser.id, {
      quote: selfProfileForm.quote,
      citaCita: selfProfileForm.citaCita,
      pesanKesan: selfProfileForm.pesanKesan,
      instagram: selfProfileForm.instagram,
      phone: selfProfileForm.phone,
      photoUrl: selfProfileForm.photoUrl || undefined,
      newPassword: selfProfileForm.newPassword ? selfProfileForm.newPassword : undefined
    });
  };

  const handleSaveTikTok = (e: React.FormEvent) => {
    e.preventDefault();
    updateClass(classRoom.id, {
      tiktokUrl: tiktokInput.trim() || undefined
    });
    setIsEditingTikTok(false);
  };

  // Schedules for this class (Syncs automatically with ClassSchedulesView)
  const [selectedScheduleDay, setSelectedScheduleDay] = useState<string>('Semua');
  const [isScheduleModalOpen, setIsScheduleModalOpen] = useState<boolean>(false);
  const [editingSchedule, setEditingSchedule] = useState<ClassScheduleItem | null>(null);
  const [scheduleForm, setScheduleForm] = useState({
    day: 'Senin' as ClassScheduleItem['day'],
    timeSlot: '07:30 - 09:00',
    subject: '',
    teacherName: '',
    room: classRoom.name
  });

  const thisClassSchedules = classSchedules.filter(s => s.classId === classRoom.id);
  const filteredClassSchedules = thisClassSchedules.filter(item => 
    selectedScheduleDay === 'Semua' || item.day === selectedScheduleDay
  );

  const daysOrder: Record<string, number> = {
    'Senin': 1, 'Selasa': 2, 'Rabu': 3, 'Kamis': 4, 'Jumat': 5, 'Sabtu': 6
  };

  const sortedClassSchedules = [...filteredClassSchedules].sort((a, b) => {
    const dayDiff = (daysOrder[a.day] || 99) - (daysOrder[b.day] || 99);
    if (dayDiff !== 0) return dayDiff;
    return a.timeSlot.localeCompare(b.timeSlot);
  });

  const handleOpenAddSchedule = () => {
    setEditingSchedule(null);
    setScheduleForm({
      day: selectedScheduleDay !== 'Semua' ? (selectedScheduleDay as any) : 'Senin',
      timeSlot: '07:30 - 09:00',
      subject: '',
      teacherName: '',
      room: classRoom.name
    });
    setIsScheduleModalOpen(true);
  };

  const handleOpenEditSchedule = (item: ClassScheduleItem) => {
    setEditingSchedule(item);
    setScheduleForm({
      day: item.day,
      timeSlot: item.timeSlot,
      subject: item.subject,
      teacherName: item.teacherName,
      room: item.room || classRoom.name
    });
    setIsScheduleModalOpen(true);
  };

  const handleSaveSchedule = (e: React.FormEvent) => {
    e.preventDefault();
    if (!scheduleForm.subject.trim() || !scheduleForm.teacherName.trim()) {
      showToast('Silakan isi nama mata pelajaran dan nama guru pengajar.', 'warning');
      return;
    }

    if (editingSchedule) {
      updateClassSchedule(editingSchedule.id, {
        day: scheduleForm.day,
        timeSlot: scheduleForm.timeSlot.trim(),
        subject: scheduleForm.subject.trim(),
        teacherName: scheduleForm.teacherName.trim(),
        room: scheduleForm.room.trim() || classRoom.name
      });
      showToast(`Mata pelajaran "${scheduleForm.subject}" berhasil diperbarui!`, 'success');
    } else {
      addClassSchedule({
        classId: classRoom.id,
        day: scheduleForm.day,
        timeSlot: scheduleForm.timeSlot.trim(),
        subject: scheduleForm.subject.trim(),
        teacherName: scheduleForm.teacherName.trim(),
        room: scheduleForm.room.trim() || classRoom.name
      });
      showToast(`Mata pelajaran "${scheduleForm.subject}" berhasil ditambahkan ke ${classRoom.name}!`, 'success');
    }
    setIsScheduleModalOpen(false);
  };

  const handleDeleteSchedule = (item: ClassScheduleItem) => {
    requestDeleteConfirm({
      itemName: `Mata Pelajaran "${item.subject}" (${item.day}, ${item.timeSlot})`,
      message: `Yakin ingin menghapus jadwal "${item.subject}" untuk kelas ${classRoom.name}? Jadwal di menu KBM juga akan disinkronkan secara otomatis.`,
      onConfirm: () => {
        deleteClassSchedule(item.id);
        showToast('Jadwal pelajaran berhasil dihapus!', 'info');
      }
    });
  };

  return (
    <div className="space-y-8 pb-24 text-[#2D2D2A]">
      {/* 🔹 BACK TO CLASSES BUTTON & CLASS HEADER BANNER */}
      <div className="space-y-4">
        <button
          onClick={onBack}
          className="inline-flex items-center gap-2 px-4 py-2 bg-white hover:bg-[#F5F5F0] text-[#5A5A40] text-xs font-bold uppercase tracking-wider rounded-xl border border-[#E0E0D8] transition-colors cursor-pointer shadow-sm"
        >
          <ArrowLeft className="w-4 h-4" />
          <span>Kembali ke Daftar Semua Kelas</span>
        </button>

        <div className="bg-white border border-[#E0E0D8] rounded-[32px] p-6 sm:p-8 shadow-sm flex flex-col md:flex-row md:items-center justify-between gap-6">
          <div className="space-y-2">
            <div className="flex items-center gap-2 flex-wrap">
              <span className="px-3 py-1 bg-[#F5F5F0] text-[#5A5A40] font-bold text-xs rounded-full border border-[#E0E0D8]">
                {classRoom.grade} • TA {classRoom.academicYear}
              </span>
              {canManageClass && (
                <span className="px-3 py-1 bg-[#8C4A32]/10 text-[#8C4A32] font-bold text-xs rounded-full border border-[#8C4A32]/20 flex items-center gap-1">
                  <UserCheck className="w-3.5 h-3.5" />
                  <span>Mode Akses Pengelola</span>
                </span>
              )}
            </div>

            <h1 className="font-serif text-3xl sm:text-4xl font-bold text-[#2D2D2A] tracking-tight">
              {classRoom.name}
            </h1>
            <p className="text-[#8E8E85] text-sm">
              Wali Kelas: <strong className="text-[#5A5A40]">{classRoom.perangkat.waliKelas}</strong>
              {classRoom.jurusan && ` • Jurusan ${classRoom.jurusan}`}
            </p>
          </div>

          {/* Quick Stats Badges & TikTok Button */}
          <div className="flex flex-wrap items-center gap-3">
            <div className="px-4 py-3 bg-[#F5F5F0] border border-[#E0E0D8] rounded-2xl text-center min-w-[90px]">
              <span className="text-[10px] font-bold uppercase tracking-wider text-[#8E8E85] block">Total Siswa</span>
              <span className="font-serif text-lg font-bold text-[#2D2D2A]">{totalStudents}</span>
            </div>
            <div className="px-4 py-3 bg-[#F5F5F0] border border-[#E0E0D8] rounded-2xl text-center min-w-[80px]">
              <span className="text-[10px] font-bold uppercase tracking-wider text-[#5A5A40] block">Laki-Laki</span>
              <span className="font-serif text-lg font-bold text-[#5A5A40]">{maleCount}</span>
            </div>
            <div className="px-4 py-3 bg-[#F5F5F0] border border-[#E0E0D8] rounded-2xl text-center min-w-[80px]">
              <span className="text-[10px] font-bold uppercase tracking-wider text-[#8C4A32] block">Perempuan</span>
              <span className="font-serif text-lg font-bold text-[#8C4A32]">{femaleCount}</span>
            </div>

            {/* 📱 TOMBOL QR CODE KENANGAN KELAS */}
            <button
              id="btn-qr-kelas"
              type="button"
              onClick={() => setIsQrModalOpen(true)}
              className="px-4 py-3 bg-[#5A5A40] hover:bg-[#464632] text-white rounded-2xl text-xs font-bold uppercase tracking-wider flex items-center gap-2 shadow-sm transition-all cursor-pointer active:scale-95 group"
              title="Buat & Pindai Kode QR Kenangan Digital Kelas Ini Langsung dari HP"
            >
              <QrCode className="w-4 h-4 text-white group-hover:rotate-12 transition-transform" />
              <span>QR Kenangan</span>
            </button>

            {/* 🎵 TOMBOL AKUN TIKTOK KELAS */}
            {classRoom.tiktokUrl && (
              <a
                id="btn-tiktok-kelas"
                href={classRoom.tiktokUrl}
                target="_blank"
                rel="noopener noreferrer"
                className="px-4 py-3 bg-black hover:bg-neutral-900 text-white rounded-2xl text-xs font-bold uppercase tracking-wider flex items-center gap-2 shadow-sm transition-all cursor-pointer active:scale-95 group border border-neutral-800"
                title={`Kunjungi Akun TikTok Resmi Kelas ${classRoom.name}`}
              >
                <TikTokIcon className="w-4 h-4 text-white group-hover:scale-110 transition-transform" />
                <span>TikTok Kelas</span>
                <ExternalLink className="w-3.5 h-3.5 opacity-70" />
              </a>
            )}

            {/* Tombol Ubah / Tambah TikTok oleh Admin Kelas & Admin Utama */}
            {canManageClass && (
              <button
                id="btn-manage-tiktok-kelas"
                type="button"
                onClick={() => {
                  setTiktokInput(classRoom.tiktokUrl || '');
                  setIsEditingTikTok(true);
                }}
                className={`px-3.5 py-3 rounded-2xl text-xs font-bold uppercase tracking-wider flex items-center gap-2 border transition-all cursor-pointer active:scale-95 ${
                  classRoom.tiktokUrl 
                    ? 'bg-[#F5F5F0] hover:bg-[#E0E0D8] text-[#5A5A40] border-[#E0E0D8]' 
                    : 'bg-black hover:bg-neutral-900 text-white border-neutral-800 shadow-sm'
                }`}
                title="Ubah tautan akun TikTok kelas ini (Admin Kelas & Admin Utama)"
              >
                <TikTokIcon className={`w-4 h-4 ${classRoom.tiktokUrl ? 'text-black' : 'text-white'}`} />
                <span>{classRoom.tiktokUrl ? 'Ubah TikTok' : '+ Akun TikTok'}</span>
                <Edit3 className="w-3.5 h-3.5 opacity-70" />
              </button>
            )}
          </div>
        </div>
      </div>

      {/* 🔹 SUB NAVIGATION TABS */}
      <div className="grid grid-cols-2 sm:grid-cols-5 gap-2 p-1.5 bg-white rounded-2xl border border-[#E0E0D8] shadow-sm">
        <button
          onClick={() => setActiveTab('gallery')}
          className={`py-3 px-3 rounded-xl text-xs sm:text-sm font-bold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'gallery'
              ? 'bg-[#5A5A40] text-white shadow-sm'
              : 'text-[#8E8E85] hover:text-[#2D2D2A]'
          }`}
        >
          <Camera className="w-4 h-4" />
          <span>📸 Galeri ({classRoom.gallery.length})</span>
        </button>

        <button
          onClick={() => setActiveTab('officers')}
          className={`py-3 px-3 rounded-xl text-xs sm:text-sm font-bold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'officers'
              ? 'bg-[#5A5A40] text-white shadow-sm'
              : 'text-[#8E8E85] hover:text-[#2D2D2A]'
          }`}
        >
          <Award className="w-4 h-4" />
          <span>👥 Perangkat</span>
        </button>

        <button
          onClick={() => setActiveTab('students')}
          className={`py-3 px-3 rounded-xl text-xs sm:text-sm font-bold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'students'
              ? 'bg-[#5A5A40] text-white shadow-sm'
              : 'text-[#8E8E85] hover:text-[#2D2D2A]'
          }`}
        >
          <Users className="w-4 h-4" />
          <span>📋 Siswa ({totalStudents})</span>
        </button>

        <button
          id="tab-class-schedules"
          onClick={() => setActiveTab('schedules')}
          className={`py-3 px-3 rounded-xl text-xs sm:text-sm font-bold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'schedules'
              ? 'bg-[#5A5A40] text-white shadow-sm'
              : 'text-[#8E8E85] hover:text-[#2D2D2A]'
          }`}
        >
          <BookOpen className="w-4 h-4" />
          <span>📚 Mata Pelajaran ({thisClassSchedules.length})</span>
        </button>

        <button
          onClick={() => setActiveTab('student_auth')}
          className={`py-3 px-3 rounded-xl text-xs sm:text-sm font-bold flex items-center justify-center gap-1.5 transition-all cursor-pointer ${
            activeTab === 'student_auth'
              ? 'bg-[#8C4A32] text-white shadow-sm'
              : 'text-[#8E8E85] hover:text-[#2D2D2A]'
          }`}
        >
          <KeyRound className="w-4 h-4" />
          <span>🔑 Login {currentStudentUser && '✓'}</span>
        </button>
      </div>

      {/* 🔹 TAB 1: GALERI KEGIATAN KELAS */}
      {activeTab === 'gallery' && (
        <section className="space-y-6">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
            <div>
              <h3 className="font-serif text-xl font-bold text-[#2D2D2A]">Dokumentasi Kenangan {classRoom.name}</h3>
              <p className="text-xs sm:text-sm text-[#8E8E85]">
                Kumpulan foto perpisahan, praktikum bersama, dan rekam jejak keceriaan di ruang kelas.
              </p>
            </div>

            {canManageClass && (
              <button
                onClick={() => setIsPhotoModalOpen(true)}
                className="px-4 py-2.5 rounded-xl bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-xs uppercase tracking-wider flex items-center gap-2 shadow-sm cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Unggah Foto Kenangan</span>
              </button>
            )}
          </div>

          {classRoom.gallery.length === 0 ? (
            <div className="bg-white border border-[#E0E0D8] rounded-[32px] p-12 text-center text-[#8E8E85]">
              <Camera className="w-12 h-12 mx-auto text-[#8E8E85] mb-3" />
              <p className="font-serif text-base font-bold text-[#2D2D2A]">Belum ada foto kenangan kelas yang diunggah.</p>
              {canManageClass && (
                <button
                  onClick={() => setIsPhotoModalOpen(true)}
                  className="mt-4 px-4 py-2 bg-[#5A5A40] text-white hover:bg-[#484833] text-xs font-bold rounded-xl uppercase tracking-wider"
                >
                  Mulai Unggah Foto Sekarang
                </button>
              )}
            </div>
          ) : (
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
              {classRoom.gallery.filter(p => canManageClass || !p.isPrivate).map(photo => (
                <div
                  key={photo.id}
                  className="bg-white border border-[#E0E0D8] rounded-[24px] overflow-hidden shadow-sm hover:shadow-md transition-all flex flex-col justify-between group relative"
                >

                  <div>

                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Ganti Kata Sandi Siswa (Opsional)</label>

                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"

                </div>

                <button
                  type="submit"
                  className="w-full py-3 bg-emerald-600 hover:bg-emerald-700 active:bg-emerald-800 text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-md cursor-pointer hover:scale-[1.01] active:scale-95 transition-all flex items-center justify-center gap-2"

                  <Save className="w-4 h-4" />
                  <span>SIMPAN PROFIL KENANGAN SAYA</span>

              </form>

          ) : (
            /* Student is NOT logged in: Direct in-class login form */

              <div className="text-center space-y-2 pb-4 border-b border-[#E0E0D8]">

                  <KeyRound className="w-7 h-7" />
                </div>
                <h3 className="font-serif text-xl font-bold text-[#2D2D2A]">Login Siswa {classRoom.name}</h3>

                  Cukup pilih nama Anda dan masukkan kata sandi tanpa perlu email. Satu nama hanya dapat aktif di satu perangkat.
                </p>

              {inClassLoginError && (
                <div className="p-3.5 bg-[#8C4A32]/10 border border-[#8C4A32]/30 rounded-2xl text-[#8C4A32] text-xs sm:text-sm flex items-start gap-2.5">

                    onChange={e => setInClassLoginStudentId(e.target.value)}

                  >
                    <option value="">-- Pilih Nama Siswa --</option>
                    {classStudents.map(s => (
                      <option key={s.id} value={s.id}>
                        {s.name} ({s.gender === 'L' ? 'L' : 'P'})

                </div>

                <div>
                  <div className="flex items-center justify-between mb-1">
                    <label className="text-xs font-bold text-[#2D2D2A]">Kata Sandi Siswa</label>

                    type="password"
                    value={inClassLoginPassword}
                    onChange={e => setInClassLoginPassword(e.target.value)}

                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"

                <button
                  type="submit"
                  className="w-full py-3 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm cursor-pointer flex items-center justify-center gap-2"

                  <KeyRound className="w-4 h-4" />

            </div>
          )}
        </section>
      )}

      {/* 🔹 TAB 5: MATA PELAJARAN & JADWAL KBM KELAS */}
      {activeTab === 'schedules' && (
        <section className="space-y-6">
          <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
            <div>
              <div className="inline-flex items-center gap-1.5 px-3 py-1 bg-[#5A5A40]/10 text-[#5A5A40] text-xs font-bold uppercase tracking-wider rounded-full mb-1">

                  type="button"

                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Catatan / Deskripsi Tambahan</label>

                  className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"

                >

          </div>
        </div>
      )}

      {/* 🔹 ADD/EDIT STUDENT MODAL */}
      {isStudentModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-md">
          <div className="relative w-full max-w-xl bg-white border border-[#E0E0D8] rounded-[32px] shadow-2xl p-6 sm:p-8 max-h-[90vh] overflow-y-auto text-[#2D2D2A]">
            <div className="flex items-center justify-between pb-3 border-b border-[#E0E0D8] mb-4">
              <h3 className="font-serif text-lg font-bold text-[#5A5A40]">
                {editingStudent ? 'Edit Data Siswa' : 'Tambah Siswa Baru ke Kelas'}
              </h3>
              <button onClick={() => setIsStudentModalOpen(false)} className="text-[#8E8E85] hover:text-[#2D2D2A] p-1.5 rounded-xl hover:bg-[#F5F5F0]">
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveStudent} className="space-y-4">
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Nama Lengkap Siswa *</label>
                  <input
                    type="text"
                    required
                    value={studentForm.name}
                    onChange={e => setStudentForm({ ...studentForm, name: e.target.value })}
                    placeholder="Contoh: Muhammad Rizky"
                    className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Jenis Kelamin *</label>
                  <select
                    value={studentForm.gender}
                    onChange={e => setStudentForm({ ...studentForm, gender: e.target.value as 'L' | 'P' })}
                    className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  >
                    <option value="L">Laki-laki (L)</option>
                    <option value="P">Perempuan (P)</option>
                  </select>
                </div>
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">NISN Siswa</label>
                  <input
                    type="text"
                    value={studentForm.nisn}
                    onChange={e => setStudentForm({ ...studentForm, nisn: e.target.value })}
                    className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Kata Sandi Login (Bawaan: 123456)</label>
                  <input
                    type="text"
                    value={studentForm.passwordHash}
                    onChange={e => setStudentForm({ ...studentForm, passwordHash: e.target.value })}
                    className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
              </div>

              <div>
                <MediaUploadField
                  id="student-form-photo-upload"
                  label="Upload Foto Profil Siswa"
                  mediaType="image"
                  value={studentForm.photoUrl}
                  onChange={url => setStudentForm({ ...studentForm, photoUrl: url })}
                  helperText="Upload pas foto atau foto santai siswa dari galeri perangkat."
                />
              </div>

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Quotes / Kata Kenangan</label>
                <input
                  type="text"
                  value={studentForm.quote}
                  onChange={e => setStudentForm({ ...studentForm, quote: e.target.value })}
                  placeholder="Kutipan inspirasi..."
                  className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                />
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Cita-cita</label>
                  <input
                    type="text"
                    value={studentForm.citaCita}
                    onChange={e => setStudentForm({ ...studentForm, citaCita: e.target.value })}
                    className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">Instagram (@username)</label>
                  <input
                    type="text"
                    value={studentForm.instagram}

                    className="w-full px-3.5 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>
              </div>

              <div className="flex items-center justify-end gap-3 pt-3 border-t border-[#E0E0D8]">
                <button
                  type="button"
                  onClick={() => setIsStudentModalOpen(false)}
                  className="px-4 py-2 bg-[#E0E0D8] hover:bg-[#D0D0C8] text-[#2D2D2A] text-xs font-bold rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="inline-flex items-center gap-2 px-6 py-2.5 bg-emerald-600 hover:bg-emerald-700 active:bg-emerald-800 text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-md cursor-pointer hover:scale-[1.02] active:scale-95 transition-all"

                  <Save className="w-4 h-4" />
                  <span>SIMPAN DATA SISWA</span>

            </form>
          </div>
        </div>

      {/* 🔹 EDIT TIKTOK KELAS MODAL (Admin Kelas & Admin Utama) */}
      {isEditingTikTok && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-md">
          <div className="relative w-full max-w-md bg-white border border-[#E0E0D8] rounded-[32px] shadow-2xl p-6 sm:p-7 text-[#2D2D2A]">
            <div className="flex items-center justify-between pb-3 border-b border-[#E0E0D8] mb-4">
              <div className="flex items-center gap-2">
                <div className="p-2 rounded-xl bg-black text-white">
                  <TikTokIcon className="w-4 h-4 text-white" />
                </div>
                <div>
                  <h3 className="font-serif text-lg font-bold text-[#2D2D2A]">
                    Akun TikTok Kelas
                  </h3>
                  <p className="text-xs text-[#8E8E85]">{classRoom.name}</p>
                </div>
              </div>
              <button
                onClick={() => setIsEditingTikTok(false)}
                className="text-[#8E8E85] hover:text-[#2D2D2A] p-1.5 rounded-xl hover:bg-[#F5F5F0] cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveTikTok} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1.5">
                  Tautan / URL Akun TikTok Kelas
                </label>
                <input
                  type="url"
                  value={tiktokInput}
                  onChange={e => setTiktokInput(e.target.value)}
                  placeholder="https://www.tiktok.com/@kelas_12ipa1"
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:border-[#5A5A40] focus:outline-none"
                  autoFocus
                />
                <p className="text-[11px] text-[#8E8E85] mt-1.5">
                  Tautan ini dapat diubah oleh Admin Kelas dan Admin Utama, dan akan disinkronkan langsung ke pusat data.
                </p>
              </div>

              <div className="flex items-center justify-end gap-2.5 pt-3 border-t border-[#E0E0D8]">
                {classRoom.tiktokUrl && (
                  <button
                    type="button"
                    onClick={() => {
                      updateClass(classRoom.id, { tiktokUrl: undefined });
                      setIsEditingTikTok(false);
                    }}
                    className="px-3.5 py-2 text-xs font-bold text-[#8C4A32] hover:bg-[#8C4A32]/10 rounded-xl cursor-pointer mr-auto"
                  >
                    Hapus Link
                  </button>
                )}
                <button
                  type="button"
                  onClick={() => setIsEditingTikTok(false)}
                  className="px-4 py-2 text-xs font-bold bg-[#E0E0D8] hover:bg-[#D0D0C8] text-[#2D2D2A] rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="inline-flex items-center gap-2 px-5 py-2 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm cursor-pointer active:scale-95 transition-all"
                >
                  <Save className="w-3.5 h-3.5" />
                  <span>Simpan Link</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 🔹 ADD/EDIT CLASS SCHEDULE & SUBJECT MODAL */}
      {isScheduleModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-md">
          <div className="relative w-full max-w-lg bg-white border border-[#E0E0D8] rounded-[28px] shadow-2xl p-6 sm:p-7 text-[#2D2D2A] max-h-[90vh] overflow-y-auto">
            <div className="flex items-center justify-between pb-3 border-b border-[#E0E0D8] mb-4">
              <div className="flex items-center gap-2.5">
                <div className="w-10 h-10 rounded-xl bg-[#5A5A40] text-white flex items-center justify-center">
                  <BookOpen className="w-5 h-5" />
                </div>
                <div>
                  <h3 className="font-serif text-lg font-bold text-[#5A5A40]">
                    {editingSchedule ? 'Edit Mata Pelajaran' : 'Tambah Mata Pelajaran'}
                  </h3>
                  <p className="text-xs text-[#8E8E85]">
                    Kelas {classRoom.name} • Sinkronisasi otomatis ke Jadwal KBM
                  </p>
                </div>
              </div>
              <button
                type="button"
                onClick={() => setIsScheduleModalOpen(false)}
                className="text-[#8E8E85] hover:text-[#2D2D2A] p-1.5 rounded-xl hover:bg-[#F5F5F0] cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveSchedule} className="space-y-4">
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1.5">
                    Hari Pelajaran *
                  </label>
                  <select
                    value={scheduleForm.day}
                    onChange={e => setScheduleForm({ ...scheduleForm, day: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:border-[#5A5A40] focus:outline-none"
                  >
                    <option value="Senin">Senin</option>
                    <option value="Selasa">Selasa</option>
                    <option value="Rabu">Rabu</option>
                    <option value="Kamis">Kamis</option>
                    <option value="Jumat">Jumat</option>
                    <option value="Sabtu">Sabtu</option>
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1.5">
                    Jam Belajar / Alokasi Waktu *
                  </label>
                  <input
                    type="text"
                    required
                    value={scheduleForm.timeSlot}
                    onChange={e => setScheduleForm({ ...scheduleForm, timeSlot: e.target.value })}
                    placeholder="Contoh: 07:30 - 09:00"
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:border-[#5A5A40] focus:outline-none"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1.5">
                  Nama Mata Pelajaran *
                </label>
                <input
                  type="text"
                  required
                  value={scheduleForm.subject}
                  onChange={e => setScheduleForm({ ...scheduleForm, subject: e.target.value })}
                  placeholder="Contoh: Matematika Peminatan / Fisika Terapan"
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:border-[#5A5A40] focus:outline-none"
                />
              </div>

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1.5">
                  Nama Guru Pengampu / Pengajar *
                </label>
                <input
                  type="text"
                  required
                  value={scheduleForm.teacherName}
                  onChange={e => setScheduleForm({ ...scheduleForm, teacherName: e.target.value })}
                  placeholder="Contoh: Dra. Hj. Siti Aminah, M.Pd"
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:border-[#5A5A40] focus:outline-none"
                />
              </div>

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1.5">
                  Ruang Belajar / Laboratorium (Opsional)
                </label>
                <input
                  type="text"
                  value={scheduleForm.room}
                  onChange={e => setScheduleForm({ ...scheduleForm, room: e.target.value })}
                  placeholder={`Contoh: Ruang ${classRoom.name} / Lab Komputer`}
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:border-[#5A5A40] focus:outline-none"
                />
              </div>

              <div className="flex items-center justify-end gap-2.5 pt-3 border-t border-[#E0E0D8]">
                <button
                  type="button"
                  onClick={() => setIsScheduleModalOpen(false)}
                  className="px-4 py-2.5 text-xs font-bold bg-[#E0E0D8] hover:bg-[#D0D0C8] text-[#2D2D2A] rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="inline-flex items-center gap-2 px-5 py-2.5 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm cursor-pointer active:scale-95 transition-all"
                >
                  <Save className="w-3.5 h-3.5" />
                  <span>{editingSchedule ? 'Simpan Perubahan' : 'Tambah Mapel'}</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 🔹 PHOTO LIGHTBOX */}
      {lightboxPhoto && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/80 backdrop-blur-md">
          <div className="relative max-w-4xl w-full bg-white border border-[#E0E0D8] rounded-[32px] overflow-hidden shadow-2xl">
            <button
              onClick={() => setLightboxPhoto(null)}
              className="absolute top-4 right-4 p-2 rounded-full bg-black/60 text-white z-20 hover:bg-black/80 cursor-pointer"
            >
              <X className="w-6 h-6" />
            </button>
            <div className="aspect-[16/10] w-full bg-[#1A1A18]">
              <img src={lightboxPhoto.imageUrl} alt={lightboxPhoto.title} className="w-full h-full object-contain" />
            </div>
            <div className="p-6 bg-white border-t border-[#E0E0D8]">
              <div className="flex items-center gap-2 text-[#5A5A40] text-xs font-bold uppercase tracking-wider mb-1">
                <Calendar className="w-4 h-4" />
                <span>{lightboxPhoto.date}</span>
              </div>
              <h3 className="font-serif text-xl font-bold text-[#2D2D2A]">{lightboxPhoto.title}</h3>
              {lightboxPhoto.description && (
                <p className="text-sm text-[#8E8E85] mt-2">{lightboxPhoto.description}</p>
              )}
            </div>
          </div>
        </div>
      )}

      {/* 📱 QR CODE MODAL UNTUK KELAS INI */}
      <QRCodeModal
        isOpen={isQrModalOpen}
        onClose={() => setIsQrModalOpen(false)}
        data={{
          type: 'class',
          id: classRoom.id,
          title: `Ruang Kelas ${classRoom.name}`,
          subtitle: `Wali Kelas: ${classRoom.perangkat.waliKelas || '-'} • ${classRoom.grade} TA ${classRoom.academicYear}`,
          category: classRoom.jurusan || classRoom.grade
        }}
      />
    </div>
  );
};
