import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { UksMedicine, UksImmunization, UksMedicOfficer } from '../types';
import { 
  HeartHandshake, 
  Activity, 
  Pill, 
  Calendar, 
  ShieldCheck, 
  Plus, 
  Edit3, 
  Trash2, 
  AlertCircle, 
  CheckCircle, 
  Sparkles, 
  Phone, 
  User, 
  Heart,
  X
} from 'lucide-react';

export const UksView: React.FC = () => {
  const { 
    uksMedics, 
    uksMedicines, 
    addUksMedicine, 
    updateUksMedicine, 
    deleteUksMedicine, 
    uksImmunizations, 
    addUksImmunization, 
    updateUksImmunization, 
    deleteUksImmunization, 
    uksHealthTips,
    authSession,
    requestDeleteConfirm,
    hasAdminScope
  } = useApp();

  const [activeTab, setActiveTab] = useState<'medics' | 'medicines' | 'immunization' | 'tips'>('medics');

  // Medicine Modal
  const [isMedModalOpen, setIsMedModalOpen] = useState<boolean>(false);
  const [editingMed, setEditingMed] = useState<UksMedicine | null>(null);
  const [medForm, setMedForm] = useState({
    name: '',
    category: 'Pertolongan Pertama' as UksMedicine['category'],
    stock: 20,
    unit: 'Strip',
    expiryDate: new Date(Date.now() + 365*24*60*60*1000).toISOString().split('T')[0],
    indication: ''
  });

  // Immunization Modal
  const [isImmModalOpen, setIsImmModalOpen] = useState<boolean>(false);
  const [editingImm, setEditingImm] = useState<UksImmunization | null>(null);
  const [immForm, setImmForm] = useState({
    vaccineName: '',
    targetGrades: 'Kelas X, XI, XII',
    scheduledDate: new Date().toISOString().split('T')[0],
    healthCenterOfficer: 'Puskesmas Kecamatan',
    description: '',
    status: 'Terjadwal' as UksImmunization['status']
  });

  const canManage = hasAdminScope('uks') || authSession.role === 'super_admin';

  // Medicine Handlers
  const handleOpenAddMed = () => {
    setEditingMed(null);
    setMedForm({
      name: '',
      category: 'Pertolongan Pertama',
      stock: 20,
      unit: 'Strip',
      expiryDate: new Date(Date.now() + 365*24*60*60*1000).toISOString().split('T')[0],
      indication: ''
    });
    setIsMedModalOpen(true);
  };

  const handleOpenEditMed = (med: UksMedicine) => {
    setEditingMed(med);
    setMedForm({
      name: med.name || med.nama,
      category: (med.category || med.kategori) as any,
      stock: med.stock !== undefined ? med.stock : med.stok,
      unit: med.unit || med.satuan || 'Strip',
      expiryDate: med.expiryDate || new Date(Date.now() + 365*24*60*60*1000).toISOString().split('T')[0],
      indication: med.indication || med.fungsi || ''
    });
    setIsMedModalOpen(true);
  };

  const handleSaveMed = (e: React.FormEvent) => {
    e.preventDefault();
    if (!medForm.name.trim()) return;

    const medStock = Number(medForm.stock);

    if (editingMed) {
      updateUksMedicine(editingMed.id, {
        name: medForm.name.trim(),
        nama: medForm.name.trim(),
        category: medForm.category,
        kategori: medForm.category as any,
        stock: medStock,
        stok: medStock,
        unit: medForm.unit.trim(),
        satuan: medForm.unit.trim(),
        expiryDate: medForm.expiryDate,
        indication: medForm.indication.trim() || undefined,
        fungsi: medForm.indication.trim() || medForm.name.trim(),
        tersedia: medStock > 0
      });
    } else {
      addUksMedicine({
        name: medForm.name.trim(),
        nama: medForm.name.trim(),
        category: medForm.category,
        kategori: medForm.category as any,
        stock: medStock,
        stok: medStock,
        unit: medForm.unit.trim(),
        satuan: medForm.unit.trim(),
        expiryDate: medForm.expiryDate,
        indication: medForm.indication.trim() || undefined,
        fungsi: medForm.indication.trim() || medForm.name.trim(),
        tersedia: medStock > 0
      });
    }
    setIsMedModalOpen(false);
  };

  const handleDeleteMed = (med: UksMedicine) => {
    const medName = med.name || med.nama;
    requestDeleteConfirm({
      itemName: `Obat "${medName}"`,
      message: `Yakin ingin menghapus obat "${medName}" dari inventaris UKS?`,
      onConfirm: () => deleteUksMedicine(med.id)
    });
  };

  // Immunization Handlers
  const handleOpenAddImm = () => {
    setEditingImm(null);
    setImmForm({
      vaccineName: '',
      targetGrades: 'Kelas X, XI, XII',
      scheduledDate: new Date().toISOString().split('T')[0],
      healthCenterOfficer: 'Puskesmas Kecamatan',
      description: '',
      status: 'Terjadwal'
    });
    setIsImmModalOpen(true);
  };

  const handleOpenEditImm = (imm: UksImmunization) => {
    setEditingImm(imm);
    setImmForm({
      vaccineName: imm.vaccineName || imm.title,
      targetGrades: imm.targetGrades || imm.targetClass,
      scheduledDate: imm.scheduledDate || imm.date,
      healthCenterOfficer: imm.healthCenterOfficer || imm.petugas,
      description: imm.description || imm.deskripsi || '',
      status: (imm.status === 'Selesai' ? 'Selesai' : 'Terjadwal') as any
    });
    setIsImmModalOpen(true);
  };

  const handleSaveImm = (e: React.FormEvent) => {
    e.preventDefault();
    if (!immForm.vaccineName.trim()) return;

    if (editingImm) {
      updateUksImmunization(editingImm.id, {
        vaccineName: immForm.vaccineName.trim(),
        title: immForm.vaccineName.trim(),
        targetGrades: immForm.targetGrades.trim(),
        targetClass: immForm.targetGrades.trim(),
        scheduledDate: immForm.scheduledDate,
        date: immForm.scheduledDate,
        healthCenterOfficer: immForm.healthCenterOfficer.trim(),
        petugas: immForm.healthCenterOfficer.trim(),
        description: immForm.description.trim() || undefined,
        deskripsi: immForm.description.trim() || immForm.vaccineName.trim(),
        status: immForm.status === 'Selesai' ? 'Selesai' : 'Akan Datang'
      });
    } else {
      addUksImmunization({
        vaccineName: immForm.vaccineName.trim(),
        title: immForm.vaccineName.trim(),
        targetGrades: immForm.targetGrades.trim(),
        targetClass: immForm.targetGrades.trim(),
        scheduledDate: immForm.scheduledDate,
        date: immForm.scheduledDate,
        healthCenterOfficer: immForm.healthCenterOfficer.trim(),
        petugas: immForm.healthCenterOfficer.trim(),
        description: immForm.description.trim() || undefined,
        deskripsi: immForm.description.trim() || immForm.vaccineName.trim(),
        status: immForm.status === 'Selesai' ? 'Selesai' : 'Akan Datang'
      });
    }
    setIsImmModalOpen(false);
  };

  const handleDeleteImm = (imm: UksImmunization) => {
    const vName = imm.vaccineName || imm.title;
    requestDeleteConfirm({
      itemName: `Jadwal Vaksinasi "${vName}"`,
      message: `Yakin ingin menghapus jadwal imunisasi "${vName}"?`,
      onConfirm: () => deleteUksImmunization(imm.id)
    });
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#5A5A40] via-[#3E3E2C] to-[#2B2B1E] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <HeartHandshake className="w-4 h-4" />
            <span>Menu 7: Unit Kesehatan Sekolah (UKS)</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            UKS & Layanan Kesehatan Siswa
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Pusat pertolongan pertama, petugas medis sekolah, ketersediaan obat-obatan esensial, jadwal imunisasi berkala bekerja sama dengan Puskesmas, dan tips kesehatan harian.
          </p>

          {/* Sub Navigation Tabs */}
          <div className="pt-2 flex flex-wrap items-center gap-2 sm:gap-3">
            <button
              onClick={() => setActiveTab('medics')}
              className={`px-4 py-2 rounded-2xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
                activeTab === 'medics'
                  ? 'bg-amber-400 text-[#2D2D2A] shadow-md'
                  : 'bg-white/20 text-white hover:bg-white/30'
              }`}
            >
              👩‍⚕️ Petugas UKS
            </button>
            <button
              onClick={() => setActiveTab('medicines')}
              className={`px-4 py-2 rounded-2xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
                activeTab === 'medicines'
                  ? 'bg-amber-400 text-[#2D2D2A] shadow-md'
                  : 'bg-white/20 text-white hover:bg-white/30'
              }`}
            >
              💊 Stok Obat ({uksMedicines.length})
            </button>
            <button
              onClick={() => setActiveTab('immunization')}
              className={`px-4 py-2 rounded-2xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
                activeTab === 'immunization'
                  ? 'bg-amber-400 text-[#2D2D2A] shadow-md'
                  : 'bg-white/20 text-white hover:bg-white/30'
              }`}
            >
              💉 Jadwal Imunisasi
            </button>
            <button
              onClick={() => setActiveTab('tips')}
              className={`px-4 py-2 rounded-2xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
                activeTab === 'tips'
                  ? 'bg-amber-400 text-[#2D2D2A] shadow-md'
                  : 'bg-white/20 text-white hover:bg-white/30'
              }`}
            >
              🌿 Tips Sehat
            </button>
          </div>
        </div>
      </div>

      {/* Tab 1: Petugas UKS */}
      {activeTab === 'medics' && (
        <div className="space-y-6">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            {uksMedics.map(medic => (
              <div
                key={medic.id}
                className="bg-white dark:bg-[#1E1E1C] p-6 rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm flex items-start gap-4"
              >
                <img
                  src={medic.photoUrl || 'https://images.unsplash.com/photo-1559839734-2b71ea197ec2?w=300&auto=format&fit=crop&q=80'}
                  alt={medic.name}
                  className="w-20 h-20 rounded-2xl object-cover ring-2 ring-[#5A5A40]/20 shrink-0"
                />
                <div className="space-y-1.5 flex-1 min-w-0">
                  <div className="inline-flex items-center gap-1 px-2.5 py-0.5 bg-emerald-100 dark:bg-emerald-950/50 text-emerald-800 dark:text-emerald-300 text-[10px] font-bold rounded-full">
                    <ShieldCheck className="w-3 h-3" />
                    <span>{medic.role}</span>
                  </div>
                  <h3 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA] truncate">
                    {medic.name}
                  </h3>
                  <p className="text-xs text-[#8E8E85]">
                    Jadwal Jaga: <strong className="text-[#5A5A40] dark:text-[#A4A485]">{medic.dutySchedule}</strong>
                  </p>
                  {medic.contact && (
                    <div className="flex items-center gap-1.5 text-xs text-[#8C4A32] dark:text-[#E07A5F] font-semibold pt-1">
                      <Phone className="w-3.5 h-3.5" />
                      <span>{medic.contact}</span>
                    </div>
                  )}
                </div>
              </div>
            ))}
          </div>

          <div className="bg-amber-50 dark:bg-amber-950/20 border border-amber-200 dark:border-amber-900/40 p-5 rounded-2xl flex items-center gap-4">
            <AlertCircle className="w-8 h-8 text-amber-600 dark:text-amber-400 shrink-0" />
            <div className="text-xs text-amber-900 dark:text-amber-200">
              <strong className="font-bold">Protokol Darurat Medis:</strong> Jika terjadi kecelakaan atau siswa mendadak pingsan/sakit parah, segera laporkan ke ruang UKS atau hubungi nomor darurat piket UKS.
            </div>
          </div>
        </div>
      )}

      {/* Tab 2: Stok Obat UKS */}
      {activeTab === 'medicines' && (
        <div className="space-y-6">
          <div className="flex items-center justify-between">
            <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
              Inventaris Ketersediaan Obat
            </h3>
            {canManage && (
              <button
                onClick={handleOpenAddMed}
                className="inline-flex items-center gap-2 px-4 py-2 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold rounded-xl text-xs shadow-sm cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Tambah Stok Obat</span>
              </button>
            )}
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
            {uksMedicines.map(med => {
              const isLowStock = med.stock <= 5;
              return (
                <div
                  key={med.id}
                  className="bg-white dark:bg-[#1E1E1C] p-5 rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs flex flex-col justify-between"
                >
                  <div className="space-y-2">
                    <div className="flex items-center justify-between">
                      <span className="px-2 py-0.5 bg-[#5A5A40]/10 dark:bg-[#5A5A40]/30 text-[#5A5A40] dark:text-[#D1D1BE] text-[10px] font-bold rounded-md">
                        {med.category}
                      </span>
                      <span className={`px-2 py-0.5 text-[10px] font-bold rounded-full ${
                        isLowStock
                          ? 'bg-rose-100 dark:bg-rose-950/50 text-rose-700 dark:text-rose-300 animate-pulse'
                          : 'bg-emerald-100 dark:bg-emerald-950/50 text-emerald-800 dark:text-emerald-300'
                      }`}>
                        {isLowStock ? 'Stok Menipis' : 'Tersedia'}
                      </span>
                    </div>

                    <h4 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                      {med.name}
                    </h4>

                    {med.indication && (
                      <p className="text-xs text-[#8E8E85] line-clamp-2">
                        Indikasi: {med.indication}
                      </p>
                    )}

                    <div className="pt-2 flex items-center justify-between text-xs font-semibold">
                      <span className="text-[#5A5A40] dark:text-[#A4A485]">
                        Sisa: <strong>{med.stock}</strong> {med.unit}
                      </span>
                      <span className="text-[11px] text-[#8E8E85]">
                        Kadaluarsa: {med.expiryDate}
                      </span>
                    </div>
                  </div>

                  {canManage && (
                    <div className="pt-3 mt-3 border-t border-[#E0E0D8]/60 dark:border-[#383832]/60 flex items-center justify-end gap-2">
                      <button
                        onClick={() => handleOpenEditMed(med)}
                        className="p-1 text-xs text-[#5A5A40] dark:text-[#D1D1BE] hover:underline font-semibold cursor-pointer"
                      >
                        Edit
                      </button>
                      <button
                        onClick={() => handleDeleteMed(med)}
                        className="p-1 text-xs text-rose-600 hover:underline font-semibold cursor-pointer"
                      >
                        Hapus
                      </button>
                    </div>
                  )}
                </div>
              );
            })}
          </div>
        </div>
      )}

      {/* Tab 3: Jadwal Imunisasi */}
      {activeTab === 'immunization' && (
        <div className="space-y-6">
          <div className="flex items-center justify-between">
            <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
              Jadwal Vaksinasi & Skrining Berkala
            </h3>
            {canManage && (
              <button
                onClick={handleOpenAddImm}
                className="inline-flex items-center gap-2 px-4 py-2 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold rounded-xl text-xs shadow-sm cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Tambah Jadwal Imunisasi</span>
              </button>
            )}
          </div>

          <div className="space-y-4">
            {uksImmunizations.map(imm => (
              <div
                key={imm.id}
                className="bg-white dark:bg-[#1E1E1C] p-5 rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs flex flex-col sm:flex-row sm:items-center justify-between gap-4"
              >
                <div className="space-y-1.5">
                  <div className="flex items-center gap-2">
                    <span className={`px-2.5 py-0.5 rounded-full text-[10px] font-bold ${
                      imm.status === 'Selesai'
                        ? 'bg-emerald-100 text-emerald-800'
                        : 'bg-amber-100 text-amber-800'
                    }`}>
                      {imm.status}
                    </span>
                    <span className="text-xs font-bold text-[#5A5A40] dark:text-[#A4A485]">
                      Sasaran: {imm.targetGrades}
                    </span>
                  </div>

                  <h4 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                    {imm.vaccineName}
                  </h4>

                  <div className="flex flex-wrap items-center gap-3 text-xs text-[#8E8E85]">
                    <div className="flex items-center gap-1">
                      <Calendar className="w-3.5 h-3.5 text-[#5A5A40]" />
                      <span>{imm.scheduledDate}</span>
                    </div>
                    <span>•</span>
                    <span>Pelaksana: <strong>{imm.healthCenterOfficer}</strong></span>
                  </div>

                  {imm.description && (
                    <p className="text-xs text-[#5A5A40] dark:text-[#A4A485] pt-1">
                      {imm.description}
                    </p>
                  )}
                </div>

                {canManage && (
                  <div className="flex items-center gap-2 shrink-0 self-end sm:self-center">
                    <button
                      onClick={() => handleOpenEditImm(imm)}
                      className="p-1.5 text-xs text-[#5A5A40] dark:text-[#D1D1BE] hover:underline font-semibold cursor-pointer"
                    >
                      Edit
                    </button>
                    <button
                      onClick={() => handleDeleteImm(imm)}
                      className="p-1.5 text-xs text-rose-600 hover:underline font-semibold cursor-pointer"
                    >
                      Hapus
                    </button>
                  </div>
                )}
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Tab 4: Tips Kesehatan */}
      {activeTab === 'tips' && (
        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          {uksHealthTips.map(tip => (
            <div
              key={tip.id}
              className="bg-white dark:bg-[#1E1E1C] p-6 rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm space-y-3"
            >
              <div className="flex items-center gap-2">
                <span className="px-2.5 py-0.5 bg-emerald-100 dark:bg-emerald-950/50 text-emerald-800 dark:text-emerald-300 text-[10px] font-bold rounded-full">
                  {tip.category}
                </span>
                <span className="text-[11px] text-[#8E8E85]">{tip.date}</span>
              </div>
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                {tip.title}
              </h3>
              <p className="text-xs sm:text-sm text-[#5A5A40] dark:text-[#C5C5B0] leading-relaxed">
                {tip.content}
              </p>
            </div>
          ))}
        </div>
      )}

      {/* Medicine Modal */}
      {isMedModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-md rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                {editingMed ? 'Edit Data Obat' : 'Tambah Obat Baru'}
              </h3>
              <button onClick={() => setIsMedModalOpen(false)} className="p-2 text-[#8E8E85] cursor-pointer">
                <X className="w-5 h-5" />
              </button>
            </div>
            <form onSubmit={handleSaveMed} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Nama Obat *</label>
                <input
                  type="text"
                  required
                  value={medForm.name}
                  onChange={e => setMedForm({ ...medForm, name: e.target.value })}
                  placeholder="Contoh: Paracetamol 500mg"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Kategori *</label>
                  <select
                    value={medForm.category}
                    onChange={e => setMedForm({ ...medForm, category: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-semibold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    <option value="Pertolongan Pertama">Pertolongan Pertama</option>
                    <option value="Demam & Flu">Demam & Flu</option>
                    <option value="Lambung & Pencernaan">Lambung & Pencernaan</option>
                    <option value="Luka Luar">Luka Luar</option>
                    <option value="Minyak & Balsem">Minyak & Balsem</option>
                  </select>
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Jumlah Stok *</label>
                  <input
                    type="number"
                    min={0}
                    required
                    value={medForm.stock}
                    onChange={e => setMedForm({ ...medForm, stock: Number(e.target.value) })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Satuan</label>
                  <input
                    type="text"
                    value={medForm.unit}
                    onChange={e => setMedForm({ ...medForm, unit: e.target.value })}
                    placeholder="Strip / Botol / Tablet"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Tanggal Expired</label>
                  <input
                    type="date"
                    value={medForm.expiryDate}
                    onChange={e => setMedForm({ ...medForm, expiryDate: e.target.value })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-semibold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Indikasi / Khasiat</label>
                <input
                  type="text"
                  value={medForm.indication}
                  onChange={e => setMedForm({ ...medForm, indication: e.target.value })}
                  placeholder="Meredakan pusing, sakit kepala, demam..."
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex justify-end gap-2">
                <button type="button" onClick={() => setIsMedModalOpen(false)} className="px-4 py-2 text-xs font-bold text-[#8E8E85]">Batal</button>
                <button type="submit" className="px-5 py-2 bg-[#5A5A40] text-white text-xs font-bold rounded-xl">Simpan</button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Immunization Modal */}
      {isImmModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-md rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                {editingImm ? 'Edit Imunisasi' : 'Tambah Jadwal Imunisasi'}
              </h3>
              <button onClick={() => setIsImmModalOpen(false)} className="p-2 text-[#8E8E85] cursor-pointer">
                <X className="w-5 h-5" />
              </button>
            </div>
            <form onSubmit={handleSaveImm} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Nama Vaksinasi / Skrining *</label>
                <input
                  type="text"
                  required
                  value={immForm.vaccineName}
                  onChange={e => setImmForm({ ...immForm, vaccineName: e.target.value })}
                  placeholder="Contoh: Skrining HB & Anemia Remaja Putri"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Sasaran Kelas *</label>
                  <input
                    type="text"
                    required
                    value={immForm.targetGrades}
                    onChange={e => setImmForm({ ...immForm, targetGrades: e.target.value })}
                    placeholder="Kelas X & XI"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Tanggal *</label>
                  <input
                    type="date"
                    required
                    value={immForm.scheduledDate}
                    onChange={e => setImmForm({ ...immForm, scheduledDate: e.target.value })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-semibold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Puskesmas / Petugas Pelaksana</label>
                <input
                  type="text"
                  value={immForm.healthCenterOfficer}
                  onChange={e => setImmForm({ ...immForm, healthCenterOfficer: e.target.value })}
                  placeholder="Puskesmas Kecamatan"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex justify-end gap-2">
                <button type="button" onClick={() => setIsImmModalOpen(false)} className="px-4 py-2 text-xs font-bold text-[#8E8E85]">Batal</button>
                <button type="submit" className="px-5 py-2 bg-[#5A5A40] text-white text-xs font-bold rounded-xl">Simpan</button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
