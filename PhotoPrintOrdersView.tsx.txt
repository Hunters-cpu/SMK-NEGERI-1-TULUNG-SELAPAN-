import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { PhotoPrintOrder } from '../types';
import { 
  Printer, 
  ShoppingBag, 
  Image as ImageIcon, 
  CheckCircle, 
  Clock, 
  Send, 
  Trash2, 
  Filter, 
  Search,
  Sparkles,
  Phone,
  User,
  X
} from 'lucide-react';
import { MediaUploadField } from '../components/MediaUploadField';

export const PhotoPrintOrdersView: React.FC = () => {
  const { 
    photoPrintOrders, 
    addPhotoPrintOrder, 
    updatePhotoPrintOrderStatus, 
    deletePhotoPrintOrder, 
    authSession,
    requestDeleteConfirm,
    hasAdminScope,
    showToast
  } = useApp();

  const [form, setForm] = useState({
    studentName: authSession.name || '',
    className: '',
    whatsappNumber: '',
    photoSourceUrl: '',
    printSize: '4R (10x15 cm)' as PhotoPrintOrder['printSize'],
    quantity: 1,
    deliveryNotes: ''
  });

  const canManage = hasAdminScope('photo_orders') || authSession.role === 'super_admin';

  const priceMap: Record<string, number> = {
    '4R (10x15 cm)': 5000,
    '10R (20x25 cm)': 20000,
    'A4 Poster': 35000,
    'Album Mini Kenangan': 75000
  };

  const estimatedTotal = (priceMap[form.printSize] || 5000) * form.quantity;

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.studentName.trim() || !form.className.trim() || !form.photoSourceUrl.trim() || !form.whatsappNumber.trim()) return;

    addPhotoPrintOrder({
      studentName: form.studentName.trim(),
      className: form.className.trim(),
      whatsappNumber: form.whatsappNumber.trim(),
      photoSourceUrl: form.photoSourceUrl.trim(),
      printSize: form.printSize,
      quantity: Number(form.quantity),
      deliveryNotes: form.deliveryNotes.trim() || undefined,
      totalPrice: estimatedTotal
    });

    setForm({
      studentName: authSession.name || '',
      className: '',
      whatsappNumber: '',
      photoSourceUrl: '',
      printSize: '4R (10x15 cm)',
      quantity: 1,
      deliveryNotes: ''
    });

    showToast('Pesanan cetak foto Anda telah berhasil dikirim! Silakan tunggu konfirmasi melalui WhatsApp.', 'success');
  };

  const handleDelete = (order: PhotoPrintOrder) => {
    requestDeleteConfirm({
      itemName: `Pesanan foto "${order.studentName}"`,
      message: `Hapus data pesanan cetak foto ini?`,
      onConfirm: () => deletePhotoPrintOrder(order.id)
    });
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#8C4A32] via-[#723B28] to-[#452115] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <Printer className="w-4 h-4" />
            <span>Menu 12: Layanan Percetakan Kenangan</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            Pesan Cetak Foto Fisik Sekolah
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Abadikan momen terindah di sekolah dalam bentuk cetak foto berkualitas studio profesional (kertas foto glossy anti luntur, poster A4, atau mini album kenangan).
          </p>
        </div>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
        {/* Order Form */}
        <div className="bg-white dark:bg-[#1E1E1C] p-6 sm:p-8 rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm space-y-5 h-fit">
          <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA] pb-3 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center gap-2">
            <ShoppingBag className="w-5 h-5 text-[#8C4A32] dark:text-[#E07A5F]" />
            <span>Formulir Pesanan Cetak</span>
          </h3>

          <form onSubmit={handleSubmit} className="space-y-4">
            <div>
              <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                Nama Pemesan (Siswa / Wali / Guru) *
              </label>
              <input
                type="text"
                required
                value={form.studentName}
                onChange={e => setForm({ ...form, studentName: e.target.value })}
                placeholder="Nama Lengkap"
                className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
              />
            </div>

            <div className="grid grid-cols-2 gap-3">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Kelas / Jabatan *
                </label>
                <input
                  type="text"
                  required
                  value={form.className}
                  onChange={e => setForm({ ...form, className: e.target.value })}
                  placeholder="Contoh: XII MIPA 1"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  No. WhatsApp *
                </label>
                <input
                  type="tel"
                  required
                  value={form.whatsappNumber}
                  onChange={e => setForm({ ...form, whatsappNumber: e.target.value })}
                  placeholder="0812-xxxx-xxxx"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>
            </div>

            <MediaUploadField
              label="Foto yang Ingin Dicetak (URL / Upload) *"
              value={form.photoSourceUrl}
              onChange={url => setForm({ ...form, photoSourceUrl: url })}
              placeholder="https://..."
            />

            <div className="grid grid-cols-2 gap-3">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Ukuran Cetak *
                </label>
                <select
                  value={form.printSize}
                  onChange={e => setForm({ ...form, printSize: e.target.value as any })}
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-bold text-[#2D2D2A] dark:text-[#EAEAE2]"
                >
                  <option value="4R (10x15 cm)">4R (10x15 cm) - Rp 5.000</option>
                  <option value="10R (20x25 cm)">10R (20x25 cm) - Rp 20.000</option>
                  <option value="A4 Poster">A4 Poster - Rp 35.000</option>
                  <option value="Album Mini Kenangan">Album Mini - Rp 75.000</option>
                </select>
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Jumlah Lembar *
                </label>
                <input
                  type="number"
                  min={1}
                  max={100}
                  required
                  value={form.quantity}
                  onChange={e => setForm({ ...form, quantity: Number(e.target.value) })}
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>
            </div>

            <div className="p-3 bg-amber-50 dark:bg-amber-950/30 rounded-xl border border-amber-200 dark:border-amber-900/40 flex items-center justify-between text-xs">
              <span className="font-semibold text-amber-900 dark:text-amber-200">Total Biaya Cetak:</span>
              <span className="font-extrabold text-sm text-[#8C4A32] dark:text-[#E07A5F]">
                Rp {estimatedTotal.toLocaleString('id-ID')}
              </span>
            </div>

            <div>
              <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                Catatan Pengambilan / Antar Kelas (Opsional)
              </label>
              <textarea
                rows={2}
                value={form.deliveryNotes}
                onChange={e => setForm({ ...form, deliveryNotes: e.target.value })}
                placeholder="Contoh: Titip di meja piket guru atau antar ke ruang kelas..."
                className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
              />
            </div>

            <button
              type="submit"
              className="w-full py-3 bg-[#8C4A32] hover:bg-[#723B28] text-white font-bold rounded-xl text-xs shadow-md transition-all flex items-center justify-center gap-2 cursor-pointer active:scale-95"
            >
              <Printer className="w-4 h-4" />
              <span>Kirim Pesanan Cetak</span>
            </button>
          </form>
        </div>

        {/* Right 2 Columns: Order History / Tracking */}
        <div className="lg:col-span-2 space-y-4">
          <div className="flex items-center justify-between">
            <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
              Status Pesanan Cetak Foto ({photoPrintOrders.length})
            </h3>
          </div>

          {photoPrintOrders.length === 0 ? (
            <div className="text-center py-16 bg-white dark:bg-[#1E1E1C] rounded-3xl border border-[#E0E0D8] dark:border-[#383832] p-8">
              <Printer className="w-12 h-12 text-[#8E8E85] mx-auto mb-3 opacity-40" />
              <p className="text-base font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">Belum ada antrean pesanan cetak</p>
              <p className="text-xs text-[#8E8E85] mt-1">Gunakan formulir untuk memesan cetak foto favorit Anda.</p>
            </div>
          ) : (
            <div className="space-y-4">
              {photoPrintOrders.map(order => (
                <div
                  key={order.id}
                  className="bg-white dark:bg-[#1E1E1C] p-5 rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs flex flex-col sm:flex-row gap-4 items-start"
                >
                  <img
                    src={order.photoSourceUrl}
                    alt="Pratinjau Cetak"
                    className="w-24 h-24 rounded-2xl object-cover ring-2 ring-[#5A5A40]/20 shrink-0"
                  />

                  <div className="flex-1 min-w-0 space-y-1.5">
                    <div className="flex items-center gap-2">
                      <span className={`px-2.5 py-0.5 rounded-full text-[10px] font-bold ${
                        order.status === 'Selesai'
                          ? 'bg-emerald-100 text-emerald-800'
                          : order.status === 'Diproses'
                            ? 'bg-blue-100 text-blue-800'
                            : 'bg-amber-100 text-amber-800'
                      }`}>
                        {order.status}
                      </span>
                      <span className="text-[11px] text-[#8E8E85]">{order.orderDate}</span>
                    </div>

                    <h4 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                      {order.studentName} ({order.className})
                    </h4>

                    <p className="text-xs text-[#5A5A40] dark:text-[#A4A485]">
                      Ukuran: <strong>{order.printSize}</strong> • Jumlah: <strong>{order.quantity} lembar</strong> • Total: <strong className="text-[#8C4A32] dark:text-[#E07A5F]">Rp {order.totalPrice?.toLocaleString('id-ID')}</strong>
                    </p>

                    {order.deliveryNotes && (
                      <p className="text-[11px] text-[#8E8E85] italic">
                        Catatan: "{order.deliveryNotes}"
                      </p>
                    )}

                    {canManage && (
                      <div className="pt-2 flex flex-wrap items-center gap-2">
                        <span className="text-[11px] font-bold text-[#8E8E85]">Ubah Status:</span>
                        {(['Menunggu', 'Diproses', 'Selesai', 'Dibatalkan'] as const).map(st => (
                          <button
                            key={st}
                            onClick={() => updatePhotoPrintOrderStatus(order.id, st)}
                            className={`px-2 py-0.5 text-[10px] font-bold rounded-md transition-colors cursor-pointer ${
                              order.status === st
                                ? 'bg-[#5A5A40] text-white'
                                : 'bg-[#F5F5F0] dark:bg-[#282825] text-[#8E8E85] hover:text-[#2D2D2A]'
                            }`}
                          >
                            {st}
                          </button>
                        ))}
                        <button
                          onClick={() => handleDelete(order)}
                          className="text-rose-600 hover:underline text-xs ml-auto cursor-pointer"
                        >
                          Hapus
                        </button>
                      </div>
                    )}
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  );
};
