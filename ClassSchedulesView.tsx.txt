import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { ClassScheduleItem, BellScheduleGroup } from '../types';
import { 
  BookOpen, 
  Clock, 
  Bell, 
  Search, 
  Plus, 
  Edit3, 
  Trash2, 
  Calendar, 
  User, 
  Layers, 
  CheckCircle,
  X
} from 'lucide-react';

export const ClassSchedulesView: React.FC = () => {
  const { 
    classes, 
    classSchedules, 
    addClassSchedule, 
    updateClassSchedule, 
    deleteClassSchedule, 
    bellSchedules, 
    updateBellSchedule,
    authSession,
    requestDeleteConfirm,
    hasAdminScope
  } = useApp();

  const [activeTab, setActiveTab] = useState<'schedule' | 'bell'>('schedule');
  const [selectedClassId, setSelectedClassId] = useState<string>(classes[0]?.id || '');
  const [selectedDay, setSelectedDay] = useState<string>('Semua');

  // Schedule Modal State
  const [isScheduleModalOpen, setIsScheduleModalOpen] = useState<boolean>(false);
  const [editingSchedule, setEditingSchedule] = useState<ClassScheduleItem | null>(null);
  const [scheduleForm, setScheduleForm] = useState({
    classId: classes[0]?.id || '',
    day: 'Senin' as ClassScheduleItem['day'],
    timeSlot: '07:30 - 09:00',
    subject: '',
    teacherName: '',
    room: ''
  });

  const canManageSchedule = 
    authSession.role === 'super_admin' || 
    hasAdminScope('class_schedules') ||
    (authSession.role === 'class_admin' && authSession.classId === selectedClassId);

  const canManageBell = 
    authSession.role === 'super_admin' || 
    hasAdminScope('bell_schedules');

  const selectedClass = classes.find(c => c.id === selectedClassId);

  // Filtered schedules
  const filteredSchedules = classSchedules.filter(item => {
    const matchClass = item.classId === selectedClassId;
    const matchDay = selectedDay === 'Semua' || item.day === selectedDay;
    return matchClass && matchDay;
  });

  const daysOrder: Record<string, number> = {
    'Senin': 1,
    'Selasa': 2,
    'Rabu': 3,
    'Kamis': 4,
    'Jumat': 5,
    'Sabtu': 6
  };

  const sortedSchedules = [...filteredSchedules].sort((a, b) => {
    const dayDiff = (daysOrder[a.day] || 99) - (daysOrder[b.day] || 99);
    if (dayDiff !== 0) return dayDiff;
    return a.timeSlot.localeCompare(b.timeSlot);
  });

  const handleOpenAddSchedule = () => {
    setEditingSchedule(null);
    setScheduleForm({
      classId: selectedClassId || classes[0]?.id || '',
      day: selectedDay !== 'Semua' ? (selectedDay as any) : 'Senin',
      timeSlot: '07:30 - 09:00',
      subject: '',
      teacherName: '',
      room: selectedClass?.name || ''
    });
    setIsScheduleModalOpen(true);
  };

  const handleOpenEditSchedule = (item: ClassScheduleItem) => {
    setEditingSchedule(item);
    setScheduleForm({
      classId: item.classId,
      day: item.day,
      timeSlot: item.timeSlot,
      subject: item.subject,
      teacherName: item.teacherName,
      room: item.room || ''
    });
    setIsScheduleModalOpen(true);
  };

  const handleSaveSchedule = (e: React.FormEvent) => {
    e.preventDefault();
    if (!scheduleForm.subject.trim() || !scheduleForm.teacherName.trim()) return;

    if (editingSchedule) {
      updateClassSchedule(editingSchedule.id, {
        classId: scheduleForm.classId,
        day: scheduleForm.day,
        timeSlot: scheduleForm.timeSlot.trim(),
        subject: scheduleForm.subject.trim(),
        teacherName: scheduleForm.teacherName.trim(),
        room: scheduleForm.room.trim() || undefined
      });
    } else {
      addClassSchedule({
        classId: scheduleForm.classId,
        day: scheduleForm.day,
        timeSlot: scheduleForm.timeSlot.trim(),
        subject: scheduleForm.subject.trim(),
        teacherName: scheduleForm.teacherName.trim(),
        room: scheduleForm.room.trim() || undefined
      });
    }

    setIsScheduleModalOpen(false);
  };

  const handleDeleteSchedule = (item: ClassScheduleItem) => {
    requestDeleteConfirm({
      itemName: `Mata Pelajaran "${item.subject}" (${item.day})`,
      message: `Yakin ingin menghapus jadwal pelajaran ${item.subject}?`,
      onConfirm: () => deleteClassSchedule(item.id)
    });
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#5A5A40] to-[#3E3E2C] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <BookOpen className="w-4 h-4" />
            <span>Menu 4 & 5: Jadwal Pelajaran & Jam Masuk / Pulang</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            Jadwal Pembelajaran & Bel Sekolah
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Lihat susunan jadwal mata pelajaran harian setiap kelas beserta guru pengampu dan rincian jam masuk, istirahat, hingga kepulangan siswa.
          </p>

          {/* Sub Navigation Tabs */}
          <div className="pt-2 flex items-center gap-3">
            <button
              onClick={() => setActiveTab('schedule')}
              className={`px-4 py-2 rounded-2xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
                activeTab === 'schedule'
                  ? 'bg-amber-400 text-[#2D2D2A] shadow-md'
                  : 'bg-white/20 text-white hover:bg-white/30'
              }`}
            >
              📚 Jadwal Pelajaran Kelas
            </button>
            <button
              onClick={() => setActiveTab('bell')}
              className={`px-4 py-2 rounded-2xl text-xs sm:text-sm font-bold transition-all cursor-pointer ${
                activeTab === 'bell'
                  ? 'bg-amber-400 text-[#2D2D2A] shadow-md'
                  : 'bg-white/20 text-white hover:bg-white/30'
              }`}
            >
              ⏰ Jam Masuk & Bel Sekolah
            </button>
          </div>
        </div>
      </div>

      {activeTab === 'schedule' ? (
        <div className="space-y-6">
          {/* Class Selector & Day Filter */}
          <div className="flex flex-col md:flex-row items-center justify-between gap-4 bg-white dark:bg-[#1E1E1C] p-4 rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs">
            <div className="flex items-center gap-3 w-full md:w-auto">
              <span className="text-xs font-bold text-[#5A5A40] dark:text-[#A4A485] shrink-0">Pilih Kelas:</span>
              <select
                value={selectedClassId}
                onChange={e => setSelectedClassId(e.target.value)}
                className="px-4 py-2 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm font-bold text-[#2D2D2A] dark:text-[#EAEAE2] focus:outline-none"
              >
                {classes.map(c => (
                  <option key={c.id} value={c.id}>
                    Kelas {c.name} ({c.grade} {c.major})
                  </option>
                ))}
              </select>
            </div>

            <div className="flex items-center gap-1.5 overflow-x-auto pb-1 md:pb-0 w-full md:w-auto">
              {['Semua', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'].map(d => (
                <button
                  key={d}
                  onClick={() => setSelectedDay(d)}
                  className={`px-3 py-1.5 rounded-xl text-xs font-bold transition-all cursor-pointer shrink-0 ${
                    selectedDay === d
                      ? 'bg-[#5A5A40] text-white shadow-xs'
                      : 'bg-[#F5F5F0] dark:bg-[#282825] text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-white'
                  }`}
                >
                  {d}
                </button>
              ))}
            </div>

            {canManageSchedule && (
              <button
                onClick={handleOpenAddSchedule}
                className="w-full md:w-auto inline-flex items-center justify-center gap-2 px-4 py-2 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold rounded-xl text-xs shadow-sm cursor-pointer shrink-0"
              >
                <Plus className="w-4 h-4" />
                <span>Tambah Jam Pelajaran</span>
              </button>
            )}
          </div>

          {/* Schedule List / Grid */}
          {sortedSchedules.length === 0 ? (
            <div className="text-center py-16 bg-white dark:bg-[#1E1E1C] rounded-3xl border border-[#E0E0D8] dark:border-[#383832] p-8">
              <BookOpen className="w-12 h-12 text-[#8E8E85] mx-auto mb-3 opacity-40" />
              <p className="text-base font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                Belum ada jadwal pelajaran untuk kelas ini pada hari yang dipilih
              </p>
              <p className="text-xs text-[#8E8E85] mt-1">
                Pilih hari lain atau tambahkan jadwal mata pelajaran baru.
              </p>
            </div>
          ) : (
            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
              {sortedSchedules.map(item => (
                <div
                  key={item.id}
                  className="p-5 bg-white dark:bg-[#1E1E1C] rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs hover:shadow-md transition-all flex flex-col justify-between"
                >
                  <div className="space-y-2.5">
                    <div className="flex items-center justify-between">
                      <span className="px-2.5 py-0.5 rounded-md text-[10px] font-bold bg-[#5A5A40]/10 dark:bg-[#5A5A40]/30 text-[#5A5A40] dark:text-[#D1D1BE]">
                        {item.day}
                      </span>
                      <div className="flex items-center gap-1 text-xs font-semibold text-[#8E8E85]">
                        <Clock className="w-3.5 h-3.5 text-[#5A5A40] dark:text-[#A4A485]" />
                        <span>{item.timeSlot}</span>
                      </div>
                    </div>

                    <div>
                      <h3 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                        {item.subject}
                      </h3>
                      <div className="flex items-center gap-1.5 text-xs text-[#8C4A32] dark:text-[#E07A5F] font-semibold mt-1">
                        <User className="w-3.5 h-3.5" />
                        <span>Guru: {item.teacherName}</span>
                      </div>
                      {item.room && (
                        <p className="text-[11px] text-[#8E8E85] mt-0.5">
                          Ruang: {item.room}
                        </p>
                      )}
                    </div>
                  </div>

                  {canManageSchedule && (
                    <div className="pt-3 mt-3 border-t border-[#E0E0D8]/60 dark:border-[#383832]/60 flex items-center justify-end gap-2">
                      <button
                        onClick={() => handleOpenEditSchedule(item)}
                        className="p-1 text-xs text-[#5A5A40] dark:text-[#D1D1BE] hover:bg-[#F5F5F0] dark:hover:bg-[#282825] rounded-lg transition-colors cursor-pointer flex items-center gap-1 font-semibold"
                      >
                        <Edit3 className="w-3.5 h-3.5" />
                        <span>Edit</span>
                      </button>
                      <button
                        onClick={() => handleDeleteSchedule(item)}
                        className="p-1 text-xs text-rose-600 hover:bg-rose-50 dark:hover:bg-rose-950/30 rounded-lg transition-colors cursor-pointer flex items-center gap-1 font-semibold"
                      >
                        <Trash2 className="w-3.5 h-3.5" />
                        <span>Hapus</span>
                      </button>
                    </div>
                  )}
                </div>
              ))}
            </div>
          )}
        </div>
      ) : (
        /* Bell Schedules View */
        <div className="space-y-6">
          <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
            {(bellSchedules || []).map(group => (
              <div
                key={group.id}
                className="bg-white dark:bg-[#1E1E1C] p-6 rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm space-y-4"
              >
                <div className="flex items-center justify-between pb-3 border-b border-[#E0E0D8] dark:border-[#383832]">
                  <div className="flex items-center gap-3">
                    <div className="w-10 h-10 rounded-2xl bg-[#5A5A40] text-white flex items-center justify-center font-bold">
                      <Bell className="w-5 h-5" />
                    </div>
                    <div>
                      <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                        {group.dayGroup}
                      </h3>
                      <p className="text-xs text-[#8E8E85]">Rincian waktu belajar & lonceng sekolah</p>
                    </div>
                  </div>
                </div>

                <div className="space-y-2.5">
                  {(group.slots || []).map((slot, idx) => (
                    <div
                      key={idx}
                      className={`p-3 rounded-xl flex items-center justify-between text-xs font-semibold ${
                        slot.isBreak
                          ? 'bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-900/40 text-amber-900 dark:text-amber-200'
                          : 'bg-[#F8F8F5] dark:bg-[#282825] border border-[#E0E0D8]/60 dark:border-[#383832]/60 text-[#2D2D2A] dark:text-[#EAEAE2]'
                      }`}
                    >
                      <div className="flex items-center gap-2">
                        <span className="w-5 h-5 rounded-full bg-[#5A5A40]/10 dark:bg-[#5A5A40]/30 text-[#5A5A40] dark:text-[#D1D1BE] flex items-center justify-center text-[10px] font-bold">
                          {idx + 1}
                        </span>
                        <span>{slot.activityName}</span>
                      </div>
                      <span className="font-mono text-xs font-bold text-[#8C4A32] dark:text-[#E07A5F]">
                        {slot.startTime} - {slot.endTime}
                      </span>
                    </div>
                  ))}
                </div>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Add / Edit Schedule Modal */}
      {isScheduleModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-lg rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <div className="flex items-center gap-2">
                <div className="w-9 h-9 rounded-xl bg-[#5A5A40] text-white flex items-center justify-center font-bold">
                  <BookOpen className="w-5 h-5" />
                </div>
                <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                  {editingSchedule ? 'Edit Jam Pelajaran' : 'Tambah Jam Pelajaran'}
                </h3>
              </div>
              <button
                onClick={() => setIsScheduleModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-white rounded-lg cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveSchedule} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Mata Pelajaran *
                </label>
                <input
                  type="text"
                  required
                  value={scheduleForm.subject}
                  onChange={e => setScheduleForm({ ...scheduleForm, subject: e.target.value })}
                  placeholder="Contoh: Matematika Peminatan / Bahasa Indonesia"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Guru Pengampu *
                </label>
                <input
                  type="text"
                  required
                  value={scheduleForm.teacherName}
                  onChange={e => setScheduleForm({ ...scheduleForm, teacherName: e.target.value })}
                  placeholder="Contoh: Drs. Bambang Sutrisno, M.Pd."
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Hari *
                  </label>
                  <select
                    value={scheduleForm.day}
                    onChange={e => setScheduleForm({ ...scheduleForm, day: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-semibold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    <option value="Senin">Senin</option>
                    <option value="Selasa">Selasa</option>
                    <option value="Rabu">Rabu</option>
                    <option value="Kamis">Kamis</option>
                    <option value="Jumat">Jumat</option>
                    <option value="Sabtu">Sabtu</option>
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Jam (Waktu) *
                  </label>
                  <input
                    type="text"
                    required
                    value={scheduleForm.timeSlot}
                    onChange={e => setScheduleForm({ ...scheduleForm, timeSlot: e.target.value })}
                    placeholder="Contoh: 07:30 - 09:00"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Ruang Belajar / Lab (Opsional)
                </label>
                <input
                  type="text"
                  value={scheduleForm.room}
                  onChange={e => setScheduleForm({ ...scheduleForm, room: e.target.value })}
                  placeholder="Contoh: Ruang Kelas / Lab Komputer 1"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex items-center justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsScheduleModalOpen(false)}
                  className="px-4 py-2.5 text-xs font-bold text-[#8E8E85] hover:bg-[#F5F5F0] dark:hover:bg-[#282825] rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 bg-[#5A5A40] hover:bg-[#484833] text-white text-xs font-bold rounded-xl shadow-md cursor-pointer"
                >
                  {editingSchedule ? 'Simpan Perubahan' : 'Tambah Jadwal'}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
