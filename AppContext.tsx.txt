function getOrCreateDeviceId(): string {
  try {
    let devId = localStorage.getItem(STORAGE_KEYS.DEVICE_ID);
    if (!devId) {
      devId = 'dev_' + Math.random().toString(36).substring(2, 9) + '_' + Date.now().toString(36);
      localStorage.setItem(STORAGE_KEYS.DEVICE_ID, devId);
    }
    return devId;
  } catch {
    return 'dev_fallback_' + Date.now();
  }
}

function loadFromStorage<T>(key: string, fallback: T): T {
  try {
    const item = localStorage.getItem(key);
    if (!item) return fallback;
    const parsed = JSON.parse(item);
    if (parsed && typeof parsed === 'object' && !Array.isArray(parsed) && typeof fallback === 'object' && !Array.isArray(fallback)) {
      return { ...fallback, ...parsed };
    }
    return parsed ?? fallback;
  } catch (e) {
    console.warn(`Error reading localStorage for key ${key}`, e);
    return fallback;
  }
}

function saveToStorage<T>(key: string, value: T): void {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch (e) {
    console.warn(`Error writing to localStorage for key ${key}`, e);
  }
}

  const requestSenderConnection = (form: {
    targetSiteName: string;

  }): { success: boolean; message: string; connection?: SenderConnection } => {
    const rawUrl = (form.targetSiteUrl || '').trim();
    const rawKey = (form.connectionKey || '').trim();
    const rawName = (form.targetSiteName || '').trim();

    if (!rawUrl) {
      return { success: false, message: 'Alamat situs penerima (URL) wajib diisi lengkap.' };
    }
    if (!rawUrl.startsWith('http://') && !rawUrl.startsWith('https://')) {
      return { success: false, message: 'Format URL salah! Harus diawali dengan https:// atau http://' };
    }
    if (!rawKey) {
      return { success: false, message: 'Kunci Sambungan UTAMA wajib ditempel dari halaman Penerima.' };
    }
    if (rawKey.length < 6) {
      return { success: false, message: 'Kunci Sambungan tidak valid! Periksa kembali huruf besar-kecil dan tanda sambung.' };
    }
    if (!rawName) {
      return { success: false, message: 'Nama instansi/sekolah tujuan wajib diisi.' };
    }

    // Check duplicate URL
    const existing = senderConnections.find(
      c => c.targetSiteUrl.replace(/\/+$/, '').toLowerCase() === rawUrl.replace(/\/+$/, '').toLowerCase()
    );
    if (existing) {
      return { success: false, message: `Sambungan ke ${rawName} (${rawUrl}) sudah terdaftar dalam daftar tujuan.` };
    }

    const newConnection: SenderConnection = {
      id: `conn_${Date.now()}_${Math.random().toString(36).substring(2, 7)}`,
      targetSiteName: rawName,
      targetSiteUrl: rawUrl,
      connectionKey: rawKey,
      status: 'pending_approval',
      requestedAt: new Date().toISOString(),
      dispatchedCount: 0,
      autoSendEnabled: true,
      allowedFileTypes: form.allowedFileTypes && form.allowedFileTypes.length > 0
        ? form.allowedFileTypes
        : ['photo', 'document'],
      maxFileSizeMb: form.maxFileSizeMb || 25
    };

    setSenderConnections(prev => [newConnection, ...prev]);
    showToast(`Permintaan sambungan ke "${rawName}" berhasil dikirim! Menunggu persetujuan Admin Utama tujuan.`, 'info');
    return { success: true, message: 'Permintaan sambungan berhasil dikirim.', connection: newConnection };

          source: `Ekskul ${e.name}`
        });
      });
    });

    // Slider items
    sliderItems.forEach(s => {
      allFiles.push({
        name: s.title || 'Foto Beranda Utama',
        type: 'photo',
        size: 1024 * 800,
        source: 'Beranda Slider'
      });
    });

    // Library books
    libraryBooks.forEach(b => {
      allFiles.push({
        name: `Buku: ${b.title}`,
        type: 'document',
        size: 1024 * 1200,
        source: 'Perpustakaan Digital'
      });
    });

    const eligibleFiles = allFiles.filter(
      f => conn.allowedFileTypes.includes(f.type) && f.size <= (conn.maxFileSizeMb * 1024 * 1024)
    );

    if (eligibleFiles.length === 0) {
      showToast('Tidak ada berkas yang memenuhi kriteria tipe berkas atau batas ukuran sambungan ini.', 'info');
      return;
    }

    const newRecords: DispatchedFileRecord[] = eligibleFiles.map((f, idx) => ({
      id: `disp_${Date.now()}_${idx}`,
      targetConnectionId: conn.id,
      targetSiteName: conn.targetSiteName,
      targetSiteUrl: conn.targetSiteUrl,
      fileName: f.name,
      fileType: f.type,
      fileSizeBytes: f.size,
      sourceModule: f.source,
      dispatchedAt: new Date().toISOString(),
      dispatchStatus: 'success',
      autoDispatched: false
    }));

    setDispatchedFileHistory(prev => [...newRecords, ...prev]);

    setSenderConnections(prev =>
      prev.map(c =>
        c.id === connectionId
          ? {
              ...c,
              dispatchedCount: (c.dispatchedCount || 0) + eligibleFiles.length,
              lastDispatchedAt: new Date().toISOString()
            }
          : c
      )
    );

    showToast(`✅ ${eligibleFiles.length} berkas lama berhasil dikirim serentak ke ${conn.targetSiteName}!`, 'success');
  };

    fileName: string;
    fileType: 'photo' | 'video' | 'document' | 'other';
    fileSizeBytes?: number;
    sourceModule: string;

    const activeConns = senderConnections.filter(
      c => c.status === 'active' && c.autoSendEnabled !== false
    );

    if (activeConns.length === 0) {
      return;
    }

    const estimatedSize = fileInfo.fileSizeBytes || 1024 * 500;
    const nowIso = new Date().toISOString();

    const targets: Array<{ name: string; status: 'sent' | 'pending' | 'failed' }> = [];
    const newRecords: DispatchedFileRecord[] = [];
    const successfullySentConnIds: string[] = [];

    activeConns.forEach(conn => {
      const typeAllowed = conn.allowedFileTypes.includes(fileInfo.fileType);
      const sizeAllowed = estimatedSize <= (conn.maxFileSizeMb * 1024 * 1024);

      if (typeAllowed && sizeAllowed) {
        newRecords.push({
          id: `disp_${Date.now()}_${Math.random().toString(36).substring(2, 6)}`,
          targetConnectionId: conn.id,
          targetSiteName: conn.targetSiteName,
          targetSiteUrl: conn.targetSiteUrl,
          fileName: fileInfo.fileName,
          fileType: fileInfo.fileType,
          fileSizeBytes: estimatedSize,
          sourceModule: fileInfo.sourceModule,
          dispatchedAt: nowIso,
          dispatchStatus: 'success',
          autoDispatched: true
        });
        successfullySentConnIds.push(conn.id);
        targets.push({ name: conn.targetSiteName, status: 'sent' });
      } else {
        targets.push({ name: conn.targetSiteName, status: 'failed' });
      }
    });

    if (newRecords.length > 0) {
      setDispatchedFileHistory(prev => [...newRecords, ...prev]);

      setSenderConnections(prev =>
        prev.map(c =>
          successfullySentConnIds.includes(c.id)
            ? {
                ...c,
                dispatchedCount: (c.dispatchedCount || 0) + 1,
                lastDispatchedAt: nowIso
              }
            : c
        )
      );

      setActiveDispatchProgress({
        isDispatching: true,
        fileName: fileInfo.fileName,
        progressPercentage: 100,
        targets
      });

      setTimeout(() => {
        setActiveDispatchProgress(null);
      }, 4000);
    }
  };
