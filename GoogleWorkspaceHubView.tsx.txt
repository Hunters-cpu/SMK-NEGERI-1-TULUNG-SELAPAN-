import React, { useState, useEffect, useCallback } from 'react';
import { useApp } from '../context/AppContext';
import { 
  signInWithGoogle, 
  getGoogleAccessToken, 
  setGoogleAccessToken,
  logoutGoogle, 
  getGoogleUser 
} from '../services/googleAuth';
import { 
  listGoogleCalendarEvents, 
  createGoogleCalendarEvent, 
  deleteGoogleCalendarEvent, 
  syncSchoolEventsToGoogleCalendar,
  GoogleCalendarEvent, 
  CreateEventInput 
} from '../services/googleCalendarService';
import { 
  listGmailMessages, 
  getGmailMessageDetails, 
  sendGmailEmail, 
  trashGmailMessage, 
  GmailMessageSummary, 
  GmailMessageFull 
} from '../services/gmailIntegrationService';
import { 
  listClassroomCourses, 
  listClassroomCourseWork, 
  listClassroomAnnouncements, 
  listClassroomStudents, 
  createClassroomAnnouncement, 
  createClassroomCourseWork, 
  deleteClassroomCourseWork, 
  ClassroomCourse, 
  ClassroomCourseWork, 
  ClassroomAnnouncement, 
  ClassroomStudent 
} from '../services/googleClassroomService';
import { GoogleSignInButton } from '../components/GoogleSignInButton';
import { WorkspaceConfirmModal } from '../components/WorkspaceConfirmModal';
import { 
  Calendar, 
  Mail, 
  GraduationCap, 
  RefreshCw, 
  Plus, 
  Trash2, 
  Send, 
  ExternalLink, 
  CheckCircle2, 
  AlertCircle, 
  Search, 
  Clock, 
  MapPin, 
  User, 
  BookOpen, 
  MessageSquare, 
  Check, 
  X, 
  Sparkles, 
  ChevronRight, 
  Inbox, 
  FileText, 
  Users, 
  ArrowRight,
  LogOut,
  CalendarDays,
  MailCheck,
  Award
} from 'lucide-react';

export type WorkspaceTab = 'calendar' | 'gmail' | 'classroom';

interface GoogleWorkspaceHubViewProps {
  initialTab?: WorkspaceTab;
}

export const GoogleWorkspaceHubView: React.FC<GoogleWorkspaceHubViewProps> = ({ initialTab = 'calendar' }) => {
  const { schoolInfo, calendarEvents, showToast } = useApp();

  const [activeTab, setActiveTab] = useState<WorkspaceTab>(initialTab);
  const [accessToken, setAccessToken] = useState<string | null>(getGoogleAccessToken());
  const [googleUser, setGoogleUserState] = useState(getGoogleUser());
  const [isAuthenticating, setIsAuthenticating] = useState<boolean>(false);

  // 📅 Calendar State
  const [calendarEventsList, setCalendarEventsList] = useState<GoogleCalendarEvent[]>([]);
  const [isLoadingCalendar, setIsLoadingCalendar] = useState<boolean>(false);
  const [calendarFilter, setCalendarFilter] = useState<'upcoming' | 'all'>('upcoming');
  const [calendarSearch, setCalendarSearch] = useState<string>('');
  const [isCreateEventModalOpen, setIsCreateEventModalOpen] = useState<boolean>(false);
  const [newEventForm, setNewEventForm] = useState<CreateEventInput>({
    summary: '',
    description: '',
    location: schoolInfo.name || 'SMA Negeri 1 Nusantara',
    startDate: new Date().toISOString().split('T')[0],
    endDate: new Date().toISOString().split('T')[0],
    allDay: true,
    startTime: '08:00',
    endTime: '10:00'
  });

  // 📧 Gmail State
  const [gmailMessages, setGmailMessages] = useState<GmailMessageSummary[]>([]);
  const [selectedMessage, setSelectedMessage] = useState<GmailMessageFull | null>(null);
  const [isLoadingGmail, setIsLoadingGmail] = useState<boolean>(false);
  const [isLoadingMessageDetail, setIsLoadingMessageDetail] = useState<boolean>(false);
  const [gmailSearchQuery, setGmailSearchQuery] = useState<string>('');
  const [isComposeModalOpen, setIsComposeModalOpen] = useState<boolean>(false);
  const [composeForm, setComposeForm] = useState({
    to: '',
    subject: '',
    body: ''
  });

  // 🎓 Classroom State
  const [classroomCourses, setClassroomCourses] = useState<ClassroomCourse[]>([]);
  const [selectedCourseId, setSelectedCourseId] = useState<string>('');
  const [courseWorkList, setCourseWorkList] = useState<ClassroomCourseWork[]>([]);
  const [courseAnnouncements, setCourseAnnouncements] = useState<ClassroomAnnouncement[]>([]);
  const [courseStudents, setCourseStudents] = useState<ClassroomStudent[]>([]);
  const [isLoadingClassroom, setIsLoadingClassroom] = useState<boolean>(false);
  const [isLoadingCourseDetails, setIsLoadingCourseDetails] = useState<boolean>(false);
  const [classroomSubTab, setClassroomSubTab] = useState<'work' | 'announcements' | 'students'>('work');
  const [isCreateAnnouncementModalOpen, setIsCreateAnnouncementModalOpen] = useState<boolean>(false);
  const [announcementText, setAnnouncementText] = useState<string>('');
  const [isCreateWorkModalOpen, setIsCreateWorkModalOpen] = useState<boolean>(false);
  const [newWorkForm, setNewWorkForm] = useState({
    title: '',
    description: '',
    maxPoints: 100,
    dueDate: new Date(Date.now() + 7 * 24 * 3600 * 1000).toISOString().split('T')[0],
    dueTime: '23:59'
  });

  // 🛡️ Destructive / Confirm Action Modal State
  const [confirmModal, setConfirmModal] = useState<{
    isOpen: boolean;
    title: string;
    description: string;
    actionType: 'delete' | 'send' | 'sync' | 'update';
    affectedCount?: number;
    affectedItems?: string[];
    confirmLabel?: string;
    onConfirm: () => Promise<void>;
  }>({
    isOpen: false,
    title: '',
    description: '',
    actionType: 'delete',
    onConfirm: async () => {}
  });
  const [isExecutingConfirm, setIsExecutingConfirm] = useState<boolean>(false);

  // Initial token check
  useEffect(() => {
    const token = getGoogleAccessToken();
    setAccessToken(token);
    setGoogleUserState(getGoogleUser());
  }, []);

  // Handle Google Login
  const handleGoogleLogin = async () => {
    setIsAuthenticating(true);
    try {
      const result = await signInWithGoogle();
      setAccessToken(result.accessToken);
      setGoogleUserState(result.user);
      showToast(`Berhasil menghubungkan Google Workspace (${result.user.email})!`, 'success');
    } catch (err: any) {
      console.error('Google login error:', err);
      showToast(err.message || 'Gagal masuk dengan Google Workspace.', 'error');
    } finally {
      setIsAuthenticating(false);
    }
  };

  const handleGoogleLogout = async () => {
    await logoutGoogle();
    setAccessToken(null);
    setGoogleUserState(null);
    setCalendarEventsList([]);
    setGmailMessages([]);
    setClassroomCourses([]);
    showToast('Telah keluar dari sesi Google Workspace.', 'info');
  };

  // 📅 Fetch Calendar Events
  const fetchCalendar = useCallback(async () => {
    if (!accessToken) return;
    setIsLoadingCalendar(true);
    try {
      const now = new Date();
      const timeMin = calendarFilter === 'upcoming' ? now.toISOString() : undefined;
      const events = await listGoogleCalendarEvents(accessToken, 'primary', {
        timeMin,
        maxResults: 40,
        q: calendarSearch || undefined
      });
      setCalendarEventsList(events);
    } catch (err: any) {
      console.error('Failed to fetch calendar:', err);
      if (err.message?.includes('401') || err.message?.includes('auth')) {
        setAccessToken(null);
        showToast('Sesi token Google telah kedaluwarsa. Silakan masuk kembali.', 'warning');
      } else {
        showToast(err.message || 'Gagal memuat agenda Google Calendar.', 'error');
      }
    } finally {
      setIsLoadingCalendar(false);
    }
  }, [accessToken, calendarFilter, calendarSearch, showToast]);

  // 📧 Fetch Gmail Messages
  const fetchGmail = useCallback(async () => {
    if (!accessToken) return;
    setIsLoadingGmail(true);
    try {
      const res = await listGmailMessages(accessToken, {
        q: gmailSearchQuery || undefined,
        maxResults: 20
      });
      setGmailMessages(res.messages);
    } catch (err: any) {
      console.error('Failed to fetch Gmail:', err);
      if (err.message?.includes('401') || err.message?.includes('auth')) {
        setAccessToken(null);
        showToast('Sesi token Google telah kedaluwarsa. Silakan masuk kembali.', 'warning');
      } else {
        showToast(err.message || 'Gagal memuat pesan Gmail.', 'error');
      }
    } finally {
      setIsLoadingGmail(false);
    }
  }, [accessToken, gmailSearchQuery, showToast]);

  // 🎓 Fetch Classroom Courses
  const fetchClassroom = useCallback(async () => {
    if (!accessToken) return;
    setIsLoadingClassroom(true);
    try {
      const courses = await listClassroomCourses(accessToken);
      setClassroomCourses(courses);
      if (courses.length > 0 && !selectedCourseId) {
        setSelectedCourseId(courses[0].id);
      }
    } catch (err: any) {
      console.error('Failed to fetch Classroom courses:', err);
      if (err.message?.includes('401') || err.message?.includes('auth')) {
        setAccessToken(null);
        showToast('Sesi token Google telah kedaluwarsa. Silakan masuk kembali.', 'warning');
      } else {
        showToast(err.message || 'Gagal memuat data Google Classroom.', 'error');
      }
    } finally {
      setIsLoadingClassroom(false);
    }
  }, [accessToken, selectedCourseId, showToast]);

  // Fetch Course details when selectedCourseId changes
  useEffect(() => {
    if (!accessToken || !selectedCourseId) return;

    let isMounted = true;
    const fetchCourseDetails = async () => {
      setIsLoadingCourseDetails(true);
      try {
        const [works, ann, stus] = await Promise.all([
          listClassroomCourseWork(accessToken, selectedCourseId).catch(() => []),
          listClassroomAnnouncements(accessToken, selectedCourseId).catch(() => []),
          listClassroomStudents(accessToken, selectedCourseId).catch(() => [])
        ]);
        if (isMounted) {
          setCourseWorkList(works);
          setCourseAnnouncements(ann);
          setCourseStudents(stus);
        }
      } catch (err) {
        console.warn('Error fetching course sub-details:', err);
      } finally {
        if (isMounted) setIsLoadingCourseDetails(false);
      }
    };

    fetchCourseDetails();
    return () => {
      isMounted = false;
    };
  }, [accessToken, selectedCourseId]);

  // Load active tab data on change
  useEffect(() => {
    if (!accessToken) return;
    if (activeTab === 'calendar') fetchCalendar();
    if (activeTab === 'gmail') fetchGmail();
    if (activeTab === 'classroom') fetchClassroom();
  }, [activeTab, accessToken, fetchCalendar, fetchGmail, fetchClassroom]);

  // View full message detail
  const handleSelectMessage = async (msg: GmailMessageSummary) => {
    if (!accessToken) return;
    setIsLoadingMessageDetail(true);
    try {
      const full = await getGmailMessageDetails(accessToken, msg.id);
      setSelectedMessage(full);
    } catch (err: any) {
      showToast(err.message || 'Gagal membuka detail email.', 'error');
    } finally {
      setIsLoadingMessageDetail(false);
    }
  };

  // ----------------------------------------------------
  // Action Handlers with Explicit User Confirmation Modals
  // ----------------------------------------------------

  // 1. Create Calendar Event
  const handleCreateEventSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!newEventForm.summary.trim()) {
      showToast('Judul agenda tidak boleh kosong!', 'error');
      return;
    }
    if (!accessToken) return;

    try {
      await createGoogleCalendarEvent(accessToken, newEventForm);
      showToast(`Agenda '${newEventForm.summary}' berhasil ditambahkan ke Google Calendar!`, 'success');
      setIsCreateEventModalOpen(false);
      setNewEventForm({
        summary: '',
        description: '',
        location: schoolInfo.name,
        startDate: new Date().toISOString().split('T')[0],
        endDate: new Date().toISOString().split('T')[0],
        allDay: true,
        startTime: '08:00',
        endTime: '10:00'
      });
      fetchCalendar();
    } catch (err: any) {
      showToast(err.message || 'Gagal membuat agenda Google Calendar.', 'error');
    }
  };

  // 2. Delete Calendar Event (Destructive -> Requires Confirmation)
  const promptDeleteEvent = (event: GoogleCalendarEvent) => {
    setConfirmModal({
      isOpen: true,
      title: 'Hapus Agenda Google Calendar?',
      description: `Apakah Anda yakin ingin menghapus agenda "${event.summary || 'Tanpa Judul'}" secara permanen dari akun Google Calendar Anda? Tindakan ini tidak dapat dibatalkan.`,
      actionType: 'delete',
      affectedCount: 1,
      affectedItems: [event.summary || 'Agenda Google Calendar'],
      confirmLabel: 'Hapus Agenda',
      onConfirm: async () => {
        if (!accessToken) return;
        await deleteGoogleCalendarEvent(accessToken, event.id);
        showToast(`Agenda "${event.summary}" telah dihapus.`, 'info');
        fetchCalendar();
      }
    });
  };

  // 3. Batch Sync School Calendar to Google Calendar
  const promptSyncAllSchoolEvents = () => {
    if (calendarEvents.length === 0) {
      showToast('Tidak ada agenda kalender akademik sekolah untuk disinkronkan.', 'warning');
      return;
    }

    setConfirmModal({
      isOpen: true,
      title: 'Sinkronkan Kalender Akademik Sekolah ke Google Calendar?',
      description: `Sistem akan menambahkan ${calendarEvents.length} agenda kegiatan akademik, ujian, dan hari libur sekolah ke akun Google Calendar pribadi Anda.`,
      actionType: 'sync',
      affectedCount: calendarEvents.length,
      affectedItems: calendarEvents.slice(0, 5).map(e => `[${e.date}] ${e.title}`),
      confirmLabel: 'Mulai Sinkronisasi',
      onConfirm: async () => {
        if (!accessToken) return;
        const result = await syncSchoolEventsToGoogleCalendar(accessToken, calendarEvents, schoolInfo.name);
        showToast(`Berhasil menyinkronkan ${result.successCount} agenda sekolah ke Google Calendar!`, 'success');
        fetchCalendar();
      }
    });
  };

  // 4. Send Gmail Email (Mutating -> Requires Confirmation)
  const promptSendEmail = (e: React.FormEvent) => {
    e.preventDefault();
    if (!composeForm.to.trim() || !composeForm.subject.trim() || !composeForm.body.trim()) {
      showToast('Semua kolom email (Penerima, Subjek, Pesan) wajib diisi!', 'error');
      return;
    }

    setConfirmModal({
      isOpen: true,
      title: 'Kirim Pesan Email Melalui Akun Gmail Anda?',
      description: `Pesan email akan dikirimkan secara langsung ke alamat "${composeForm.to}" menggunakan akun Google Workspace terhubung. Pastikan isi surat sudah sesuai.`,
      actionType: 'send',
      affectedCount: 1,
      affectedItems: [`Kepada: ${composeForm.to}`, `Subjek: ${composeForm.subject}`],
      confirmLabel: 'Kirim Email Sekarang',
      onConfirm: async () => {
        if (!accessToken) return;
        await sendGmailEmail(accessToken, {
          to: composeForm.to,
          subject: composeForm.subject,
          body: composeForm.body,
          isHtml: true
        });
        showToast(`Email berhasil dikirim ke ${composeForm.to}!`, 'success');
        setIsComposeModalOpen(false);
        setComposeForm({ to: '', subject: '', body: '' });
        fetchGmail();
      }
    });
  };

  // 5. Trash Gmail Message (Destructive -> Requires Confirmation)
  const promptTrashEmail = (msg: GmailMessageSummary | GmailMessageFull) => {
    setConfirmModal({
      isOpen: true,
      title: 'Pindahkan Email ke Kotak Sampah?',
      description: `Surat email "${msg.subject || '(Tanpa Subjek)'}" dari ${msg.fromName || msg.from} akan dipindahkan ke folder Sampah (Trash) di Gmail Anda.`,
      actionType: 'delete',
      affectedCount: 1,
      affectedItems: [`Subjek: ${msg.subject}`, `Dari: ${msg.from}`],
      confirmLabel: 'Pindahkan ke Sampah',
      onConfirm: async () => {
        if (!accessToken) return;
        await trashGmailMessage(accessToken, msg.id);
        showToast('Email telah dipindahkan ke folder Sampah.', 'info');
        setSelectedMessage(null);
        fetchGmail();
      }
    });
  };

  // 6. Post Classroom Announcement
  const handlePostAnnouncement = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!announcementText.trim() || !selectedCourseId || !accessToken) return;

    try {
      await createClassroomAnnouncement(accessToken, selectedCourseId, announcementText);
      showToast('Pengumuman berhasil diposting ke Google Classroom!', 'success');
      setAnnouncementText('');
      setIsCreateAnnouncementModalOpen(false);
      // reload announcements
      const ann = await listClassroomAnnouncements(accessToken, selectedCourseId);
      setCourseAnnouncements(ann);
    } catch (err: any) {
      showToast(err.message || 'Gagal memposting pengumuman.', 'error');
    }
  };

  // 7. Create CourseWork Assignment
  const handleCreateCourseWork = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!newWorkForm.title.trim() || !selectedCourseId || !accessToken) return;

    try {
      const [year, month, day] = newWorkForm.dueDate.split('-').map(Number);
      const [hours, minutes] = newWorkForm.dueTime.split(':').map(Number);

      await createClassroomCourseWork(accessToken, selectedCourseId, {
        title: newWorkForm.title,
        description: newWorkForm.description,
        maxPoints: Number(newWorkForm.maxPoints) || 100,
        workType: 'ASSIGNMENT',
        dueYear: year,
        dueMonth: month,
        dueDay: day,
        dueHours: hours,
        dueMinutes: minutes
      });

      showToast(`Tugas '${newWorkForm.title}' berhasil dibuat di Google Classroom!`, 'success');
      setIsCreateWorkModalOpen(false);
      setNewWorkForm({
        title: '',
        description: '',
        maxPoints: 100,
        dueDate: new Date(Date.now() + 7 * 24 * 3600 * 1000).toISOString().split('T')[0],
        dueTime: '23:59'
      });
      // reload works
      const works = await listClassroomCourseWork(accessToken, selectedCourseId);
      setCourseWorkList(works);
    } catch (err: any) {
      showToast(err.message || 'Gagal membuat tugas di Google Classroom.', 'error');
    }
  };

  // 8. Delete CourseWork (Destructive -> Requires Confirmation)
  const promptDeleteCourseWork = (work: ClassroomCourseWork) => {
    setConfirmModal({
      isOpen: true,
      title: 'Hapus Tugas Google Classroom?',
      description: `Apakah Anda yakin ingin menghapus tugas "${work.title}" dari kelas Google Classroom ini? Semua lampiran dan nilai terkait akan dihapus.`,
      actionType: 'delete',
      affectedCount: 1,
      affectedItems: [`Tugas: ${work.title}`],
      confirmLabel: 'Hapus Tugas',
      onConfirm: async () => {
        if (!accessToken || !selectedCourseId) return;
        await deleteClassroomCourseWork(accessToken, selectedCourseId, work.id);
        showToast(`Tugas "${work.title}" berhasil dihapus.`, 'info');
        const works = await listClassroomCourseWork(accessToken, selectedCourseId);
        setCourseWorkList(works);
      }
    });
  };

  // Modal confirm executor
  const handleExecuteModalConfirm = async () => {
    setIsExecutingConfirm(true);
    try {
      await confirmModal.onConfirm();
      setConfirmModal(prev => ({ ...prev, isOpen: false }));
    } catch (err: any) {
      showToast(err.message || 'Terjadi kesalahan saat mengeksekusi operasi.', 'error');
    } finally {
      setIsExecutingConfirm(false);
    }
  };

  const selectedCourse = classroomCourses.find(c => c.id === selectedCourseId);

  return (
    <div className="space-y-6 pb-16 animate-fade-in">
      {/* Header Banner & Title */}
      <div className="bg-gradient-to-r from-[#5A5A40] via-[#4D4D36] to-[#383827] rounded-3xl p-6 sm:p-8 text-white shadow-xl relative overflow-hidden">
        <div className="absolute right-0 top-0 translate-x-8 -translate-y-8 w-64 h-64 bg-white/10 rounded-full blur-2xl pointer-events-none" />
        <div className="relative z-10 flex flex-col md:flex-row md:items-center justify-between gap-6">
          <div className="space-y-2">
            <div className="inline-flex items-center gap-2 px-3 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold tracking-wider uppercase">
              <Sparkles className="w-3.5 h-3.5 text-amber-300" />
              <span>Integrasi Resmi Google Workspace</span>
            </div>
            <h1 className="font-serif text-2xl sm:text-3xl font-bold tracking-tight">
              Google Workspace Hub
            </h1>
            <p className="text-sm text-white/85 max-w-2xl leading-relaxed">
              Akses, kelola, dan sinkronkan agenda <strong>Google Calendar</strong>, kotak masuk surat <strong>Gmail</strong>, dan materi kelas <strong>Google Classroom</strong> secara terintegrasi langsung di portal sekolah.
            </p>
          </div>

          {/* Account Status / Connect Button */}
          <div className="shrink-0">
            {accessToken && googleUser ? (
              <div className="bg-white/15 backdrop-blur-md border border-white/25 rounded-2xl p-3.5 flex items-center gap-3">
                {googleUser.photoURL ? (
                  <img
                    src={googleUser.photoURL}
                    alt={googleUser.displayName || 'Google User'}
                    className="w-10 h-10 rounded-full object-cover border-2 border-emerald-400 shrink-0"
                    referrerPolicy="no-referrer"
                  />
                ) : (
                  <div className="w-10 h-10 rounded-full bg-emerald-500 text-white flex items-center justify-center font-bold text-base">
                    {googleUser.displayName?.[0] || 'G'}
                  </div>
                )}
                <div className="text-left">
                  <div className="flex items-center gap-1.5">
                    <span className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
                    <span className="text-xs font-bold text-emerald-300">Terhubung Resmi</span>
                  </div>
                  <div className="font-semibold text-xs text-white line-clamp-1 max-w-[180px]">
                    {googleUser.displayName || googleUser.email}
                  </div>
                  <div className="text-[10px] text-white/70 line-clamp-1 max-w-[180px]">
                    {googleUser.email}
                  </div>
                </div>
                <button
                  onClick={handleGoogleLogout}
                  className="p-1.5 hover:bg-white/20 rounded-lg text-white/80 hover:text-white transition-colors cursor-pointer ml-1"
                  title="Putuskan Hubungan Google"
                >
                  <LogOut className="w-4 h-4" />
                </button>
              </div>
            ) : (
              <GoogleSignInButton
                onClick={handleGoogleLogin}
                isLoading={isAuthenticating}
                text="Hubungkan Google Workspace"
                className="shadow-lg"
              />
            )}
          </div>
        </div>
      </div>

      {/* If Not Connected Notice */}
      {!accessToken && (
        <div className="bg-white dark:bg-[#1E1E1C] rounded-2xl p-8 border border-[#E0E0D8] dark:border-[#383832] shadow-sm text-center space-y-4">
          <div className="w-16 h-16 rounded-2xl bg-[#5A5A40]/10 dark:bg-[#5A5A40]/30 text-[#5A5A40] dark:text-[#C5C5B0] flex items-center justify-center mx-auto">
            <GraduationCap className="w-8 h-8" />
          </div>
          <div className="max-w-md mx-auto space-y-2">
            <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
              Hubungkan Akun Google untuk Mengaktifkan Fitur
            </h3>
            <p className="text-xs text-[#666] dark:text-[#AAA] leading-relaxed">
              Akses Google Calendar, Gmail, dan Google Classroom membutuhkan izin resmi dari akun Google Anda. Klik tombol di bawah ini untuk memulai integrasi.
            </p>
          </div>
          <div className="pt-2">
            <GoogleSignInButton
              onClick={handleGoogleLogin}
              isLoading={isAuthenticating}
              text="Masuk dengan Google (Calendar, Gmail, Classroom)"
            />
          </div>
        </div>
      )}

      {/* Main Feature Tabs (When Connected) */}
      {accessToken && (
        <>
          {/* Navigation Bar Tabs */}
          <div className="flex items-center justify-between border-b border-[#E0E0D8] dark:border-[#383832] pb-2 overflow-x-auto gap-2">
            <div className="flex items-center gap-2">
              <button
                onClick={() => setActiveTab('calendar')}
                className={`inline-flex items-center gap-2 px-4 py-2.5 rounded-xl text-xs sm:text-sm font-bold transition-all cursor-pointer whitespace-nowrap ${
                  activeTab === 'calendar'
                    ? 'bg-[#5A5A40] text-white shadow-md'
                    : 'bg-white dark:bg-[#1E1E1C] text-[#666] dark:text-[#AAA] hover:bg-[#F5F5F0] dark:hover:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B]'
                }`}
              >
                <Calendar className="w-4 h-4" />
                <span>Google Calendar</span>
                <span className={`px-1.5 py-0.5 rounded-md text-[10px] ${activeTab === 'calendar' ? 'bg-white/20 text-white' : 'bg-[#F5F5F0] dark:bg-[#282825] text-[#555]'}`}>
                  {calendarEventsList.length}
                </span>
              </button>

              <button
                onClick={() => setActiveTab('gmail')}

                    value={newWorkForm.dueDate}
                    onChange={e => setNewWorkForm({ ...newWorkForm, dueDate: e.target.value })}
                    className="w-full px-3 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-[#2D2D2A] dark:text-[#EAEAE2] focus:outline-none focus:ring-2 focus:ring-[#2E6F40]"
                  />
                </div>
              </div>

              <div>
                <label className="block font-bold text-[#2D2D2A] dark:text-[#EAEAE2] mb-1.5">
                  Petunjuk & Instruksi Pengerjaan
                </label>
                <textarea
                  rows={4}
                  placeholder="Jelaskan instruksi pengerjaan, format berkas yang diterima, dan kriteria penilaian..."
                  value={newWorkForm.description}
                  onChange={e => setNewWorkForm({ ...newWorkForm, description: e.target.value })}
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-[#2D2D2A] dark:text-[#EAEAE2] focus:outline-none focus:ring-2 focus:ring-[#2E6F40]"
                />
              </div>

              <div className="flex items-center justify-end gap-3 pt-3 border-t border-[#E0E0D8] dark:border-[#383832]">
                <button
                  type="button"
                  onClick={() => setIsCreateWorkModalOpen(false)}
                  className="px-4 py-2.5 rounded-xl font-bold bg-[#F5F5F0] dark:bg-[#282825] hover:bg-[#EAEAE5] text-[#2D2D2A] dark:text-[#EAEAE2] transition-colors cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 rounded-xl font-bold bg-[#2E6F40] hover:bg-[#255833] text-white transition-all cursor-pointer shadow-sm active:scale-95"
                >
                  Simpan & Terbitkan Tugas
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 5. Destructive Action Confirmation Modal (Crucial Workspace Mandate) */}
      <WorkspaceConfirmModal
        isOpen={confirmModal.isOpen}
        title={confirmModal.title}
        description={confirmModal.description}
        actionType={confirmModal.actionType}
        affectedCount={confirmModal.affectedCount}
        affectedItems={confirmModal.affectedItems}
        confirmLabel={confirmModal.confirmLabel}
        isLoading={isExecutingConfirm}
        onConfirm={handleExecuteModalConfirm}
        onClose={() => setConfirmModal(prev => ({ ...prev, isOpen: false }))}
      />
    </div>
  );
};
