                <button
                  type="button"
                  onClick={() => setShowSuperAdminPass(!showSuperAdminPass)}
                  className="absolute right-3 top-3 text-[#8E8E85] hover:text-[#2D2D2A] cursor-pointer"

                  {showSuperAdminPass ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}

              <button
                type="submit"
                className="px-5 py-2.5 bg-[#8C4A32] hover:bg-[#723C28] text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm transition-all cursor-pointer"

                type="button"

            {(customAdmins || []).length === 0 ? (

                {(customAdmins || []).map(adm => (

                    <div className="flex items-center justify-between">

                        requestDeleteConfirm({

                      className="px-3 py-1.5 bg-[#FAF8F5] hover:bg-[#E0E0D8] text-[#8C4A32] text-xs font-bold rounded-xl flex items-center gap-1.5 cursor-pointer"

                      type="button"
                      onClick={() => handleOpenEditClassAdmin(cls)}
                      className="px-3.5 py-1.5 bg-[#5A5A40] hover:bg-[#484833] text-white text-xs font-bold rounded-xl flex items-center gap-1.5 cursor-pointer shadow-sm"

          </div>

          {/* Edit Class Admin Modal */}
          {isClassAdminModalOpen && editingClassAdmin && (
            <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
              <div className="relative w-full max-w-md bg-white border border-[#E0E0D8] rounded-[32px] shadow-2xl p-6 sm:p-8 space-y-5 text-[#2D2D2A]">
                <div className="flex items-center justify-between pb-3 border-b border-[#E0E0D8]">
                  <div className="flex items-center gap-2.5">
                    <div className="p-2 rounded-xl bg-[#5A5A40] text-white">
                      <UserCheck className="w-5 h-5" />
                    </div>
                    <div>
                      <span className="text-[10px] font-bold uppercase tracking-widest text-[#8E8E85]">Pengaturan Akun</span>
                      <h4 className="font-serif text-lg font-bold text-[#2D2D2A]">
                        Ubah Admin {editingClassAdmin.className}

                  </div>
                  <button
                    onClick={() => {
                      setIsClassAdminModalOpen(false);
                      setEditingClassAdmin(null);

                    className="text-[#8E8E85] hover:text-[#2D2D2A] p-1.5 rounded-xl hover:bg-[#F5F5F0] cursor-pointer"

                    <X className="w-5 h-5" />
                  </button>
                </div>

                <form onSubmit={handleSaveClassAdmin} className="space-y-4">
                  <div>
                    <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                      Username Login Admin Kelas *
                    </label>
                    <input
                      type="text"
                      required
                      value={editingClassAdmin.username}
                      onChange={e => setEditingClassAdmin({ ...editingClassAdmin, username: e.target.value })}

                      className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"

                  <div>
                    <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                      Kata Sandi Admin Kelas *
                    </label>
                    <input
                      type="text"
                      required
                      value={editingClassAdmin.password}
                      onChange={e => setEditingClassAdmin({ ...editingClassAdmin, password: e.target.value })}
                      placeholder="Masukkan kata sandi baru..."
                      className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                    />
                  </div>

                  <div className="flex items-center justify-end gap-2 pt-3 border-t border-[#E0E0D8]">
                    <button
                      type="button"
                      onClick={() => {
                        setIsClassAdminModalOpen(false);
                        setEditingClassAdmin(null);
                      }}
                      className="px-4 py-2 text-xs font-bold text-[#8E8E85] hover:text-[#2D2D2A] cursor-pointer"
                    >
                      Batal
                    </button>
                    <button
                      type="submit"
                      className="px-5 py-2.5 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm cursor-pointer"
                    >
                      Simpan Kredensial
                    </button>
                  </div>
                </form>
              </div>
            </div>
          )}
        </div>
      )}

      {/* 🔹 TAB 4: SESI LOGIN & KUNCI PERANGKAT (1 AKUN = 1 PERANGKAT) */}
      {activeTab === 'students' && (() => {
        // Build list of all system accounts
        const allAccountsList: Array<{
          key: string;
          type: 'super_admin' | 'admin' | 'class_admin' | 'eskul_admin' | 'student';
          typeName: string;
          typeBadgeColor: string;
          name: string;
          subtext: string;
          session?: {
            deviceId: string;
            deviceLabel?: string;
            loginTime?: string;
          };
          rawAdmin?: CustomAdmin;
        }> = [];

        // 1. Super Admin Utama
        const superAdminSession = activeSessions?.['super_admin'];
        allAccountsList.push({
          key: 'super_admin',
          type: 'super_admin',
          typeName: 'Super Admin Utama',
          typeBadgeColor: 'bg-[#5A5A40] text-white',
          name: 'Super Admin Sekolah',
          subtext: 'mumumimi353@gmail.com / superadmin',
          session: superAdminSession
        });

        // 2. Custom Admins
        (customAdmins || []).forEach(adm => {
          const admKey = `custom_admin_${adm.username.toLowerCase()}`;
          allAccountsList.push({
            key: admKey,
            type: 'admin',
            typeName: 'Admin Khusus',
            typeBadgeColor: 'bg-[#8E8E85] text-white',
            name: adm.name,
            subtext: `Username: ${adm.username}`,
            session: activeSessions?.[admKey],
            rawAdmin: adm
          });
        });

        // 3. Admin Kelas
        (classes || []).forEach(c => {
          const cKey = `class_admin_${c.id}`;

            key: cKey,
            type: 'class_admin',
            typeName: 'Admin Kelas',
            typeBadgeColor: 'bg-stone-700 text-white',
            name: `Admin Kelas ${c.name}`,
            subtext: `User: ${c.adminUsername} • Kelas ${c.name}`,
            session: activeSessions?.[cKey]
          });
        });

        // 4. Admin Eskul
        (eskulList || []).forEach(e => {
          const eKey = `eskul_admin_${e.id}`;
          allAccountsList.push({
            key: eKey,
            type: 'eskul_admin',
            typeName: 'Admin Eskul',
            typeBadgeColor: 'bg-[#5A5A40] text-white',
            name: `Admin Eskul ${e.name}`,
            subtext: `User: ${e.adminUsername || 'admin_' + e.name.toLowerCase()} • Eskul ${e.name}`,
            session: activeSessions?.[eKey]
          });

        // 5. Siswa
        (students || []).forEach(s => {
          const sKey = `student_${s.id}`;
          const cls = (classes || []).find(c => c.id === s.classId);
          const studentSession = activeSessions?.[sKey] || activeSessions?.[s.id] || (s.currentDeviceId ? { deviceId: s.currentDeviceId, loginTime: s.lastLoginAt } : undefined);

            key: sKey,
            type: 'student',
            typeName: 'Siswa',
            typeBadgeColor: 'bg-[#F5F5F0] text-[#5A5A40] border border-[#E0E0D8]',

        const filteredAccounts = allAccountsList.filter(acc => {
          // Role filter
          if (sessionRoleFilter === 'admin' && acc.type !== 'super_admin' && acc.type !== 'admin') return false;
          if (sessionRoleFilter === 'class_admin' && acc.type !== 'class_admin') return false;
          if (sessionRoleFilter === 'eskul_admin' && acc.type !== 'eskul_admin') return false;
          if (sessionRoleFilter === 'student' && acc.type !== 'student') return false;

          // Status filter
          const isLocked = Boolean(acc.session?.deviceId);
          if (sessionStatusFilter === 'locked' && !isLocked) return false;
          if (sessionStatusFilter === 'unlocked' && isLocked) return false;

              acc.subtext.toLowerCase().includes(q) ||

                  <span>Reset Semua Sesi ({activeLocksCount})</span>
                </button>
              </div>
            </div>

            {/* Quick Metrics Bar */}
            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
              <div className="p-3.5 bg-[#F8F8F5] rounded-2xl border border-[#E0E0D8]">
                <p className="text-[11px] text-[#8E8E85] font-medium">Total Akun Terdaftar</p>
                <p className="text-xl font-bold text-[#2D2D2A] mt-0.5">{allAccountsList.length}</p>
              </div>
              <div className="p-3.5 bg-[#F8F8F5] rounded-2xl border border-[#E0E0D8]">
                <p className="text-[11px] text-[#8E8E85] font-medium">Sesi Terkunci Aktif</p>
                <p className="text-xl font-bold text-amber-700 mt-0.5">{activeLocksCount}</p>
              </div>
              <div className="p-3.5 bg-[#F8F8F5] rounded-2xl border border-[#E0E0D8]">
                <p className="text-[11px] text-[#8E8E85] font-medium">Siswa Terkunci</p>
                <p className="text-xl font-bold text-[#5A5A40] mt-0.5">
                  {allAccountsList.filter(a => a.type === 'student' && Boolean(a.session?.deviceId)).length}
                </p>
              </div>
              <div className="p-3.5 bg-[#F8F8F5] rounded-2xl border border-[#E0E0D8]">
                <p className="text-[11px] text-[#8E8E85] font-medium">Admin Terkunci</p>
                <p className="text-xl font-bold text-[#5A5A40] mt-0.5">
                  {allAccountsList.filter(a => a.type !== 'student' && Boolean(a.session?.deviceId)).length}
                </p>
              </div>
            </div>

            {/* Search & Filters */}
            <div className="flex flex-col md:flex-row gap-3 items-center justify-between">
              {/* Search */}
              <div className="relative w-full md:w-80">
                <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-[#8E8E85]" />
                <input

                    onClick={() => setSessionSearchQuery('')}

                    <X className="w-3.5 h-3.5" />

                    sessionRoleFilter === 'all'

                  type="button"

                  placeholder="contoh: Anggota Kelas 7A / Peserta Ekstrakurikuler"
                  value={bulkLoginForm.notes}
                  onChange={e => setBulkLoginForm({ ...bulkLoginForm, notes: e.target.value })}
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-[#5A5A40]"
                />
              </div>

              <div className="flex items-center justify-end gap-3 pt-4 border-t border-[#E0E0D8]">
                <button
                  type="button"
                  onClick={() => setIsBulkLoginModalOpen(false)}
                  className="px-5 py-2.5 rounded-xl border border-[#E0E0D8] text-xs font-bold text-[#8E8E85] hover:text-[#2D2D2A] cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 bg-[#8C4A32] hover:bg-[#723C28] text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm cursor-pointer inline-flex items-center gap-2"
                >
                  <Download className="w-4 h-4" />
                  <span>Tambahkan Semua Sekaligus</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 🔹 MODAL: UBAH DATA AKUN LOGIN */}
      {isEditLoginModalOpen && editingLoginAccount && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm animate-fadeIn">
          <div className="bg-white rounded-[32px] max-w-lg w-full p-6 sm:p-8 border border-[#E0E0D8] shadow-2xl space-y-5">
            <div className="flex items-center justify-between border-b border-[#E0E0D8] pb-4">
              <div className="flex items-center gap-2.5">
                <div className="p-2 rounded-xl bg-blue-600 text-white">
                  <Edit3 className="w-5 h-5" />
                </div>
                <div>
                  <span className="text-[10px] font-bold uppercase tracking-widest text-[#8E8E85]">Pengaturan Akun</span>
                  <h3 className="font-serif text-lg font-bold text-[#2D2D2A]">
                    Ubah Data Akun Login
                  </h3>
                </div>
              </div>
              <button
                type="button"
                onClick={() => {
                  setIsEditLoginModalOpen(false);
                  setEditingLoginAccount(null);
                }}
                className="p-2 hover:bg-[#F5F5F0] rounded-xl text-[#8E8E85] cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveEditLogin} className="space-y-4">
              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                  Email Google (Terkunci)
                </label>
                <input
                  type="text"
                  disabled
                  value={editingLoginAccount.email}
                  className="w-full px-3.5 py-2.5 bg-[#EBEBE5] border border-[#E0E0D8] rounded-xl text-[#8E8E85] text-sm font-mono cursor-not-allowed"
                />
              </div>

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                  Nama Lengkap *
                </label>
                <input
                  type="text"
                  required
                  value={editingLoginAccount.name}
                  onChange={e => setEditingLoginAccount({ ...editingLoginAccount, name: e.target.value })}
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-blue-600"
                />
              </div>

              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                    Kelas
                  </label>
                  <select
                    value={editingLoginAccount.classId || 'none'}
                    onChange={e => {
                      const cId = e.target.value;
                      const sel = classes.find(c => c.id === cId);
                      setEditingLoginAccount({
                        ...editingLoginAccount,
                        classId: cId,
                        className: sel ? sel.name : (cId === 'none' ? 'Tanpa Kelas' : 'Semua Kelas')
                      });
                    }}
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-blue-600"
                  >
                    <option value="none">Tanpa Kelas (-)</option>
                    {classes.map(c => (
                      <option key={c.id} value={c.id}>{c.name}</option>
                    ))}
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                    Status / Peran *
                  </label>
                  <select
                    value={editingLoginAccount.statusRole}
                    onChange={e => setEditingLoginAccount({ ...editingLoginAccount, statusRole: e.target.value as LoginRoleStatus })}
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-blue-600"
                  >
                    <option value="student">Siswa</option>
                    <option value="class_admin">Admin Kelas</option>
                    <option value="eskul_admin">Admin Ekskul</option>
                    <option value="custom_admin">Admin Pengelola</option>
                  </select>
                </div>
              </div>

              {editingLoginAccount.statusRole === 'eskul_admin' && (
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                    Pilih Ekstrakurikuler yang Dikelola *
                  </label>
                  <select
                    value={editingLoginAccount.eskulId || ''}
                    onChange={e => setEditingLoginAccount({ ...editingLoginAccount, eskulId: e.target.value })}
                    className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-blue-600"
                  >
                    <option value="">-- Pilih Eskul --</option>
                    {eskulList.map(eskul => (
                      <option key={eskul.id} value={eskul.id}>{eskul.name}</option>
                    ))}
                  </select>
                </div>
              )}

              <div>
                <label className="block text-xs font-bold text-[#2D2D2A] mb-1">
                  Keterangan
                </label>
                <input
                  type="text"
                  value={editingLoginAccount.notes || ''}
                  onChange={e => setEditingLoginAccount({ ...editingLoginAccount, notes: e.target.value })}
                  className="w-full px-3.5 py-2.5 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] text-sm focus:outline-none focus:border-blue-600"
                />
              </div>

              <div className="flex items-center justify-end gap-3 pt-4 border-t border-[#E0E0D8]">
                <button
                  type="button"
                  onClick={() => {
                    setIsEditLoginModalOpen(false);
                    setEditingLoginAccount(null);
                  }}
                  className="px-5 py-2.5 rounded-xl border border-[#E0E0D8] text-xs font-bold text-[#8E8E85] hover:text-[#2D2D2A] cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 bg-blue-600 hover:bg-blue-700 text-white font-bold text-xs uppercase tracking-wider rounded-xl shadow-sm cursor-pointer inline-flex items-center gap-2"
                >
                  <Save className="w-4 h-4" />
                  <span>Simpan Perubahan</span>
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 👑 MODAL KHUSUS ADMIN UTAMA: INSPEKSI & SALIN KODE HTML WEBSITE */}
      {isHtmlCodeModalOpen && isSuperAdmin && (() => {
        const portalCode = generateSchoolPortalHtml({
          schoolInfo,
          classes,
          students,
          teachers,
          announcements,
          eskulList,
          calendarEvents,
          facilities: schoolInfo.fasilitas
        });
        const indexCode = getWebsiteIndexHtml(schoolInfo);
        const activeCode = htmlCodeTab === 'portal' ? portalCode : indexCode;
        const lineCount = activeCode.split('\n').length;
        const sizeKb = (new Blob([activeCode]).size / 1024).toFixed(1);

        return (
          <div className="fixed inset-0 z-50 flex items-center justify-center p-3 sm:p-4 bg-black/70 backdrop-blur-md animate-fadeIn">
            <div className="bg-white rounded-[32px] max-w-5xl w-full p-5 sm:p-7 border border-[#E0E0D8] shadow-2xl space-y-4 max-h-[94vh] flex flex-col">
              {/* Header Modal */}
              <div className="flex items-center justify-between border-b border-[#E0E0D8] pb-3.5">
                <div className="flex items-center gap-3">
                  <div className="w-10 h-10 rounded-2xl bg-[#5A5A40] text-white flex items-center justify-center shadow-sm">
                    <Crown className="w-5 h-5 text-amber-300" />
                  </div>
                  <div>
                    <div className="flex items-center gap-2">
                      <h3 className="font-serif text-lg font-bold text-[#2D2D2A]">
                        Inspeksi & Ekspor Kode HTML Website
                      </h3>
                      <span className="px-2 py-0.5 rounded-full text-[10px] font-bold bg-amber-100 text-amber-900 border border-amber-300">
                        Khusus Admin Utama
                      </span>
                    </div>
                    <p className="text-xs text-[#8E8E85]">
                      Kode sumber HTML website sekolah siap pakai, dapat disalin, diuji pratinjau, atau diunduh langsung.
                    </p>
                  </div>
                </div>

                <button
                  type="button"
                  onClick={() => setIsHtmlCodeModalOpen(false)}
                  className="p-2 hover:bg-[#F5F5F0] rounded-xl text-[#8E8E85] hover:text-[#2D2D2A] transition-colors cursor-pointer"
                >
                  <X className="w-5 h-5" />
                </button>
              </div>

              {/* Tab Selector & Stats Info */}
              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 bg-[#F8F8F5] p-2 rounded-2xl border border-[#E0E0D8]">
                <div className="flex items-center gap-1 bg-white p-1 rounded-xl border border-[#E0E0D8]">
                  <button
                    type="button"
                    onClick={() => setHtmlCodeTab('portal')}
                    className={`px-3.5 py-1.5 rounded-lg text-xs font-bold transition-all cursor-pointer flex items-center gap-1.5 ${
                      htmlCodeTab === 'portal'
                        ? 'bg-[#5A5A40] text-white shadow-xs'
                        : 'text-[#8E8E85] hover:text-[#2D2D2A]'
                    }`}
                  >
                    <FileCode className="w-3.5 h-3.5" />
                    <span>Portal Lengkap (.HTML)</span>
                  </button>

                  <button
                    type="button"
                    onClick={() => setHtmlCodeTab('index')}
                    className={`px-3.5 py-1.5 rounded-lg text-xs font-bold transition-all cursor-pointer flex items-center gap-1.5 ${
                      htmlCodeTab === 'index'
                        ? 'bg-[#5A5A40] text-white shadow-xs'
                        : 'text-[#8E8E85] hover:text-[#2D2D2A]'
                    }`}
                  >
                    <Globe className="w-3.5 h-3.5" />
                    <span>Entry Point (index.html)</span>
                  </button>
                </div>

                <div className="flex items-center gap-2 text-[11px] font-semibold text-[#8E8E85] px-2">
                  <span>📊 {lineCount} Baris</span>
                  <span>•</span>
                  <span>💾 {sizeKb} KB</span>
                </div>
              </div>

              {/* Toolbar Aksi */}
              <div className="flex flex-wrap items-center justify-between gap-2.5">
                <div className="relative flex-1 min-w-[200px]">
                  <Search className="w-4 h-4 text-[#8E8E85] absolute left-3 top-2.5" />
                  <input
                    type="text"
                    placeholder="Cari tag, kata kunci, atau teks di dalam HTML..."
                    value={htmlSearchQuery}
                    onChange={e => setHtmlSearchQuery(e.target.value)}
                    className="w-full pl-9 pr-3 py-2 bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-xs text-[#2D2D2A] focus:outline-none focus:border-[#5A5A40]"
                  />
                </div>

                <div className="flex items-center gap-2">
                  <button
                    type="button"
                    onClick={() => handleCopyHtml(activeCode)}
                    className="px-4 py-2 bg-[#5A5A40] hover:bg-[#484833] text-white text-xs font-bold rounded-xl shadow-sm cursor-pointer transition-all flex items-center gap-1.5 active:scale-95"
                  >
                    {copiedHtml ? <Check className="w-4 h-4 text-emerald-300" /> : <Copy className="w-4 h-4" />}
                    <span>{copiedHtml ? 'Kode Tersalin!' : 'Salin Kode HTML'}</span>
                  </button>

                  <button
                    type="button"
                    onClick={() => handlePreviewHtmlInNewTab(htmlCodeTab)}
                    className="px-3.5 py-2 bg-[#F5F5F0] hover:bg-[#EBEBE5] text-[#5A5A40] border border-[#E0E0D8] text-xs font-bold rounded-xl shadow-xs cursor-pointer transition-all flex items-center gap-1.5"
                    title="Buka Pratinjau Dokumen HTML di Tab Baru"
                  >
                    <ExternalLink className="w-4 h-4" />
                    <span>Uji Pratinjau</span>
                  </button>

                  <button
                    type="button"
                    onClick={htmlCodeTab === 'portal' ? handleDownloadPortalHtmlFile : handleDownloadIndexHtmlFile}
                    className="px-3.5 py-2 bg-emerald-700 hover:bg-emerald-800 text-white text-xs font-bold rounded-xl shadow-xs cursor-pointer transition-all flex items-center gap-1.5 active:scale-95"
                    title="Unduh Berkas .HTML ke Komputer"
                  >
                    <Download className="w-4 h-4" />
                    <span>Unduh Berkas</span>
                  </button>
                </div>
              </div>

              {/* Code Box Area */}
              <div className="flex-1 min-h-[300px] max-h-[460px] bg-[#1E1E24] rounded-2xl border border-[#2D2D38] p-4 overflow-auto relative font-mono text-xs text-[#D8D8E0] leading-relaxed shadow-inner">
                <pre className="whitespace-pre overflow-x-auto selection:bg-[#5A5A40] selection:text-white">
                  <code>{activeCode}</code>
                </pre>
              </div>

              {/* Footer Modal */}
              <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 pt-3 border-t border-[#E0E0D8] text-xs text-[#8E8E85]">
                <p>
                  💡 <strong>Tips:</strong> Berkas HTML mandiri ini dapat Anda host di hosting statis (GitHub Pages, Netlify, cPanel) atau langsung dibuka secara luring tanpa internet.
                </p>
                <button
                  type="button"
                  onClick={() => setIsHtmlCodeModalOpen(false)}
                  className="px-5 py-2 rounded-xl bg-[#F5F5F0] hover:bg-[#EBEBE5] text-[#2D2D2A] text-xs font-bold cursor-pointer transition-all self-end sm:self-auto"
                >
                  Tutup
                </button>
              </div>
            </div>
          </div>
        );
      })()}
    </div>
  );
};
