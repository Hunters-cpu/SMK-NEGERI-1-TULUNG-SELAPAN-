import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { LibraryBook } from '../types';
import { 
  BookOpen, 
  Search, 
  Filter, 
  Plus, 
  Edit3, 
  Trash2, 
  Download, 
  BookMarked, 
  Sparkles, 
  User, 
  Eye, 
  Layers,
  CheckCircle,
  ExternalLink,
  X
} from 'lucide-react';
import { MediaUploadField } from '../components/MediaUploadField';

export const LibraryView: React.FC = () => {
  const { 
    libraryBooks, 
    addLibraryBook, 
    updateLibraryBook, 
    deleteLibraryBook, 
    borrowBook,
    authSession,
    requestDeleteConfirm,
    hasAdminScope,
    showToast
  } = useApp();

  const [searchQuery, setSearchQuery] = useState<string>('');
  const [selectedCategory, setSelectedCategory] = useState<string>('Semua');
  const [isModalOpen, setIsModalOpen] = useState<boolean>(false);
  const [editingItem, setEditingItem] = useState<LibraryBook | null>(null);

  const [form, setForm] = useState({
    title: '',
    author: '',
    category: 'Pelajaran' as LibraryBook['category'],
    year: new Date().getFullYear(),
    isbn: '',
    coverUrl: '',
    pdfUrl: '',
    description: '',
    stock: 10
  });

  const canManage = hasAdminScope('library') || authSession.role === 'super_admin';

  // Filtered
  const filteredBooks = libraryBooks.filter(b => {
    const matchSearch = b.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
      b.author.toLowerCase().includes(searchQuery.toLowerCase()) ||
      (b.isbn && b.isbn.toLowerCase().includes(searchQuery.toLowerCase()));

    const matchCat = selectedCategory === 'Semua' || b.category === selectedCategory;
    return matchSearch && matchCat;
  });

  const handleOpenAdd = () => {
    setEditingItem(null);
    setForm({
      title: '',
      author: '',
      category: 'Pelajaran',
      year: new Date().getFullYear(),
      isbn: '',
      coverUrl: '',
      pdfUrl: '',
      description: '',
      stock: 10
    });
    setIsModalOpen(true);
  };

  const handleOpenEdit = (item: LibraryBook) => {
    setEditingItem(item);
    setForm({
      title: item.title,
      author: item.author,
      category: item.category,
      year: item.year || new Date().getFullYear(),
      isbn: item.isbn || '',
      coverUrl: item.coverUrl || '',
      pdfUrl: item.pdfUrl || item.ebookUrl || '',
      description: item.description || item.synopsis || '',
      stock: item.stock !== undefined ? item.stock : (item.stockAvailable ?? 5)
    });
    setIsModalOpen(true);
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.title.trim() || !form.author.trim()) return;

    const bookStock = Number(form.stock);

    if (editingItem) {
      updateLibraryBook(editingItem.id, {
        title: form.title.trim(),
        author: form.author.trim(),
        category: form.category as any,
        year: Number(form.year),
        isbn: form.isbn.trim() || undefined,
        coverUrl: form.coverUrl.trim() || undefined,
        pdfUrl: form.pdfUrl.trim() || undefined,
        ebookUrl: form.pdfUrl.trim() || undefined,
        description: form.description.trim() || undefined,
        synopsis: form.description.trim() || undefined,
        stock: bookStock,
        stockAvailable: bookStock,
        totalStock: Math.max(bookStock, editingItem.totalStock || bookStock)
      });
    } else {
      addLibraryBook({
        title: form.title.trim(),
        author: form.author.trim(),
        category: form.category as any,
        year: Number(form.year),
        isbn: form.isbn.trim() || undefined,
        coverUrl: form.coverUrl.trim() || undefined,
        pdfUrl: form.pdfUrl.trim() || undefined,
        ebookUrl: form.pdfUrl.trim() || undefined,
        description: form.description.trim() || undefined,
        synopsis: form.description.trim() || undefined,
        stock: bookStock,
        stockAvailable: bookStock,
        totalStock: bookStock
      });
    }

    setIsModalOpen(false);
  };

  const handleDelete = (item: LibraryBook) => {
    requestDeleteConfirm({
      itemName: `Buku "${item.title}"`,
      message: `Yakin ingin menghapus buku "${item.title}" dari katalog perpustakaan?`,
      onConfirm: () => deleteLibraryBook(item.id)
    });
  };

  const handleBorrow = (item: LibraryBook) => {
    const currentStock = item.stock !== undefined ? item.stock : (item.stockAvailable ?? 0);
    if (currentStock <= 0) {
      showToast('Stok fisik buku ini sedang habis dipinjam.', 'warning');
      return;
    }
    borrowBook(item.id);
    showToast(`Berhasil mengajukan peminjaman buku "${item.title}". Silakan ambil di perpustakaan.`, 'success');
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#5A5A40] to-[#2B2B1E] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <BookOpen className="w-4 h-4" />
            <span>Menu 6: Perpustakaan Digital & Fisik</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            Perpustakaan Digital Sekolah
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Katalog buku pelajaran kurikulum merdeka, referensi umum, novel sastra, ensiklopedia, dan e-book digital yang dapat dibaca dan dipinjam oleh seluruh civitas akademika sekolah.
          </p>

          {canManage && (
            <div className="pt-2">
              <button
                onClick={handleOpenAdd}
                className="inline-flex items-center gap-2 px-4 py-2.5 bg-amber-400 hover:bg-amber-300 text-[#2D2D2A] font-bold rounded-2xl text-xs sm:text-sm shadow-md transition-transform active:scale-95 cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Tambah Koleksi Buku</span>
              </button>
            </div>
          )}
        </div>
      </div>

      {/* Filter & Search Bar */}
      <div className="flex flex-col sm:flex-row items-center gap-4 bg-white dark:bg-[#1E1E1C] p-4 rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs">
        <div className="relative flex-1 w-full">
          <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-[#8E8E85]" />
          <input
            type="text"
            placeholder="Cari judul buku, nama penulis, ISBN, atau topik..."
            value={searchQuery}
            onChange={e => setSearchQuery(e.target.value)}
            className="w-full pl-10 pr-4 py-2 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2] focus:outline-none focus:ring-2 focus:ring-[#5A5A40]"
          />
        </div>

        <div className="flex items-center gap-1.5 overflow-x-auto pb-1 sm:pb-0 w-full sm:w-auto">
          {['Semua', 'Pelajaran', 'Referensi', 'Novel', 'Ensiklopedia'].map(cat => (
            <button
              key={cat}
              onClick={() => setSelectedCategory(cat)}
              className={`px-3 py-1.5 rounded-xl text-xs font-bold transition-all cursor-pointer shrink-0 ${
                selectedCategory === cat
                  ? 'bg-[#5A5A40] text-white shadow-xs'
                  : 'bg-[#F5F5F0] dark:bg-[#282825] text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-white'
              }`}
            >
              {cat}
            </button>
          ))}
        </div>
      </div>

      {/* Books Grid */}
      {filteredBooks.length === 0 ? (
        <div className="text-center py-16 bg-white dark:bg-[#1E1E1C] rounded-3xl border border-[#E0E0D8] dark:border-[#383832] p-8">
          <BookOpen className="w-12 h-12 text-[#8E8E85] mx-auto mb-3 opacity-40" />
          <p className="text-base font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">Belum ada koleksi buku yang cocok</p>
          <p className="text-xs text-[#8E8E85] mt-1">Coba sesuaikan kata kunci pencarian atau kategori buku.</p>
        </div>
      ) : (
        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-4 sm:gap-6">
          {filteredBooks.map(book => (
            <div
              key={book.id}
              className="bg-white dark:bg-[#1E1E1C] rounded-[24px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm hover:shadow-md transition-all overflow-hidden flex flex-col justify-between group"
            >
              <div>
                {/* Book Cover */}
                <div className="relative aspect-[3/4] w-full overflow-hidden bg-neutral-100 dark:bg-neutral-800">
                  <img
                    src={book.coverUrl || 'https://images.unsplash.com/photo-1544716278-ca5e3f4abd8c?w=400&auto=format&fit=crop&q=80'}
                    alt={book.title}
                    className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
                  />
                  <div className="absolute top-2 left-2 px-2 py-0.5 bg-black/60 backdrop-blur-xs text-white text-[9px] font-bold rounded-md uppercase">
                    {book.category}
                  </div>
                  {book.pdfUrl && (
                    <div className="absolute top-2 right-2 px-2 py-0.5 bg-emerald-600 text-white text-[9px] font-bold rounded-md">
                      E-Book PDF
                    </div>
                  )}
                </div>

                {/* Book Metadata */}
                <div className="p-3.5 space-y-1.5">
                  <h3 className="font-serif text-xs sm:text-sm font-bold text-[#2D2D2A] dark:text-[#F0F0EA] line-clamp-2">
                    {book.title}
                  </h3>
                  <p className="text-[11px] text-[#8C4A32] dark:text-[#E07A5F] font-semibold truncate">
                    {book.author}
                  </p>
                  <div className="flex items-center justify-between text-[10px] text-[#8E8E85] pt-1">
                    <span>Tersedia: <strong>{book.stock}</strong> buku</span>
                    <span>{book.borrowCount || 0}x dipinjam</span>
                  </div>
                </div>
              </div>

              {/* Actions Footer */}
              <div className="p-3 bg-[#F8F8F5] dark:bg-[#282825] border-t border-[#E0E0D8]/60 dark:border-[#383832]/60 space-y-2">
                <div className="flex items-center gap-1.5">
                  {book.pdfUrl ? (
                    <a
                      href={book.pdfUrl}
                      target="_blank"
                      rel="noopener noreferrer"
                      className="flex-1 py-1.5 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-[11px] rounded-lg text-center transition-colors flex items-center justify-center gap-1"
                    >
                      <Eye className="w-3 h-3" />
                      <span>Baca Digital</span>
                    </a>
                  ) : (
                    <button
                      onClick={() => handleBorrow(book)}
                      className="flex-1 py-1.5 bg-[#5A5A40] hover:bg-[#484833] text-white font-bold text-[11px] rounded-lg transition-colors cursor-pointer flex items-center justify-center gap-1 active:scale-95"
                    >
                      <BookMarked className="w-3 h-3" />
                      <span>Pinjam Fisik</span>
                    </button>
                  )}
                </div>

                {canManage && (
                  <div className="flex items-center justify-end gap-2 pt-1 border-t border-[#E0E0D8]/40 dark:border-[#383832]/40">
                    <button
                      onClick={() => handleOpenEdit(book)}
                      className="text-[10px] text-[#5A5A40] dark:text-[#D1D1BE] hover:underline font-semibold cursor-pointer"
                    >
                      Edit
                    </button>
                    <button
                      onClick={() => handleDelete(book)}
                      className="text-[10px] text-rose-600 hover:underline font-semibold cursor-pointer"
                    >
                      Hapus
                    </button>
                  </div>
                )}
              </div>
            </div>
          ))}
        </div>
      )}

      {/* Add / Edit Modal */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-lg rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <div className="flex items-center gap-2">
                <div className="w-9 h-9 rounded-xl bg-[#5A5A40] text-white flex items-center justify-center font-bold">
                  <BookOpen className="w-5 h-5" />
                </div>
                <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                  {editingItem ? 'Edit Data Buku' : 'Tambah Buku Perpustakaan'}
                </h3>
              </div>
              <button
                onClick={() => setIsModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-white rounded-lg cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSubmit} className="p-5 space-y-4 max-h-[75vh] overflow-y-auto">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Judul Buku *
                </label>
                <input
                  type="text"
                  required
                  value={form.title}
                  onChange={e => setForm({ ...form, title: e.target.value })}
                  placeholder="Contoh: Fisika Kuantum & Penerapan Modern"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Nama Penulis / Pengarang *
                  </label>
                  <input
                    type="text"
                    required
                    value={form.author}
                    onChange={e => setForm({ ...form, author: e.target.value })}
                    placeholder="Contoh: Prof. Dr. Yohanes Surya"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Kategori Buku *
                  </label>
                  <select
                    value={form.category}
                    onChange={e => setForm({ ...form, category: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-semibold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    <option value="Pelajaran">Buku Pelajaran</option>
                    <option value="Referensi">Buku Referensi</option>
                    <option value="Novel">Novel / Cerpen</option>
                    <option value="Ensiklopedia">Ensiklopedia</option>
                  </select>
                </div>
              </div>

              <div className="grid grid-cols-3 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Tahun Terbit
                  </label>
                  <input
                    type="number"
                    value={form.year}
                    onChange={e => setForm({ ...form, year: Number(e.target.value) })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    ISBN
                  </label>
                  <input
                    type="text"
                    value={form.isbn}
                    onChange={e => setForm({ ...form, isbn: e.target.value })}
                    placeholder="978-602-..."
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Stok Fisik
                  </label>
                  <input
                    type="number"
                    min={0}
                    value={form.stock}
                    onChange={e => setForm({ ...form, stock: Number(e.target.value) })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <MediaUploadField
                label="Foto Sampul Buku (Cover URL / Upload)"
                value={form.coverUrl}
                onChange={url => setForm({ ...form, coverUrl: url })}
                placeholder="https://..."
              />

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Link PDF / E-Book Digital (Opsional)
                </label>
                <input
                  type="url"
                  value={form.pdfUrl}
                  onChange={e => setForm({ ...form, pdfUrl: e.target.value })}
                  placeholder="https://drive.google.com/... atau link file PDF"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Sinopsis / Ringkasan Buku
                </label>
                <textarea
                  rows={3}
                  value={form.description}
                  onChange={e => setForm({ ...form, description: e.target.value })}
                  placeholder="Ringkasan isi buku..."
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex items-center justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsModalOpen(false)}
                  className="px-4 py-2.5 text-xs font-bold text-[#8E8E85] hover:bg-[#F5F5F0] dark:hover:bg-[#282825] rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 bg-[#5A5A40] hover:bg-[#484833] text-white text-xs font-bold rounded-xl shadow-md cursor-pointer"
                >
                  {editingItem ? 'Simpan Perubahan' : 'Tambah Buku'}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
