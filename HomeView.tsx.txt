                    onChange={e => setFormData({ ...formData, akreditasi: e.target.value })}
                    className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  />
                </div>
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Slogan / Tagline</label>
                <input
                  type="text"
                  value={formData.tagline}
                  onChange={e => setFormData({ ...formData, tagline: e.target.value })}
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Link Akun TikTok Resmi Sekolah</label>
                <input
                  type="url"
                  value={formData.tiktokUrl}
                  onChange={e => setFormData({ ...formData, tiktokUrl: e.target.value })}
                  placeholder="https://www.tiktok.com/@namasekolah"
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Visi Sekolah</label>
                <textarea
                  rows={3}
                  value={formData.visi}
                  onChange={e => setFormData({ ...formData, visi: e.target.value })}
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Misi Sekolah (Satu baris per poin)</label>
                <textarea
                  rows={4}
                  value={formData.misiText}
                  onChange={e => setFormData({ ...formData, misiText: e.target.value })}
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Sejarah Sekolah</label>
                <textarea
                  rows={4}
                  value={formData.sejarah}
                  onChange={e => setFormData({ ...formData, sejarah: e.target.value })}
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div className="flex justify-end gap-3 pt-4 border-t border-[#E0E0D8] dark:border-[#383832]">
                <button
                  type="button"
                  onClick={() => setIsEditingInfo(false)}
                  className="px-5 py-2.5 rounded-xl border border-[#E0E0D8] text-xs font-bold"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 bg-[#5A5A40] text-white rounded-xl text-xs font-bold uppercase shadow-sm"
                >
                  Simpan Perubahan
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 🔹 ADD/EDIT ACTIVITY MODAL (WITH BATCH MULTI-MEDIA UPLOAD) */}
      {isActivityModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm overflow-y-auto">
          <div className="bg-white dark:bg-[#242422] border border-[#E0E0D8] dark:border-[#383832] rounded-3xl max-w-xl w-full p-6 shadow-2xl space-y-5 my-auto max-h-[90vh] overflow-y-auto">
            <div className="flex items-center justify-between border-b border-[#E0E0D8] dark:border-[#383832] pb-4">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                {editingActivity ? 'Ubah Kegiatan Sekolah' : 'Upload Foto / Video Kegiatan'}
              </h3>
              <button
                onClick={() => setIsActivityModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:bg-[#F5F5F0] dark:hover:bg-[#1E1E1C] rounded-xl"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveActivity} className="space-y-4">
              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Judul Kegiatan</label>
                <input
                  type="text"
                  value={activityForm.title}
                  onChange={e => setActivityForm({ ...activityForm, title: e.target.value })}
                  placeholder="Contoh: Upacara Hari Pahlawan Nasional"
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  required
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Kategori</label>
                  <select
                    value={activityForm.category}
                    onChange={e => setActivityForm({ ...activityForm, category: e.target.value })}
                    className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  >
                    {(activityCategories || []).map(c => (
                      <option key={c} value={c}>{c}</option>
                    ))}
                  </select>
                </div>
                <div>
                  <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Tanggal</label>
                  <input
                    type="text"
                    value={activityForm.date}
                    onChange={e => setActivityForm({ ...activityForm, date: e.target.value })}
                    className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  />
                </div>
              </div>

              {/* Multi-Media Upload Component */}
              <MediaUploadField
                label="Foto / Video Kegiatan (Bisa Upload Beberapa Sekaligus)"
                value={activityForm.mediaUrl}
                mediaType={activityForm.mediaType}
                onChange={(url, type) => {
                  setActivityForm(prev => ({
                    ...prev,
                    mediaUrl: url,
                    mediaType: type || 'image',
                    batchUrls: [...prev.batchUrls, url]
                  }));
                }}
              />

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Deskripsi / Cerita Singkat</label>
                <textarea
                  rows={3}
                  value={activityForm.description}
                  onChange={e => setActivityForm({ ...activityForm, description: e.target.value })}
                  placeholder="Ceritakan momen seru kegiatan ini..."
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div className="flex justify-end gap-3 pt-4 border-t border-[#E0E0D8] dark:border-[#383832]">
                <button
                  type="button"
                  onClick={() => setIsActivityModalOpen(false)}
                  className="px-5 py-2.5 rounded-xl border border-[#E0E0D8] text-xs font-bold"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 bg-[#5A5A40] text-white rounded-xl text-xs font-bold uppercase shadow-sm"
                >
                  Simpan & Publikasikan
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 🔹 ADD/EDIT ANNOUNCEMENT MODAL */}
      {isAnnouncementModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm overflow-y-auto">
          <div className="bg-white dark:bg-[#242422] border border-[#E0E0D8] dark:border-[#383832] rounded-3xl max-w-xl w-full p-6 shadow-2xl space-y-5 my-auto">
            <div className="flex items-center justify-between border-b border-[#E0E0D8] dark:border-[#383832] pb-4">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                {editingAnnouncement ? 'Ubah Pengumuman' : 'Buat Pengumuman Baru'}
              </h3>
              <button
                onClick={() => setIsAnnouncementModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:bg-[#F5F5F0] dark:hover:bg-[#1E1E1C] rounded-xl"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveAnnouncement} className="space-y-4">
              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Judul Pengumuman</label>
                <input
                  type="text"
                  value={announcementForm.title}
                  onChange={e => setAnnouncementForm({ ...announcementForm, title: e.target.value })}
                  placeholder="Contoh: Jadwal Ujian Tengah Semester Genap"
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  required
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Kategori</label>
                  <input
                    type="text"
                    value={announcementForm.category}
                    onChange={e => setAnnouncementForm({ ...announcementForm, category: e.target.value })}
                    className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  />
                </div>
                <div>
                  <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Penulis / Penerbit</label>
                  <input
                    type="text"
                    value={announcementForm.author}
                    onChange={e => setAnnouncementForm({ ...announcementForm, author: e.target.value })}
                    className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  />
                </div>
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Isi Pengumuman</label>
                <textarea
                  rows={5}
                  value={announcementForm.content}
                  onChange={e => setAnnouncementForm({ ...announcementForm, content: e.target.value })}
                  placeholder="Rincian informasi pengumuman..."
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  required
                />
              </div>

              <div className="flex items-center gap-2">
                <input
                  type="checkbox"
                  id="ann-pinned"
                  checked={announcementForm.isPinned}
                  onChange={e => setAnnouncementForm({ ...announcementForm, isPinned: e.target.checked })}
                  className="rounded border-[#E0E0D8]"
                />
                <label htmlFor="ann-pinned" className="text-xs text-[#2D2D2A] dark:text-[#EAEAE2] cursor-pointer">
                  Sematkan pengumuman ini di bagian paling atas
                </label>
              </div>

              <div className="flex justify-end gap-3 pt-4 border-t border-[#E0E0D8] dark:border-[#383832]">
                <button
                  type="button"
                  onClick={() => setIsAnnouncementModalOpen(false)}
                  className="px-5 py-2.5 rounded-xl border border-[#E0E0D8] text-xs font-bold"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 bg-[#5A5A40] text-white rounded-xl text-xs font-bold uppercase shadow-sm"
                >
                  Terbitkan Pengumuman
                </button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* 🔹 ADD/EDIT FACILITY MODAL */}
      {isFacilityModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm overflow-y-auto">
          <div className="bg-white dark:bg-[#242422] border border-[#E0E0D8] dark:border-[#383832] rounded-3xl max-w-lg w-full p-6 shadow-2xl space-y-5 my-auto">
            <div className="flex items-center justify-between border-b border-[#E0E0D8] dark:border-[#383832] pb-4">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                {editingFacility ? 'Ubah Data Fasilitas' : 'Tambah Fasilitas Sekolah'}
              </h3>
              <button
                onClick={() => setIsFacilityModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:bg-[#F5F5F0] dark:hover:bg-[#1E1E1C] rounded-xl"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSaveFacility} className="space-y-4">
              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Nama Fasilitas</label>
                <input
                  type="text"
                  value={facilityForm.nama}
                  onChange={e => setFacilityForm({ ...facilityForm, nama: e.target.value })}
                  placeholder="Contoh: Laboratorium Komputer AI"
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                  required
                />
              </div>

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Lokasi Gedung / Lantai</label>
                <input
                  type="text"
                  value={facilityForm.lokasi}
                  onChange={e => setFacilityForm({ ...facilityForm, lokasi: e.target.value })}
                  placeholder="Gedung C Lantai 2"
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <MediaUploadField
                label="Foto Fasilitas (Opsional)"
                value={facilityForm.foto}
                onChange={url => setFacilityForm({ ...facilityForm, foto: url })}
              />

              <div>
                <label className="text-xs font-bold uppercase text-[#8E8E85] block mb-1">Deskripsi Fasilitas</label>
                <textarea
                  rows={3}
                  value={facilityForm.deskripsi}
                  onChange={e => setFacilityForm({ ...facilityForm, deskripsi: e.target.value })}
                  placeholder="Dilengkapi 40 unit PC berkecepatan tinggi..."
                  className="w-full px-4 py-2.5 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-xl text-xs"
                />
              </div>

              <div className="flex justify-end gap-3 pt-4 border-t border-[#E0E0D8] dark:border-[#383832]">
                <button
                  type="button"
                  onClick={() => setIsFacilityModalOpen(false)}
                  className="px-5 py-2.5 rounded-xl border border-[#E0E0D8] text-xs font-bold"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-6 py-2.5 bg-[#5A5A40] text-white rounded-xl text-xs font-bold uppercase shadow-sm"
                >
                  Simpan Fasilitas
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
      {/* 📁 HTML EXPORT MODAL */}
      <HtmlExportModal
        isOpen={isExportModalOpen}
        onClose={() => setIsExportModalOpen(false)}
      />

      {/* 📱 QR CODE MODAL */}
      <QRCodeModal
        isOpen={!!qrModalData}
        onClose={() => setQrModalData(null)}
        data={qrModalData}
      />
    </div>
  );
};
