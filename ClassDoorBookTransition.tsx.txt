import React, { useEffect, useState } from 'react';
import { ClassRoom, EskulItem } from '../types';
import { Sparkles, BookOpen, GraduationCap, Clock, Flame, Users, DoorOpen, Feather } from 'lucide-react';

interface ClassDoorBookTransitionProps {
  classRoom?: ClassRoom | null;
  eskulItem?: EskulItem | null;
  className?: string;
  classGrade?: string;
  academicYear?: string;
  onComplete: () => void;
  onSkip?: () => void;
  onCancel?: () => void;
}

export const ClassDoorBookTransition: React.FC<ClassDoorBookTransitionProps> = ({
  classRoom,
  eskulItem,
  className,
  classGrade,
  academicYear,
  onComplete,
  onSkip,
  onCancel
}) => {
  const isEskul = !!eskulItem;
  
  // Safe resolved values
  const resolvedTitle = isEskul
    ? eskulItem?.name || 'Ekstrakurikuler'
    : classRoom?.name || className || 'Ruang Kelas';

  const resolvedSubtitle = isEskul
    ? `Kategori: ${eskulItem?.category || 'Bakat & Prestasi'}`
    : `${classRoom?.grade || classGrade || 'Kelas'} • TP ${classRoom?.academicYear || academicYear || '2026/2027'}`;

  const resolvedLeaderLabel = isEskul ? 'Pembina / Pelatih' : 'Wali Kelas';
  const resolvedLeader = isEskul
    ? eskulItem?.pembina || eskulItem?.coach || 'Pembina Eskul'
    : classRoom?.perangkat?.waliKelas || 'Bpk/Ibu Guru Wali Kelas';

  const resolvedKetua = isEskul
    ? eskulItem?.ketua || 'Ketua Eskul'
    : classRoom?.perangkat?.ketuaKelas || 'Ketua Kelas';

  // 10 Seconds total duration:
  // Phase 1: 0s - 3.8s (Membuka Pintu Kayu 3D & Efek Cahaya Terang)
  // Phase 2: 3.8s - 8.5s (Membuka Buku Jurnal Kenangan & Pena Menggores Catatan)
  // Phase 3: 8.5s - 10s (Penyambutan Masuk Ruangan)
  const [phase, setPhase] = useState<1 | 2 | 3>(1);
  const [progress, setProgress] = useState<number>(0);
  const [typedText, setTypedText] = useState<string>('');
  const [secondsRemaining, setSecondsRemaining] = useState<number>(10);

  const fullText = isEskul
    ? `Jurnal Prestasi & Dokumentasi ${resolvedTitle}`
    : `Buku Catatan Kenangan & Galeri ${resolvedTitle}`;

  useEffect(() => {
    const startTime = Date.now();
    const duration = 10000; // 10 Detik

    const interval = setInterval(() => {
      const elapsed = Date.now() - startTime;
      const pct = Math.min(100, (elapsed / duration) * 100);
      setProgress(pct);
      
      const left = Math.max(0, Math.ceil((duration - elapsed) / 1000));
      setSecondsRemaining(left);

      if (elapsed < 3800) {
        setPhase(1);
      } else if (elapsed < 8500) {
        setPhase(2);
      } else {
        setPhase(3);
      }

      if (elapsed >= duration) {
        clearInterval(interval);
        onComplete();
      }
    }, 40);

    return () => clearInterval(interval);
  }, [onComplete]);

  // Typing effect for the book phase
  useEffect(() => {
    if (phase >= 2) {
      let currentIndex = 0;
      const typeInterval = setInterval(() => {
        if (currentIndex <= fullText.length) {
          setTypedText(fullText.slice(0, currentIndex));
          currentIndex++;
        } else {
          clearInterval(typeInterval);
        }
      }, 55);

      return () => clearInterval(typeInterval);
    }
  }, [phase, fullText]);

  return (
    <div className="fixed inset-0 z-[9999] flex items-center justify-center bg-[#141412] text-white overflow-hidden select-none">
      {/* Ambient Classroom Glow Background */}
      <div className="absolute inset-0 bg-radial from-[#5A5A40]/30 via-[#22221E]/90 to-[#0D0D0B] pointer-events-none" />

      {/* Floating Sparkle Particles */}
      <div className="absolute inset-0 overflow-hidden pointer-events-none opacity-40">
        <div className="absolute top-1/4 left-1/5 w-2 h-2 bg-amber-200 rounded-full blur-[1px] animate-ping" />
        <div className="absolute top-1/3 right-1/4 w-3 h-3 bg-amber-100 rounded-full blur-[1px] animate-pulse" />
        <div className="absolute bottom-1/3 left-1/3 w-2.5 h-2.5 bg-yellow-300 rounded-full blur-[1px] animate-bounce" />
        <div className="absolute top-2/3 right-1/5 w-2 h-2 bg-amber-400 rounded-full blur-[1px] animate-pulse" />
      </div>

      {/* Main Container */}
      <div className="relative w-full max-w-4xl h-[600px] px-4 flex flex-col items-center justify-center">

        {/* 🚪 FASE 1: MEMBUKA PINTU RUANGAN 3D (0s - 3.8s) */}
        {phase === 1 && (
          <div className="relative w-full max-w-md h-[460px] flex flex-col items-center justify-center perspective-1200 preserve-3d animate-fadeIn">
            {/* Wooden Door Frame */}
            <div className="relative w-72 sm:w-80 h-96 bg-[#2B1B17] border-4 border-[#8B5A2B] rounded-t-2xl shadow-2xl p-2 flex justify-between overflow-hidden">
              
              {/* Classroom Nameplate on Top of Door */}
              <div className="absolute -top-3.5 left-1/2 -translate-x-1/2 z-30 bg-gradient-to-r from-amber-700 via-amber-600 to-amber-800 border-2 border-amber-300 px-5 py-1 rounded-full shadow-lg text-center whitespace-nowrap">
                <span className="font-serif text-xs sm:text-sm font-bold text-amber-50 tracking-wider uppercase flex items-center gap-1.5">
                  {isEskul ? <Flame className="w-3.5 h-3.5 text-amber-300" /> : <DoorOpen className="w-3.5 h-3.5 text-amber-300" />}
                  {resolvedTitle}
                </span>
              </div>

              {/* Ambient Light from inside room */}
              <div className="absolute inset-0 bg-gradient-to-t from-amber-200/90 via-amber-100/70 to-yellow-50 z-0 flex flex-col items-center justify-center p-6 text-center">
                <div className="w-16 h-16 rounded-2xl bg-amber-600/10 border border-amber-600/30 flex items-center justify-center text-amber-900 mb-2 animate-pulse">
                  {isEskul ? <Users className="w-8 h-8" /> : <GraduationCap className="w-8 h-8" />}
                </div>
                <h3 className="font-serif text-lg font-bold text-stone-900">{resolvedTitle}</h3>
                <p className="text-xs text-stone-700 mt-1">{resolvedSubtitle}</p>
                <div className="mt-3 text-[11px] bg-white/80 px-3 py-1.5 rounded-lg text-stone-800 border border-stone-300 shadow-2xs">
                  {resolvedLeaderLabel}: <strong>{resolvedLeader}</strong>
                </div>
              </div>

              {/* Left Door Leaf (Opens backward to left) */}
              <div className="relative w-1/2 h-full bg-gradient-to-b from-[#6A4126] via-[#50321D] to-[#362113] border-r border-[#24150C] rounded-tl-xl shadow-lg z-10 origin-left anim-door-left flex flex-col justify-center p-3">
                <div className="w-full h-24 border border-amber-900/60 rounded-md bg-black/10 mb-4" />
                <div className="w-full h-36 border border-amber-900/60 rounded-md bg-black/10 flex items-center justify-end pr-1">
                  {/* Brass Door Knob */}
                  <div className="w-4 h-4 rounded-full bg-gradient-to-tr from-amber-600 via-yellow-300 to-amber-500 shadow-md border border-amber-700 anim-knob-turn" />
                </div>
              </div>

              {/* Right Door Leaf (Opens backward to right) */}
              <div className="relative w-1/2 h-full bg-gradient-to-b from-[#6A4126] via-[#50321D] to-[#362113] border-l border-[#24150C] rounded-tr-xl shadow-lg z-10 origin-right anim-door-right flex flex-col justify-center p-3">
                <div className="w-full h-24 border border-amber-900/60 rounded-md bg-black/10 mb-4" />
                <div className="w-full h-36 border border-amber-900/60 rounded-md bg-black/10 flex items-center justify-start pl-1">
                  {/* Brass Lock Plate */}
                  <div className="w-3 h-5 rounded-xs bg-gradient-to-b from-yellow-500 to-amber-700 shadow-md" />
                </div>
              </div>
            </div>

            {/* Bottom Status text */}
            <div className="mt-6 flex items-center gap-2 px-4 py-2 bg-white/10 backdrop-blur-md rounded-full border border-white/15 text-xs text-amber-200">
              <Sparkles className="w-3.5 h-3.5 text-amber-400 animate-spin" />
              <span>Membuka Ruangan & Mengakses Data Terpusat...</span>
            </div>
          </div>
        )}

        {/* 📖 FASE 2: MEMBUKA BUKU KENANGAN & MENULIS DENGAN PENA (3.8s - 8.5s) */}
        {phase === 2 && (
          <div className="relative w-full max-w-xl flex flex-col items-center justify-center animate-fadeIn">
            
            {/* Open Memory Book 3D */}
            <div className="relative w-full h-80 sm:h-92 bg-[#F4F1EA] text-[#2D2D2A] rounded-2xl shadow-2xl border-4 border-[#5A402D] p-6 sm:p-8 flex flex-col sm:flex-row gap-6 overflow-hidden anim-page-flutter">
              
              {/* Left Page (Header & Identity) */}
              <div className="w-full sm:w-1/2 border-b sm:border-b-0 sm:border-r border-[#D5CFBE] pb-4 sm:pb-0 sm:pr-4 flex flex-col justify-between">
                <div>
                  <div className="flex items-center gap-2 text-amber-800 mb-2">
                    <BookOpen className="w-5 h-5" />
                    <span className="text-[11px] font-bold uppercase tracking-widest">
                      {isEskul ? 'EKSTRAKURIKULER' : 'BUKU KENANGAN RESMI'}
                    </span>
                  </div>
                  <h2 className="font-serif text-xl sm:text-2xl font-bold text-[#1E1E1C] leading-tight">
                    {resolvedTitle}
                  </h2>
                  <p className="text-xs text-[#6E6E64] mt-1 font-medium">{resolvedSubtitle}</p>
                </div>

                <div className="space-y-1.5 text-xs bg-[#EAE5D8]/70 p-3 rounded-xl border border-[#D5CFBE]/60">
                  <div className="flex justify-between">
                    <span className="text-stone-600">{resolvedLeaderLabel}:</span>
                    <strong className="text-stone-900">{resolvedLeader}</strong>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-stone-600">{isEskul ? 'Ketua' : 'Ketua Kelas'}:</span>
                    <strong className="text-stone-900">{resolvedKetua}</strong>
                  </div>
                </div>
              </div>

              {/* Right Page (Typing Journal Note & Animated Feather Pen) */}
              <div className="w-full sm:w-1/2 sm:pl-2 flex flex-col justify-between relative">
                <div>
                  <div className="text-[10px] text-amber-800 font-bold uppercase tracking-wider mb-2 flex items-center gap-1">
                    <Feather className="w-3.5 h-3.5 text-amber-700" />
                    <span>Catatan Berita Acara:</span>
                  </div>
                  
                  {/* Handwritten Typing Effect */}
                  <p className="font-serif text-base sm:text-lg text-amber-950 font-bold italic leading-relaxed min-h-[70px]">
                    "{typedText}"
                    <span className="inline-block w-1.5 h-5 bg-amber-800 ml-1 animate-pulse" />
                  </p>

                  <p className="text-xs text-stone-600 mt-3 italic">
                    Memuat galeri momen, daftar profil anggota, dan forum interaksi terpusat...
                  </p>
                </div>

                {/* Animated Floating Feather Pen */}
                <div className="absolute -bottom-2 right-2 text-amber-800 opacity-90 anim-pen-glide">
                  <div className="flex items-center gap-1 px-3 py-1 bg-amber-100/90 rounded-full border border-amber-300 text-[10px] font-bold text-amber-900 shadow-xs">
                    <Feather className="w-3.5 h-3.5" />
                    <span>Menulis Jurnal...</span>
                  </div>
                </div>
              </div>
            </div>

            {/* Sub text */}
            <div className="mt-5 flex items-center gap-2 px-4 py-1.5 bg-amber-500/10 border border-amber-400/30 rounded-full text-xs text-amber-200">
              <Sparkles className="w-3.5 h-3.5 text-amber-300 animate-spin" />
              <span>Menyiapkan Kenangan & Arsip Terintegrasi</span>
            </div>
          </div>
        )}

        {/* 🌟 FASE 3: PENYAMBUTAN & MEMASUKI RUANGAN (8.5s - 10s) */}
        {phase === 3 && (
          <div className="text-center space-y-4 animate-scaleUp">
            <div className="w-20 h-20 mx-auto rounded-3xl bg-gradient-to-tr from-amber-500 to-yellow-300 flex items-center justify-center text-stone-900 shadow-2xl animate-bounce">
              <Sparkles className="w-10 h-10" />
            </div>
            <h2 className="font-serif text-2xl sm:text-3xl font-bold text-white tracking-wide">
              Selamat Datang di {resolvedTitle}!
            </h2>
            <p className="text-sm text-amber-200/90 max-w-md mx-auto">
              Portal ruangan telah siap dibuka. Menampilkan seluruh data, galeri, dan profil siswa...
            </p>
          </div>
        )}

        {/* ⏳ BOTTOM PROGRESS BAR & TIMER CONTROLS */}
        <div className="absolute bottom-6 left-6 right-6 max-w-xl mx-auto flex flex-col gap-2 z-40">
          <div className="flex items-center justify-between text-xs text-stone-400">
            <span className="flex items-center gap-1.5">
              <Clock className="w-3.5 h-3.5 text-amber-400" />
              <span>Transisi Sinematik 10 Detik ({secondsRemaining}s)</span>
            </span>
            <span className="font-mono text-amber-300 font-bold">{Math.round(progress)}%</span>
          </div>

          <div className="w-full h-1.5 bg-stone-800 rounded-full overflow-hidden border border-white/10">
            <div
              className="h-full bg-gradient-to-r from-amber-600 via-yellow-400 to-amber-300 transition-all duration-75 rounded-full"
              style={{ width: `${progress}%` }}
            />
          </div>

          <div className="flex items-center justify-between pt-1">
            {onCancel ? (
              <button
                type="button"
                onClick={onCancel}
                className="text-stone-400 hover:text-white text-xs underline cursor-pointer hover:scale-105 transition-all"
              >
                &larr; Batalkan
              </button>
            ) : <span />}

            {onSkip && (
              <button
                type="button"
                onClick={onSkip}
                className="px-3 py-1 bg-white/10 hover:bg-white/20 text-white rounded-lg text-xs font-semibold cursor-pointer hover:scale-105 transition-all border border-white/20 shadow-xs"
              >
                Lewati Animasi &raquo;
              </button>
            )}
          </div>
        </div>

      </div>
    </div>
  );
};
