import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { TeacherEvaluationNote } from '../types';
import { 
  FileText, 
  Plus, 
  Edit3, 
  Trash2, 
  CheckCircle, 
  Search, 
  User, 
  Sparkles, 
  Award, 
  Lock, 
  ShieldCheck,
  X
} from 'lucide-react';

export const TeacherNotesView: React.FC = () => {
  const { 
    teacherNotes, 
    addTeacherNote, 
    deleteTeacherNote, 
    classes,
    authSession,
    requestDeleteConfirm,
    hasAdminScope,
    showToast
  } = useApp();

  const [selectedClass, setSelectedClass] = useState<string>('Semua');
  const [isModalOpen, setIsModalOpen] = useState<boolean>(false);

  const [form, setForm] = useState({
    studentName: '',
    className: classes[0]?.name || 'XII MIPA 1',
    teacherName: authSession.name || 'Wali Kelas',
    subject: 'Wali Kelas & Bimbingan Konseling',
    category: 'Prestasi' as TeacherEvaluationNote['category'],
    note: '',
    isPrivate: true
  });

  const canManage = hasAdminScope('teacher_notes') || authSession.role === 'super_admin' || authSession.role === 'class_admin';

  const filteredNotes = (teacherNotes || []).filter(n => {
    return selectedClass === 'Semua' || n.className === selectedClass;
  });

  const handleSave = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.studentName.trim() || !form.note.trim()) return;

    addTeacherNote({
      studentName: form.studentName.trim(),
      className: form.className.trim(),
      teacherName: form.teacherName.trim(),
      subject: form.subject.trim(),
      category: form.category,
      note: form.note.trim(),
      isPrivate: form.isPrivate
    });

    setIsModalOpen(false);
    setForm({
      studentName: '',
      className: classes[0]?.name || 'XII MIPA 1',
      teacherName: authSession.name || 'Wali Kelas',
      subject: 'Wali Kelas & Bimbingan Konseling',
      category: 'Prestasi',
      note: '',
      isPrivate: true
    });
    showToast('Catatan evaluasi karakter berhasil disimpan.', 'success');
  };

  const handleDelete = (note: TeacherEvaluationNote) => {
    requestDeleteConfirm({
      itemName: `Catatan "${note.studentName}"`,
      message: `Hapus catatan evaluasi untuk ${note.studentName}?`,
      onConfirm: () => deleteTeacherNote(note.id)
    });
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#5A5A40] via-[#484833] to-[#2B2B1E] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <FileText className="w-4 h-4" />
            <span>Menu 17: Rekam Jejak Karakter Siswa</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            Catatan Guru & Evaluasi Perkembangan
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Pencatatan rekam jejak perilaku, apresiasi prestasi sikap, pembinaan kedisiplinan, dan rekomendasi kepribadian dari dewan guru dan wali kelas.
          </p>

          {canManage && (
            <div className="pt-2">
              <button
                onClick={() => setIsModalOpen(true)}
                className="inline-flex items-center gap-2 px-4 py-2.5 bg-amber-400 hover:bg-amber-300 text-[#2D2D2A] font-bold rounded-2xl text-xs sm:text-sm shadow-md transition-transform active:scale-95 cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Tulis Catatan Evaluasi</span>
              </button>
            </div>
          )}
        </div>
      </div>

      {/* Filter Tabs */}
      <div className="flex items-center gap-2 overflow-x-auto pb-1">
        <button
          onClick={() => setSelectedClass('Semua')}
          className={`px-3.5 py-1.5 rounded-xl text-xs font-bold transition-all cursor-pointer shrink-0 ${
            selectedClass === 'Semua'
              ? 'bg-[#5A5A40] text-white shadow-xs'
              : 'bg-white dark:bg-[#1E1E1C] text-[#8E8E85] border border-[#E0E0D8] dark:border-[#383832]'
          }`}
        >
          Semua Kelas
        </button>
        {classes.map(cls => (
          <button
            key={cls.id}
            onClick={() => setSelectedClass(cls.name)}
            className={`px-3.5 py-1.5 rounded-xl text-xs font-bold transition-all cursor-pointer shrink-0 ${
              selectedClass === cls.name
                ? 'bg-[#5A5A40] text-white shadow-xs'
                : 'bg-white dark:bg-[#1E1E1C] text-[#8E8E85] border border-[#E0E0D8] dark:border-[#383832]'
            }`}
          >
            {cls.name}
          </button>
        ))}
      </div>

      {/* Grid of Notes */}
      {filteredNotes.length === 0 ? (
        <div className="text-center py-16 bg-white dark:bg-[#1E1E1C] rounded-3xl border border-[#E0E0D8] dark:border-[#383832] p-8">
          <FileText className="w-12 h-12 text-[#8E8E85] mx-auto mb-3 opacity-40" />
          <p className="text-base font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">Belum ada catatan evaluasi guru</p>
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {filteredNotes.map(note => {
            const isPraise = note.category === 'Prestasi' || note.category === 'Sikap Positif';
            const isNeedAttention = note.category === 'Perlu Perhatian' || note.category === 'Pelanggaran';

            return (
              <div
                key={note.id}
                className="bg-white dark:bg-[#1E1E1C] p-6 rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm flex flex-col justify-between space-y-4"
              >
                <div className="space-y-3">
                  <div className="flex items-center justify-between">
                    <span className={`px-2.5 py-0.5 rounded-md text-[10px] font-bold ${
                      isPraise
                        ? 'bg-emerald-100 dark:bg-emerald-950/50 text-emerald-800 dark:text-emerald-300'
                        : isNeedAttention
                          ? 'bg-rose-100 dark:bg-rose-950/50 text-rose-800 dark:text-rose-300'
                          : 'bg-amber-100 dark:bg-amber-950/50 text-amber-800 dark:text-amber-300'
                    }`}>
                      {note.category}
                    </span>
                    <span className="text-[11px] text-[#8E8E85]">{note.date}</span>
                  </div>

                  <div>
                    <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
                      {note.studentName}
                    </h3>
                    <p className="text-xs font-semibold text-[#8C4A32] dark:text-[#E07A5F]">
                      Kelas {note.className}
                    </p>
                  </div>

                  <p className="text-xs sm:text-sm text-[#2D2D2A] dark:text-[#ECECE4] leading-relaxed bg-[#F8F8F5] dark:bg-[#282825] p-3.5 rounded-xl border border-[#E0E0D8]/60 dark:border-[#383832]/60 whitespace-pre-wrap">
                    "{note.note}"
                  </p>

                  <div className="text-[11px] text-[#8E8E85] pt-1">
                    Guru Penilai: <strong className="text-[#5A5A40] dark:text-[#A4A485]">{note.teacherName}</strong> ({note.subject})
                  </div>
                </div>

                {canManage && (
                  <div className="pt-3 border-t border-[#E0E0D8]/60 dark:border-[#383832]/60 flex items-center justify-end">
                    <button
                      onClick={() => handleDelete(note)}
                      className="text-xs text-rose-600 hover:underline font-semibold cursor-pointer"
                    >
                      Hapus Catatan
                    </button>
                  </div>
                )}
              </div>
            );
          })}
        </div>
      )}

      {/* Modal */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-md rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                Tulis Catatan Evaluasi Siswa
              </h3>
              <button onClick={() => setIsModalOpen(false)} className="p-2 text-[#8E8E85] cursor-pointer">
                <X className="w-5 h-5" />
              </button>
            </div>
            <form onSubmit={handleSave} className="p-5 space-y-4">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Nama Siswa *</label>
                <input
                  type="text"
                  required
                  value={form.studentName}
                  onChange={e => setForm({ ...form, studentName: e.target.value })}
                  placeholder="Nama Lengkap Siswa"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Kelas *</label>
                  <select
                    value={form.className}
                    onChange={e => setForm({ ...form, className: e.target.value })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-bold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    {classes.map(c => (
                      <option key={c.id} value={c.name}>{c.name}</option>
                    ))}
                  </select>
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Kategori *</label>
                  <select
                    value={form.category}
                    onChange={e => setForm({ ...form, category: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-bold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    <option value="Prestasi">Prestasi</option>
                    <option value="Sikap Positif">Sikap Positif</option>
                    <option value="Perlu Perhatian">Perlu Perhatian</option>
                    <option value="Pelanggaran">Pelanggaran</option>
                  </select>
                </div>
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Mata Pelajaran / Peran Guru</label>
                <input
                  type="text"
                  value={form.subject}
                  onChange={e => setForm({ ...form, subject: e.target.value })}
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Isi Catatan & Rekomendasi *</label>
                <textarea
                  required
                  rows={4}
                  value={form.note}
                  onChange={e => setForm({ ...form, note: e.target.value })}
                  placeholder="Deskripsikan sikap, progres belajar, atau hal yang perlu ditingkatkan..."
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex justify-end gap-2">
                <button type="button" onClick={() => setIsModalOpen(false)} className="px-4 py-2 text-xs font-bold text-[#8E8E85]">Batal</button>
                <button type="submit" className="px-5 py-2 bg-[#5A5A40] text-white text-xs font-bold rounded-xl">Simpan Catatan</button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
