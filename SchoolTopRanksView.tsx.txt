import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { SchoolTopRankItem } from '../types';
import { 
  Crown, 
  Trophy, 
  Medal, 
  Award, 
  Plus, 
  Edit3, 
  Trash2, 
  Sparkles, 
  Star, 
  User, 
  Search, 
  School,
  X
} from 'lucide-react';
import { MediaUploadField } from '../components/MediaUploadField';

export const SchoolTopRanksView: React.FC = () => {
  const { 
    schoolTopRanks, 
    addSchoolTopRank, 
    updateSchoolTopRank, 
    deleteSchoolTopRank, 
    classes,
    authSession,
    requestDeleteConfirm,
    hasAdminScope
  } = useApp();

  const [selectedSemester, setSelectedSemester] = useState<string>('Semua');
  const [isModalOpen, setIsModalOpen] = useState<boolean>(false);
  const [editingItem, setEditingItem] = useState<SchoolTopRankItem | null>(null);

  const [form, setForm] = useState({
    studentName: '',
    className: classes[0]?.name || 'XII MIPA 1',
    overallRank: 1,
    averageScore: 95.8,
    academicYear: '2026/2027',
    semester: 'Ganjil' as SchoolTopRankItem['semester'],
    photoUrl: '',
    specialAchievement: ''
  });

  const canManage = hasAdminScope('school_ranks') || authSession.role === 'super_admin';

  const filteredRanks = schoolTopRanks.filter(item => {
    return selectedSemester === 'Semua' || item.semester === selectedSemester;
  }).sort((a, b) => a.overallRank - b.overallRank);

  const handleOpenAdd = () => {
    setEditingItem(null);
    setForm({
      studentName: '',
      className: classes[0]?.name || 'XII MIPA 1',
      overallRank: (schoolTopRanks.length % 3) + 1,
      averageScore: 95.5,
      academicYear: '2026/2027',
      semester: 'Ganjil',
      photoUrl: '',
      specialAchievement: ''
    });
    setIsModalOpen(true);
  };

  const handleOpenEdit = (item: SchoolTopRankItem) => {
    setEditingItem(item);
    setForm({
      studentName: item.studentName,
      className: item.className,
      overallRank: item.overallRank,
      averageScore: item.averageScore,
      academicYear: item.academicYear,
      semester: item.semester,
      photoUrl: item.photoUrl || '',
      specialAchievement: item.specialAchievement || ''
    });
    setIsModalOpen(true);
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.studentName.trim()) return;

    if (editingItem) {
      updateSchoolTopRank(editingItem.id, {
        studentName: form.studentName.trim(),
        className: form.className.trim(),
        overallRank: Number(form.overallRank),
        averageScore: Number(form.averageScore),
        academicYear: form.academicYear.trim(),
        semester: form.semester,
        photoUrl: form.photoUrl.trim() || undefined,
        specialAchievement: form.specialAchievement.trim() || undefined
      });
    } else {
      addSchoolTopRank({
        studentName: form.studentName.trim(),
        className: form.className.trim(),
        overallRank: Number(form.overallRank),
        averageScore: Number(form.averageScore),
        academicYear: form.academicYear.trim(),
        semester: form.semester,
        photoUrl: form.photoUrl.trim() || undefined,
        specialAchievement: form.specialAchievement.trim() || undefined
      });
    }
    setIsModalOpen(false);
  };

  const handleDelete = (item: SchoolTopRankItem) => {
    requestDeleteConfirm({
      itemName: `Peringkat "${item.studentName}"`,
      message: `Hapus catatan juara umum untuk ${item.studentName}?`,
      onConfirm: () => deleteSchoolTopRank(item.id)
    });
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#5A5A40] via-[#484833] to-[#2B2B1E] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <Crown className="w-4 h-4" />
            <span>Menu 13: Papan Penghargaan Bintang Pelajar</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            Juara Umum & Peringkat Bintang Sekolah
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Apresiasi tertinggi untuk siswa-siswi dengan nilai akademik rata-rata tertinggi dan prestasi teladan tingkat sekolah.
          </p>

          {canManage && (
            <div className="pt-2">
              <button
                onClick={handleOpenAdd}
                className="inline-flex items-center gap-2 px-4 py-2.5 bg-amber-400 hover:bg-amber-300 text-[#2D2D2A] font-bold rounded-2xl text-xs sm:text-sm shadow-md transition-transform active:scale-95 cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Tambah Juara Umum</span>
              </button>
            </div>
          )}
        </div>
      </div>

      {/* Top Podium Display (Top 3) */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6 pt-4">
        {filteredRanks.slice(0, 3).map((rankItem, idx) => {
          const isFirst = rankItem.overallRank === 1;
          const isSecond = rankItem.overallRank === 2;
          const isThird = rankItem.overallRank === 3;

          let medalBg = 'bg-amber-400 text-[#2D2D2A]';
          let borderHighlight = 'border-amber-400/60 ring-2 ring-amber-400/20';

          if (isSecond) {
            medalBg = 'bg-slate-300 text-slate-800';
            borderHighlight = 'border-slate-300/60';
          } else if (isThird) {
            medalBg = 'bg-amber-700 text-white';
            borderHighlight = 'border-amber-700/60';
          }

          return (
            <div
              key={rankItem.id}
              className={`bg-white dark:bg-[#1E1E1C] p-6 rounded-[28px] border ${borderHighlight} shadow-lg text-center flex flex-col justify-between relative overflow-hidden group ${
                isFirst ? 'md:-translate-y-3' : ''
              }`}
            >
              <div className="space-y-4">
                {/* Crown or Rank Badge */}
                <div className="relative inline-block mx-auto">
                  <img
                    src={rankItem.photoUrl || 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=300&auto=format&fit=crop&q=80'}
                    alt={rankItem.studentName}
                    className="w-24 h-24 rounded-full object-cover ring-4 ring-[#5A5A40]/20 mx-auto shadow-md"
                  />
                  <div className={`absolute -bottom-2 -right-1 w-8 h-8 rounded-full ${medalBg} flex items-center justify-center font-extrabold text-sm shadow-md ring-2 ring-white dark:ring-[#1E1E1C]`}>
                    #{rankItem.overallRank}
                  </div>
                </div>

                <div>
                  <span className="px-3 py-0.5 rounded-full text-[10px] font-bold bg-[#5A5A40]/10 dark:bg-[#5A5A40]/30 text-[#5A5A40] dark:text-[#D1D1BE] uppercase">
                    Juara Umum #{rankItem.overallRank} Sekolah
                  </span>
                  <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#F0F0EA] mt-1">
                    {rankItem.studentName}
                  </h3>
                  <p className="text-xs font-semibold text-[#8C4A32] dark:text-[#E07A5F]">
                    Kelas {rankItem.className} • TA {rankItem.academicYear} ({rankItem.semester})
                  </p>
                </div>

                <div className="p-3 bg-[#F8F8F5] dark:bg-[#282825] rounded-2xl border border-[#E0E0D8]/60 dark:border-[#383832]/60">
                  <span className="text-[10px] font-bold uppercase text-[#8E8E85] block">Nilai Rata-Rata Rapor</span>
                  <span className="font-serif text-2xl font-extrabold text-[#5A5A40] dark:text-[#A4A485]">
                    {rankItem.averageScore.toFixed(1)}
                  </span>
                </div>

                {rankItem.specialAchievement && (
                  <p className="text-xs text-[#5A5A40] dark:text-[#C5C5B0] italic">
                    "{rankItem.specialAchievement}"
                  </p>
                )}
              </div>

              {canManage && (
                <div className="pt-4 mt-4 border-t border-[#E0E0D8]/60 dark:border-[#383832]/60 flex items-center justify-center gap-3 text-xs font-semibold">
                  <button onClick={() => handleOpenEdit(rankItem)} className="text-[#5A5A40] dark:text-[#D1D1BE] hover:underline cursor-pointer">
                    Edit
                  </button>
                  <span>•</span>
                  <button onClick={() => handleDelete(rankItem)} className="text-rose-600 hover:underline cursor-pointer">
                    Hapus
                  </button>
                </div>
              )}
            </div>
          );
        })}
      </div>

      {/* Add / Edit Modal */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-lg rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                {editingItem ? 'Edit Juara Umum' : 'Tambah Juara Umum Sekolah'}
              </h3>
              <button onClick={() => setIsModalOpen(false)} className="p-2 text-[#8E8E85] cursor-pointer">
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSubmit} className="p-5 space-y-4 max-h-[75vh] overflow-y-auto">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Nama Lengkap Siswa *</label>
                <input
                  type="text"
                  required
                  value={form.studentName}
                  onChange={e => setForm({ ...form, studentName: e.target.value })}
                  placeholder="Contoh: Muhammad Rizky Pratama"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Kelas *</label>
                  <input
                    type="text"
                    required
                    value={form.className}
                    onChange={e => setForm({ ...form, className: e.target.value })}
                    placeholder="XII MIPA 1"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Peringkat Ke *</label>
                  <input
                    type="number"
                    min={1}
                    max={10}
                    required
                    value={form.overallRank}
                    onChange={e => setForm({ ...form, overallRank: Number(e.target.value) })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <div className="grid grid-cols-3 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Nilai Rata-Rata *</label>
                  <input
                    type="number"
                    step="0.01"
                    required
                    value={form.averageScore}
                    onChange={e => setForm({ ...form, averageScore: Number(e.target.value) })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Tahun Ajaran</label>
                  <input
                    type="text"
                    value={form.academicYear}
                    onChange={e => setForm({ ...form, academicYear: e.target.value })}
                    placeholder="2026/2027"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Semester</label>
                  <select
                    value={form.semester}
                    onChange={e => setForm({ ...form, semester: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-bold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    <option value="Ganjil">Ganjil</option>
                    <option value="Genap">Genap</option>
                  </select>
                </div>
              </div>

              <MediaUploadField
                label="Foto Siswa Berprestasi (URL / Upload)"
                value={form.photoUrl}
                onChange={url => setForm({ ...form, photoUrl: url })}
                placeholder="https://..."
              />

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">Prestasi Spesial / Catatan Pujian</label>
                <input
                  type="text"
                  value={form.specialAchievement}
                  onChange={e => setForm({ ...form, specialAchievement: e.target.value })}
                  placeholder="Contoh: Peraih Medali Emas OSN & Bintang Kelas Berturut-turut"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex justify-end gap-2">
                <button type="button" onClick={() => setIsModalOpen(false)} className="px-4 py-2 text-xs font-bold text-[#8E8E85]">Batal</button>
                <button type="submit" className="px-5 py-2 bg-[#5A5A40] text-white text-xs font-bold rounded-xl">Simpan</button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
