import React, { useState, useEffect } from 'react';
import { useApp } from '../context/AppContext';
import { 
  ShieldCheck, 
  UserCheck, 
  GraduationCap, 
  Award,
  Lock, 
  User, 
  Search, 
  AlertCircle, 
  X, 
  CheckCircle2, 
  Info,
  KeyRound,
  Mail,
  Send,
  RefreshCw,
  Clock,
  ArrowRight,
  Sparkles
} from 'lucide-react';

interface AuthModalProps {
  isOpen?: boolean;
  onClose?: () => void;
}

export const AuthModal: React.FC<AuthModalProps> = ({ isOpen, onClose }) => {
  const { 
    isAuthModalOpen, 
    setIsAuthModalOpen, 
    classes, 
    students, 
    eskulList,
    loginStudent, 
    loginClassAdmin, 
    loginEskulAdmin,
    loginSuperAdmin,
    loginWithGoogle,
    checkEmailRegistration,
    sendEmailVerificationCode,
    verifyEmailCodeAndLogin,
    emailVerificationState,
    cancelEmailVerification,
    showToast
  } = useApp();

  const isModalVisible = isOpen !== undefined ? isOpen : isAuthModalOpen;

  // Primary Login Mode: 'email_code' | 'google' | 'legacy_password'
  const [loginMode, setLoginMode] = useState<'email_code' | 'google' | 'legacy_password'>('email_code');

  // 📧 Email Passwordless Flow States
  const [inputEmail, setInputEmail] = useState<string>('');
  const [isSendingCode, setIsSendingCode] = useState<boolean>(false);
  const [verificationCodeInput, setVerificationCodeInput] = useState<string>('');
  const [timeLeftSeconds, setTimeLeftSeconds] = useState<number>(600); // 10 minutes = 600s
  const [resendCooldown, setResendCooldown] = useState<number>(0); // 60 seconds cooldown
  const [emailFlowError, setEmailFlowError] = useState<string>('');

  // 🔑 Legacy Password Form States
  const [legacyRole, setLegacyRole] = useState<'student' | 'class_admin' | 'eskul_admin' | 'super_admin'>('student');
  const [chosenClassId, setChosenClassId] = useState<string>(() => classes[0]?.id || '');
  const [selectedStudentId, setSelectedStudentId] = useState<string>('');
  const [studentPassword, setStudentPassword] = useState<string>('');
  const [chosenAdminClassId, setChosenAdminClassId] = useState<string>(() => classes[0]?.id || '');
  const [classUsername, setClassUsername] = useState<string>('');
  const [classPassword, setClassPassword] = useState<string>('');
  const [chosenAdminEskulId, setChosenAdminEskulId] = useState<string>(() => eskulList[0]?.id || '');
  const [eskulUsername, setEskulUsername] = useState<string>('');
  const [eskulPassword, setEskulPassword] = useState<string>('');
  const [superAdminInputPassword, setSuperAdminInputPassword] = useState<string>('');
  const [legacyError, setLegacyError] = useState<string>('');

  // ⏱️ 10-Minute Countdown Timer for Email Code Expiration
  useEffect(() => {
    if (!emailVerificationState) return;

    const interval = setInterval(() => {
      const remainingMs = emailVerificationState.expiresAt - Date.now();
      const remainingSec = Math.max(0, Math.floor(remainingMs / 1000));
      setTimeLeftSeconds(remainingSec);
      if (remainingSec <= 0) {
        clearInterval(interval);
      }
    }, 1000);

    return () => clearInterval(interval);
  }, [emailVerificationState]);

  // ⏱️ 60-Second Cooldown Timer for Resending Verification Code
  useEffect(() => {
    if (resendCooldown <= 0) return;

    const interval = setInterval(() => {
      setResendCooldown(prev => {
        if (prev <= 1) {
          clearInterval(interval);
          return 0;
        }
        return prev - 1;
      });
    }, 1000);

    return () => clearInterval(interval);
  }, [resendCooldown]);

  if (!isModalVisible) return null;

  const handleClose = () => {
    if (onClose) {
      onClose();
    } else {
      setIsAuthModalOpen(false);
    }
    setEmailFlowError('');
    setLegacyError('');
  };

  // 1. Kirim Kode Verifikasi ke Email (Dengan Validasi Pendaftaran Ketat)
  const handleSendCode = async (e?: React.FormEvent) => {
    if (e) e.preventDefault();
    setEmailFlowError('');

    const cleanEmail = (emailVerificationState?.email || inputEmail).trim().toLowerCase();
    if (!cleanEmail || !cleanEmail.includes('@')) {
      setEmailFlowError('Silakan masukkan alamat email yang valid.');
      return;
    }

    // Step 1: PERIKSA apakah email SUDAH ADA di Daftar Data Login Admin?
    const check = checkEmailRegistration(cleanEmail);
    if (!check.isRegistered || !check.account) {
      // ❌ Belum terdaftar: Berhenti di sini!
      setEmailFlowError('Maaf, email ini belum disetujui. Hubungi Admin Sekolah.');
      return;
    }

    // ✅ Terdaftar: Kirim kode 6 digit via layanan email
    setIsSendingCode(true);
    try {
      const res = await sendEmailVerificationCode(cleanEmail);
      if (!res.success) {
        setEmailFlowError(res.message || 'Gagal mengirim kode verifikasi.');
      } else {
        setVerificationCodeInput(''); // HARUS KOSONG MURNI
        setResendCooldown(60); // Batasi tiap 60 detik
        setTimeLeftSeconds(600); // 10 menit
        setEmailFlowError('');
      }
    } catch (err: any) {
      setEmailFlowError(err.message || 'Terjadi gangguan saat mengirim email verifikasi.');
    } finally {
      setIsSendingCode(false);
    }
  };

  // 2. Verifikasi Kode 6 Digit & Masuk
  const handleVerifyCode = (e: React.FormEvent) => {
    e.preventDefault();
    setEmailFlowError('');

    const targetEmail = (emailVerificationState?.email || inputEmail).trim().toLowerCase();
    const cleanCode = verificationCodeInput.trim();

    if (!cleanCode) {
      setEmailFlowError('Silakan ketik atau tempel 6 digit kode dari email Anda.');
      return;
    }

    if (cleanCode.length !== 6) {
      setEmailFlowError('Kode verifikasi harus tepat berjumlah 6 digit angka.');
      return;
    }

    const res = verifyEmailCodeAndLogin(targetEmail, cleanCode);
    if (!res.success) {
      setEmailFlowError(res.message || '❌ Kode salah atau sudah habis masa berlakunya — tekan Kirim Ulang Kode');
    } else {
      handleClose();
    }
  };

  // Format time MM:SS
  const formatTime = (totalSeconds: number) => {
    const mins = Math.floor(totalSeconds / 60);
    const secs = totalSeconds % 60;
    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
  };

  // Legacy Submit Handlers
  const handleLegacySubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setLegacyError('');

    if (legacyRole === 'student') {
      if (!chosenClassId || !selectedStudentId || !studentPassword) {
        setLegacyError('Lengkapi pilihan kelas, nama siswa, dan kata sandi.');
        return;
      }
      const res = loginStudent(selectedStudentId, studentPassword);
      if (!res.success) setLegacyError(res.message || 'Gagal masuk');
      else handleClose();
    } else if (legacyRole === 'class_admin') {
      if (!chosenAdminClassId || !classUsername.trim() || !classPassword) {
        setLegacyError('Lengkapi kelas, username, dan kata sandi admin kelas.');
        return;
      }
      const res = loginClassAdmin(classUsername.trim(), classPassword, chosenAdminClassId);
      if (!res.success) setLegacyError(res.message || 'Gagal masuk');
      else handleClose();
    } else if (legacyRole === 'eskul_admin') {
      if (!chosenAdminEskulId || !eskulUsername.trim() || !eskulPassword) {
        setLegacyError('Lengkapi eskul, username, dan kata sandi admin eskul.');
        return;
      }
      const res = loginEskulAdmin(chosenAdminEskulId, eskulUsername.trim(), eskulPassword);
      if (!res.success) setLegacyError(res.message || 'Gagal masuk');
      else handleClose();
    } else if (legacyRole === 'super_admin') {
      if (!superAdminInputPassword) {
        setLegacyError('Masukkan kata sandi Admin Utama.');
        return;
      }
      const res = loginSuperAdmin(superAdminInputPassword);
      if (!res.success) setLegacyError(res.message || 'Gagal masuk');
      else handleClose();
    }
  };

  return (
    <div
      id="auth-login-modal"
      className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm animate-in fade-in"
    >
      <div className="relative w-full max-w-lg bg-white dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-[32px] shadow-2xl overflow-hidden p-6 sm:p-8 text-[#2D2D2A] dark:text-[#EAEAE2] max-h-[90vh] overflow-y-auto">
        {/* Modal Header */}
        <div className="flex items-start justify-between pb-4 border-b border-[#E0E0D8] dark:border-[#383832]">
          <div className="space-y-1">
            <span className="text-[10px] font-bold uppercase tracking-widest text-[#5A5A40] dark:text-[#C2C2B0] bg-[#F5F5F0] dark:bg-[#2A2A26] px-2.5 py-0.5 rounded-full border border-[#E0E0D8] dark:border-[#42423B]">
              Sistem Login Terkontrol
            </span>
            <h2 className="font-serif text-xl sm:text-2xl font-bold text-[#2D2D2A] dark:text-[#F0EFEB]">
              Masuk Portal Sekolah
            </h2>
          </div>
          <button
            onClick={handleClose}
            className="text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-[#F0EFEB] p-1.5 rounded-xl hover:bg-[#F5F5F0] dark:hover:bg-[#2A2A26] transition-colors cursor-pointer"
            aria-label="Tutup modal"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        {/* Primary Method Tabs: Email (Kode OTP) vs Google */}
        <div className="grid grid-cols-2 gap-1.5 p-1 bg-[#F5F5F0] dark:bg-[#2A2A26] rounded-2xl border border-[#E0E0D8] dark:border-[#383832] my-4">
          <button
            id="tab-login-email-code"
            type="button"
            onClick={() => {
              setLoginMode('email_code');
              setEmailFlowError('');
            }}
            className={`py-2.5 px-3 rounded-xl text-xs font-bold transition-all flex items-center justify-center gap-1.5 cursor-pointer ${
              loginMode === 'email_code'
                ? 'bg-white dark:bg-[#1E1E1C] text-[#5A5A40] dark:text-[#C2C2B0] shadow-sm border border-[#E0E0D8] dark:border-[#42423B]'
                : 'text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-[#EAEAE2]'
            }`}
          >
            <Mail className="w-4 h-4" />
            <span>📧 Masuk dengan Email</span>
          </button>

          <button
            id="tab-login-google"
            type="button"
            onClick={() => {
              setLoginMode('google');
              setEmailFlowError('');
            }}
            className={`py-2.5 px-3 rounded-xl text-xs font-bold transition-all flex items-center justify-center gap-1.5 cursor-pointer ${
              loginMode === 'google'
                ? 'bg-white dark:bg-[#1E1E1C] text-[#2D2D2A] dark:text-[#F0EFEB] shadow-sm border border-[#E0E0D8] dark:border-[#42423B]'
                : 'text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-[#EAEAE2]'
            }`}
          >
            <svg className="w-3.5 h-3.5" viewBox="0 0 24 24">
              <path fill="#4285F4" d="M23.745 12.27c0-.7-.06-1.4-.19-2.07H12v4.51h6.6c-.29 1.52-1.14 2.82-2.4 3.68v3.05h3.88c2.27-2.09 3.66-5.17 3.66-9.17z" />
              <path fill="#34A853" d="M12 24c3.24 0 5.95-1.08 7.93-2.91l-3.88-3.05c-1.08.72-2.45 1.16-4.05 1.16-3.12 0-5.77-2.1-6.72-4.93H1.25v3.15C3.26 21.36 7.33 24 12 24z" />
              <path fill="#FBBC05" d="M5.28 14.27c-.25-.72-.38-1.49-.38-2.27s.13-1.55.38-2.27V6.58H1.25C.45 8.16 0 9.97 0 12s.45 3.84 1.25 5.42l4.03-3.15z" />
              <path fill="#EA4335" d="M12 4.75c1.77 0 3.35.61 4.6 1.8l3.42-3.42C17.95 1.19 15.24 0 12 0 7.33 0 3.26 2.64 1.25 6.58l4.03 3.15c.95-2.83 3.6-4.98 6.72-4.98z" />
            </svg>
            <span>🔑 Masuk Akun Google</span>
          </button>
        </div>

        {/* ========================================================= */}
        {/* 📧 MODE 1: MASUK DENGAN EMAIL & KODE VERIFIKASI (TANPA PASSWORD) */}
        {/* ========================================================= */}
        {loginMode === 'email_code' && (
          <div className="space-y-4">
            {/* Info Box */}
            <div className="p-3 bg-[#FAF8F5] dark:bg-[#262622] border border-[#E0E0D8] dark:border-[#383832] rounded-2xl flex items-start gap-2.5 text-xs text-[#5A5A40] dark:text-[#C2C2B0]">
              <Info className="w-4 h-4 text-[#8C4A32] flex-shrink-0 mt-0.5" />
              <p className="leading-relaxed">
                <strong>Verifikasi Email Resmi:</strong> Masukkan alamat email Anda yang terdaftar di Data Login Admin. Kode verifikasi 6-digit rahasia akan dikirimkan ke kotak masuk email Anda.
              </p>
            </div>

            {/* Error Message Alert */}
            {emailFlowError && (
              <div className="p-3.5 bg-rose-50 dark:bg-rose-950/40 border border-rose-200 dark:border-rose-800 text-rose-800 dark:text-rose-300 rounded-2xl text-xs flex items-start gap-2.5 animate-in fade-in">
                <AlertCircle className="w-4 h-4 flex-shrink-0 mt-0.5 text-rose-600 dark:text-rose-400" />
                <div className="leading-relaxed font-semibold">{emailFlowError}</div>
              </div>
            )}

            {/* ------------------------------------------------------------- */}
            {/* LANGKAH 1: Pengguna Ketik Email → Tekan Kirim Kode             */}
            {/* ------------------------------------------------------------- */}
            {!emailVerificationState ? (
              <form onSubmit={handleSendCode} className="space-y-4">
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] dark:text-[#EAEAE2] mb-1.5">
                    Alamat Email Terdaftar *
                  </label>
                  <div className="relative">
                    <Mail className="w-4 h-4 absolute left-3.5 top-3 text-[#8E8E85]" />
                    <input
                      id="input-email-login"
                      type="email"
                      required
                      placeholder="Masukkan alamat emailmu..."
                      value={inputEmail}
                      onChange={e => {
                        setInputEmail(e.target.value);
                        setEmailFlowError('');
                      }}
                      className="w-full pl-10 pr-4 py-2.5 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-sm text-[#2D2D2A] dark:text-[#F0EFEB] focus:outline-none focus:border-[#5A5A40]"
                    />
                  </div>
                </div>

                <button
                  id="btn-send-verification-code"
                  type="submit"
                  disabled={isSendingCode || !inputEmail.trim()}
                  className="w-full py-3 px-4 bg-[#5A5A40] hover:bg-[#484833] disabled:opacity-50 text-white text-xs font-bold uppercase tracking-wider rounded-xl shadow-sm flex items-center justify-center gap-2 transition-all cursor-pointer active:scale-95"
                >
                  {isSendingCode ? (
                    <>
                      <RefreshCw className="w-4 h-4 animate-spin" />
                      <span>Memeriksa & Mengirim Kode...</span>
                    </>
                  ) : (
                    <>
                      <Send className="w-4 h-4" />
                      <span>Kirim Kode Verifikasi</span>
                    </>
                  )}
                </button>
              </form>
            ) : (
              /* ------------------------------------------------------------- */
              /* LANGKAH 2: Kotak Masuk Kode 6-Digit (KOSONG MURNI & TANPA BOCORAN) */
              /* ------------------------------------------------------------- */
              <form onSubmit={handleVerifyCode} className="space-y-4 animate-in fade-in">
                {/* Banner Status Pengiriman — TIDAK ADA ANGKA SATU PUN YANG DITAMPILKAN */}
                <div className="p-4 bg-emerald-50 dark:bg-emerald-950/40 border border-emerald-200 dark:border-emerald-800 rounded-2xl text-xs space-y-2">
                  <div className="flex items-center justify-between text-emerald-900 dark:text-emerald-200 font-bold">
                    <span className="flex items-center gap-1.5">
                      <CheckCircle2 className="w-4 h-4 text-emerald-600 dark:text-emerald-400" />
                      <span>Kode Verifikasi Telah Dikirim</span>
                    </span>
                    <span className={`font-mono text-xs px-2 py-0.5 rounded-md ${timeLeftSeconds < 60 ? 'bg-rose-100 text-rose-700' : 'bg-emerald-100 dark:bg-emerald-900/60 text-emerald-800 dark:text-emerald-200'}`}>
                      <Clock className="w-3 h-3 inline mr-1" />
                      {formatTime(timeLeftSeconds)}
                    </span>
                  </div>
                  <p className="text-emerald-800 dark:text-emerald-300 leading-relaxed font-medium">
                    ✅ Kode verifikasi TELAH DIKIRIM ke alamat emailmu (<strong>{emailVerificationState.email}</strong>) — silakan cek Kotak Masuk atau folder Spam/Junk
                  </p>
                </div>

                {/* Input Kode 6 Digit Murni Kosong */}
                <div>
                  <label className="block text-xs font-bold text-[#2D2D2A] dark:text-[#EAEAE2] mb-1.5 text-center">
                    Masukkan 6 Digit Kode Verifikasi *
                  </label>
                  <input
                    id="input-verification-code"
                    type="text"
                    inputMode="numeric"
                    pattern="[0-9]*"
                    required
                    maxLength={6}
                    autoFocus
                    placeholder="• • • • • •"
                    value={verificationCodeInput}
                    onChange={e => {
                      setVerificationCodeInput(e.target.value.replace(/[^0-9]/g, ''));
                      setEmailFlowError('');
                    }}
                    className="w-full text-center text-3xl font-mono font-extrabold tracking-[0.4em] px-4 py-3 bg-[#F8F8F5] dark:bg-[#2A2A26] border-2 border-[#5A5A40] dark:border-[#A3A380] rounded-2xl text-[#2D2D2A] dark:text-[#F0EFEB] focus:outline-none shadow-inner"
                  />
                  <p className="text-[11px] text-[#8E8E85] text-center mt-1.5">
                    Ketik atau tempel 6 digit angka yang Anda terima dari pesan email Anda.
                  </p>
                </div>

                <div className="flex flex-col gap-2 pt-1">
                  <button
                    id="btn-confirm-verification-login"
                    type="submit"
                    disabled={verificationCodeInput.length !== 6}
                    className="w-full py-3 px-4 bg-[#5A5A40] hover:bg-[#484833] disabled:opacity-50 text-white text-xs font-bold uppercase tracking-wider rounded-xl shadow-sm flex items-center justify-center gap-2 transition-all cursor-pointer active:scale-95"
                  >
                    <CheckCircle2 className="w-4 h-4" />
                    <span>Verifikasi & Masuk</span>
                  </button>

                  <div className="flex items-center justify-between text-xs pt-1.5">
                    {/* Tombol Kirim Ulang Kode (Dengan Cooldown 60 Detik) */}
                    <button
                      id="btn-resend-verification-code"
                      type="button"
                      onClick={() => handleSendCode()}
                      disabled={isSendingCode || resendCooldown > 0}
                      className="text-[#5A5A40] dark:text-[#C2C2B0] font-bold hover:underline disabled:opacity-50 disabled:no-underline flex items-center gap-1.5 cursor-pointer"
                    >
                      <RefreshCw className={`w-3.5 h-3.5 ${isSendingCode ? 'animate-spin' : ''}`} />
                      <span>
                        {resendCooldown > 0 
                          ? `Kirim Ulang Kode (${resendCooldown}s)` 
                          : 'Kirim Ulang Kode'
                        }
                      </span>
                    </button>

                    {/* Tombol Ganti Email */}
                    <button
                      type="button"
                      onClick={() => {
                        cancelEmailVerification();
                        setVerificationCodeInput('');
                        setEmailFlowError('');
                      }}
                      className="text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-[#F0EFEB] font-medium hover:underline cursor-pointer"
                    >
                      Ganti Alamat Email
                    </button>
                  </div>
                </div>
              </form>
            )}
          </div>
        )}

        {/* ========================================================= */}
        {/* 🔑 MODE 2: MASUK DENGAN AKUN GOOGLE (OFFICIAL OAUTH) */}
        {/* ========================================================= */}
        {loginMode === 'google' && (
          <div className="space-y-4">
            <div className="p-3.5 bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-800/40 rounded-2xl text-xs text-emerald-800 dark:text-emerald-300 flex items-start gap-2.5">
              <CheckCircle2 className="w-4 h-4 flex-shrink-0 mt-0.5 text-emerald-600 dark:text-emerald-400" />
              <div>
                <p className="font-bold">Masuk Otomatis dengan Akun Google</p>
                <p className="text-[11px] text-emerald-700 dark:text-emerald-400/90 leading-relaxed mt-0.5">
                  Sistem langsung memvalidasi apakah email Google Anda terdaftar di Data Login Admin. Jika terdaftar, Anda otomatis diizinkan masuk sesuai peran yang telah disetujui.
                </p>
              </div>
            </div>

            {/* Google Sign In Direct Action */}
            <button
              id="btn-login-google-passwordless"
              type="button"
              onClick={() => {
                const userEmail = prompt(
                  '🔐 Masuk Dengan Google (accounts.google.com)\n\nMasukkan alamat email Google Anda untuk verifikasi identitas resmi (misal: mumumimi353@gmail.com, guru@sekolah.sch.id, atau siswa@gmail.com):',
                  'mumumimi353@gmail.com'
                );
                if (userEmail && userEmail.trim()) {
                  const cleanEmail = userEmail.trim().toLowerCase();
                  const userName = cleanEmail.split('@')[0].replace(/[._]/g, ' ');
                  const avatar = `https://api.dicebear.com/7.x/initials/svg?seed=${encodeURIComponent(cleanEmail)}&backgroundColor=047857,059669,10b981`;
                  loginWithGoogle({
                    email: cleanEmail,
                    name: userName,
                    avatarUrl: avatar,
                    sub: `google_oauth_${Date.now()}`
                  });
                  handleClose();
                }
              }}
              className="w-full py-3.5 px-4 rounded-xl text-xs sm:text-sm font-bold bg-white dark:bg-[#2A2A26] hover:bg-[#F5F5F0] dark:hover:bg-[#32322D] text-[#2D2D2A] dark:text-[#EAEAE2] border-2 border-[#E0E0D8] dark:border-[#42423B] shadow-xs flex items-center justify-center gap-3 transition-all cursor-pointer active:scale-95"
            >
              <svg className="w-4 h-4" viewBox="0 0 24 24">
                <path fill="#4285F4" d="M23.745 12.27c0-.7-.06-1.4-.19-2.07H12v4.51h6.6c-.29 1.52-1.14 2.82-2.4 3.68v3.05h3.88c2.27-2.09 3.66-5.17 3.66-9.17z" />
                <path fill="#34A853" d="M12 24c3.24 0 5.95-1.08 7.93-2.91l-3.88-3.05c-1.08.72-2.45 1.16-4.05 1.16-3.12 0-5.77-2.1-6.72-4.93H1.25v3.15C3.26 21.36 7.33 24 12 24z" />
                <path fill="#FBBC05" d="M5.28 14.27c-.25-.72-.38-1.49-.38-2.27s.13-1.55.38-2.27V6.58H1.25C.45 8.16 0 9.97 0 12s.45 3.84 1.25 5.42l4.03-3.15z" />
                <path fill="#EA4335" d="M12 4.75c1.77 0 3.35.61 4.6 1.8l3.42-3.42C17.95 1.19 15.24 0 12 0 7.33 0 3.26 2.64 1.25 6.58l4.03 3.15c.95-2.83 3.6-4.98 6.72-4.98z" />
              </svg>
              <span className="font-bold tracking-wide">Lanjutkan dengan Akun Google</span>
            </button>
          </div>
        )}

        {/* ========================================================= */}
        {/* ⚙️ MODE 3: KREDENSIAL CADANGAN KHUSUS (OPSIONAL)           */}
        {/* ========================================================= */}
        <div className="mt-5 pt-4 border-t border-[#E0E0D8] dark:border-[#383832]">
          {loginMode !== 'legacy_password' ? (
            <button
              type="button"
              onClick={() => {
                setLoginMode('legacy_password');
                setLegacyError('');
              }}
              className="text-[11px] font-bold text-[#8E8E85] hover:text-[#5A5A40] dark:hover:text-[#C2C2B0] flex items-center justify-center gap-1.5 mx-auto cursor-pointer"
            >
              <KeyRound className="w-3.5 h-3.5" />
              <span>Masuk Kredensial Khusus (Khusus Akun Offline/Password)</span>
            </button>
          ) : (
            <div className="space-y-4 animate-in fade-in">
              <div className="flex items-center justify-between">
                <span className="text-xs font-bold text-[#5A5A40] dark:text-[#C2C2B0]">
                  Masuk dengan Kredensial Khusus
                </span>
                <button
                  type="button"
                  onClick={() => setLoginMode('email_code')}
                  className="text-xs text-[#8C4A32] font-bold hover:underline cursor-pointer"
                >
                  ← Kembali ke Email & Google
                </button>
              </div>

              {/* Sub-Role Selector */}
              <div className="grid grid-cols-4 gap-1 p-1 bg-[#F5F5F0] dark:bg-[#2A2A26] rounded-xl border border-[#E0E0D8] dark:border-[#383832]">
                <button
                  type="button"
                  onClick={() => setLegacyRole('student')}
                  className={`py-1.5 px-2 rounded-lg text-[11px] font-bold ${legacyRole === 'student' ? 'bg-white dark:bg-[#1E1E1C] text-[#5A5A40]' : 'text-[#8E8E85]'}`}
                >
                  Siswa
                </button>
                <button
                  type="button"
                  onClick={() => setLegacyRole('class_admin')}
                  className={`py-1.5 px-2 rounded-lg text-[11px] font-bold ${legacyRole === 'class_admin' ? 'bg-white dark:bg-[#1E1E1C] text-[#8C4A32]' : 'text-[#8E8E85]'}`}
                >
                  Admin Kelas
                </button>
                <button
                  type="button"
                  onClick={() => setLegacyRole('eskul_admin')}
                  className={`py-1.5 px-2 rounded-lg text-[11px] font-bold ${legacyRole === 'eskul_admin' ? 'bg-white dark:bg-[#1E1E1C] text-[#5A5A40]' : 'text-[#8E8E85]'}`}
                >
                  Admin Eskul
                </button>
                <button
                  type="button"
                  onClick={() => setLegacyRole('super_admin')}
                  className={`py-1.5 px-2 rounded-lg text-[11px] font-bold ${legacyRole === 'super_admin' ? 'bg-[#5A5A40] text-white' : 'text-[#8E8E85]'}`}
                >
                  Admin Utama
                </button>
              </div>

              {legacyError && (
                <div className="p-2.5 bg-rose-50 border border-rose-200 text-rose-800 rounded-xl text-xs">
                  {legacyError}
                </div>
              )}

              <form onSubmit={handleLegacySubmit} className="space-y-3">
                {legacyRole === 'student' && (
                  <>
                    <div>
                      <label className="block text-[11px] font-bold mb-1">Pilih Kelas *</label>
                      <select
                        value={chosenClassId}
                        onChange={e => setChosenClassId(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      >
                        {classes.map(c => <option key={c.id} value={c.id}>{c.name}</option>)}
                      </select>
                    </div>

                    <div>
                      <label className="block text-[11px] font-bold mb-1">Pilih Nama Siswa *</label>
                      <select
                        value={selectedStudentId}
                        onChange={e => setSelectedStudentId(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      >
                        <option value="">-- Pilih Siswa --</option>
                        {students.filter(s => s.classId === chosenClassId).map(s => (
                          <option key={s.id} value={s.id}>{s.name} ({s.nisn})</option>
                        ))}
                      </select>
                    </div>

                    <div>
                      <label className="block text-[11px] font-bold mb-1">Kata Sandi Siswa *</label>
                      <input
                        type="password"
                        placeholder="Default: 123456"
                        value={studentPassword}
                        onChange={e => setStudentPassword(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      />
                    </div>
                  </>
                )}

                {legacyRole === 'class_admin' && (
                  <>
                    <div>
                      <label className="block text-[11px] font-bold mb-1">Pilih Kelas *</label>
                      <select
                        value={chosenAdminClassId}
                        onChange={e => setChosenAdminClassId(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      >
                        {classes.map(c => <option key={c.id} value={c.id}>{c.name}</option>)}
                      </select>
                    </div>

                    <div>
                      <label className="block text-[11px] font-bold mb-1">Username Admin Kelas *</label>
                      <input
                        type="text"
                        placeholder="Contoh: admin_12mipa1"
                        value={classUsername}
                        onChange={e => setClassUsername(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      />
                    </div>

                    <div>
                      <label className="block text-[11px] font-bold mb-1">Kata Sandi *</label>
                      <input
                        type="password"
                        placeholder="Kata sandi admin kelas..."
                        value={classPassword}
                        onChange={e => setClassPassword(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      />
                    </div>
                  </>
                )}

                {legacyRole === 'eskul_admin' && (
                  <>
                    <div>
                      <label className="block text-[11px] font-bold mb-1">Pilih Ekstrakurikuler *</label>
                      <select
                        value={chosenAdminEskulId}
                        onChange={e => setChosenAdminEskulId(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      >
                        {eskulList.map(e => <option key={e.id} value={e.id}>{e.name}</option>)}
                      </select>
                    </div>

                    <div>
                      <label className="block text-[11px] font-bold mb-1">Username Admin Eskul *</label>
                      <input
                        type="text"
                        placeholder="Username admin eskul..."
                        value={eskulUsername}
                        onChange={e => setEskulUsername(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      />
                    </div>

                    <div>
                      <label className="block text-[11px] font-bold mb-1">Kata Sandi *</label>
                      <input
                        type="password"
                        placeholder="Kata sandi admin eskul..."
                        value={eskulPassword}
                        onChange={e => setEskulPassword(e.target.value)}
                        className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                      />
                    </div>
                  </>
                )}

                {legacyRole === 'super_admin' && (
                  <div>
                    <label className="block text-[11px] font-bold mb-1">Kata Sandi Admin Utama *</label>
                    <input
                      type="password"
                      placeholder="Masukkan kata sandi Admin Utama..."
                      value={superAdminInputPassword}
                      onChange={e => setSuperAdminInputPassword(e.target.value)}
                      className="w-full px-3 py-2 bg-[#F8F8F5] dark:bg-[#2A2A26] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs"
                    />
                  </div>
                )}

                <button
                  type="submit"
                  className="w-full py-2.5 bg-[#5A5A40] text-white font-bold text-xs rounded-xl shadow-xs cursor-pointer"
                >
                  Masuk
                </button>
              </form>
            </div>
          )}
        </div>
      </div>
    </div>
  );
};
