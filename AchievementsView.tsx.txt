import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { SchoolAchievement } from '../types';
import { 
  Trophy, 
  Award, 
  Medal, 
  Search, 
  Filter, 
  Plus, 
  Edit3, 
  Trash2, 
  Calendar, 
  Users, 
  Sparkles,
  MapPin,
  X
} from 'lucide-react';
import { MediaUploadField } from '../components/MediaUploadField';

export const AchievementsView: React.FC = () => {
  const { 
    achievements, 
    addAchievement, 
    updateAchievement, 
    deleteAchievement, 
    authSession,
    requestDeleteConfirm,
    hasAdminScope
  } = useApp();

  const [searchQuery, setSearchQuery] = useState<string>('');
  const [selectedLevel, setSelectedLevel] = useState<string>('Semua');
  const [selectedYear, setSelectedYear] = useState<string>('Semua');
  const [isModalOpen, setIsModalOpen] = useState<boolean>(false);
  const [editingItem, setEditingItem] = useState<SchoolAchievement | null>(null);

  const [form, setForm] = useState({
    title: '',
    competitionName: '',
    winnerName: '',
    rank: 'Juara 1',
    level: 'Nasional',
    year: String(new Date().getFullYear()),
    category: 'Akademik',
    photoUrl: '',
    description: ''
  });

  const canManage = hasAdminScope('achievements') || authSession.role === 'super_admin';

  // Distinct Years
  const distinctYears = Array.from(new Set(achievements.map(a => Number(a.year || a.tahunAjaran?.split('/')[0] || 2024)))).sort((a, b) => Number(b) - Number(a));

  // Filtered List
  const filteredList = achievements.filter(item => {
    const compName = item.competitionName || item.title || '';
    const winName = item.winnerName || item.namaPemenang || '';
    const rk = item.rank || item.juara || '';
    const lvl = item.level || item.tingkat || '';
    const yr = String(item.year || item.tahunAjaran || '');

    const matchSearch = item.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
      compName.toLowerCase().includes(searchQuery.toLowerCase()) ||
      winName.toLowerCase().includes(searchQuery.toLowerCase()) ||
      rk.toLowerCase().includes(searchQuery.toLowerCase());

    const matchLevel = selectedLevel === 'Semua' || lvl === selectedLevel;
    const matchYear = selectedYear === 'Semua' || yr.includes(selectedYear);

    return matchSearch && matchLevel && matchYear;
  });

  const handleOpenAdd = () => {
    setEditingItem(null);
    setForm({
      title: '',
      competitionName: '',
      winnerName: '',
      rank: 'Juara 1',
      level: 'Nasional',
      year: String(new Date().getFullYear()),
      category: 'Akademik',
      photoUrl: '',
      description: ''
    });
    setIsModalOpen(true);
  };

  const handleOpenEdit = (item: SchoolAchievement) => {
    setEditingItem(item);
    setForm({
      title: item.title,
      competitionName: item.competitionName || item.title,
      winnerName: item.winnerName || item.namaPemenang,
      rank: item.rank || item.juara,
      level: item.level || item.tingkat || 'Nasional',
      year: String(item.year || item.tahunAjaran || '2024/2025'),
      category: item.category || item.kategori || 'Akademik',
      photoUrl: item.photoUrl || item.fotoUrl || '',
      description: item.description || item.deskripsi || ''
    });
    setIsModalOpen(true);
  };

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!form.title.trim() || !form.winnerName.trim()) return;

    if (editingItem) {
      updateAchievement(editingItem.id, {
        title: form.title.trim(),
        competitionName: form.competitionName.trim(),
        winnerName: form.winnerName.trim(),
        namaPemenang: form.winnerName.trim(),
        rank: form.rank.trim(),
        juara: form.rank.trim(),
        level: form.level,
        tingkat: form.level as any,
        year: form.year,
        tahunAjaran: form.year,
        category: form.category,
        kategori: form.category as any,
        photoUrl: form.photoUrl.trim() || undefined,
        fotoUrl: form.photoUrl.trim() || undefined,
        description: form.description.trim() || undefined,
        deskripsi: form.description.trim() || undefined
      });
    } else {
      addAchievement({
        title: form.title.trim(),
        competitionName: form.competitionName.trim(),
        winnerName: form.winnerName.trim(),
        namaPemenang: form.winnerName.trim(),
        rank: form.rank.trim(),
        juara: form.rank.trim(),
        level: form.level,
        tingkat: form.level as any,
        year: form.year,
        tahunAjaran: form.year,
        category: form.category,
        kategori: form.category as any,
        photoUrl: form.photoUrl.trim() || undefined,
        fotoUrl: form.photoUrl.trim() || undefined,
        description: form.description.trim() || undefined,
        deskripsi: form.description.trim() || undefined,
        tanggal: new Date().toISOString().split('T')[0]
      });
    }

    setIsModalOpen(false);
  };

  const handleDelete = (item: SchoolAchievement) => {
    requestDeleteConfirm({
      itemName: `Prestasi "${item.title}"`,
      message: `Yakin ingin menghapus catatan prestasi "${item.title}"?`,
      onConfirm: () => deleteAchievement(item.id)
    });
  };

  return (
    <div className="space-y-8 pb-20 animate-fade-in">
      {/* Header Banner */}
      <div className="relative overflow-hidden bg-gradient-to-br from-[#8C4A32] to-[#5C2B1C] text-white p-6 sm:p-10 rounded-[32px] shadow-lg">
        <div className="relative z-10 max-w-3xl space-y-3">
          <div className="inline-flex items-center gap-2 px-3.5 py-1 bg-white/20 backdrop-blur-md rounded-full text-xs font-bold uppercase tracking-widest text-amber-200">
            <Trophy className="w-4 h-4" />
            <span>Menu 2: Galeri Kebanggaan Sekolah</span>
          </div>
          <h1 className="font-serif text-2xl sm:text-4xl font-bold tracking-tight">
            Prestasi & Kejuaraan
          </h1>
          <p className="text-sm sm:text-base text-white/85 leading-relaxed">
            Daftar lengkap penghargaan, piala, dan medali kejuaraan yang dipersembahkan oleh putra-putri terbaik sekolah dari tingkat Kabupaten, Provinsi, hingga Nasional dan Internasional.
          </p>

          {canManage && (
            <div className="pt-2">
              <button
                onClick={handleOpenAdd}
                className="inline-flex items-center gap-2 px-4 py-2.5 bg-amber-400 hover:bg-amber-300 text-[#2D2D2A] font-bold rounded-2xl text-xs sm:text-sm shadow-md transition-transform active:scale-95 cursor-pointer"
              >
                <Plus className="w-4 h-4" />
                <span>Tambah Catatan Prestasi</span>
              </button>
            </div>
          )}
        </div>
      </div>

      {/* Filter & Search Bar */}
      <div className="flex flex-col sm:flex-row items-center gap-4 bg-white dark:bg-[#1E1E1C] p-4 rounded-2xl border border-[#E0E0D8] dark:border-[#383832] shadow-xs">
        <div className="relative flex-1 w-full">
          <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 w-4 h-4 text-[#8E8E85]" />
          <input
            type="text"
            placeholder="Cari nama siswa peraih juara, lomba, atau kejuaraan..."
            value={searchQuery}
            onChange={e => setSearchQuery(e.target.value)}
            className="w-full pl-10 pr-4 py-2 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2] focus:outline-none focus:ring-2 focus:ring-[#8C4A32]"
          />
        </div>

        <div className="flex items-center gap-2 w-full sm:w-auto overflow-x-auto pb-1 sm:pb-0">
          <Filter className="w-4 h-4 text-[#8E8E85] shrink-0" />
          <select
            value={selectedLevel}
            onChange={e => setSelectedLevel(e.target.value)}
            className="px-3 py-2 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-bold text-[#8C4A32] dark:text-[#E07A5F] focus:outline-none"
          >
            <option value="Semua">Semua Tingkat</option>
            <option value="Kabupaten">Tingkat Kabupaten</option>
            <option value="Provinsi">Tingkat Provinsi</option>
            <option value="Nasional">Tingkat Nasional</option>
            <option value="Internasional">Tingkat Internasional</option>
          </select>

          <select
            value={selectedYear}
            onChange={e => setSelectedYear(e.target.value)}
            className="px-3 py-2 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-bold text-[#8C4A32] dark:text-[#E07A5F] focus:outline-none"
          >
            <option value="Semua">Semua Tahun</option>
            {distinctYears.map(year => (
              <option key={year} value={year.toString()}>Tahun {year}</option>
            ))}
          </select>
        </div>
      </div>

      {/* Achievement Cards Grid */}
      {filteredList.length === 0 ? (
        <div className="text-center py-16 bg-white dark:bg-[#1E1E1C] rounded-3xl border border-[#E0E0D8] dark:border-[#383832] p-8">
          <Trophy className="w-12 h-12 text-[#8E8E85] mx-auto mb-3 opacity-40" />
          <p className="text-base font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">Belum ada catatan kejuaraan</p>
          <p className="text-xs text-[#8E8E85] mt-1">Coba sesuaikan kata kunci pencarian atau filter tingkat lomba.</p>
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {filteredList.map(item => {
            const isGold = item.rank.toLowerCase().includes('1') || item.rank.toLowerCase().includes('emas');
            const isSilver = item.rank.toLowerCase().includes('2') || item.rank.toLowerCase().includes('perak');

            return (
              <div
                key={item.id}
                className="bg-white dark:bg-[#1E1E1C] rounded-[24px] border border-[#E0E0D8] dark:border-[#383832] shadow-sm hover:shadow-md transition-all overflow-hidden flex flex-col justify-between group"
              >
                <div>
                  {/* Photo Container */}
                  <div className="relative aspect-[16/10] w-full overflow-hidden bg-neutral-100 dark:bg-neutral-800">
                    <img
                      src={item.photoUrl || 'https://images.unsplash.com/photo-1578269174936-2709b6aeb913?w=600&auto=format&fit=crop&q=80'}
                      alt={item.title}
                      className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
                    />
                    <div className="absolute inset-0 bg-gradient-to-t from-black/70 via-transparent to-transparent pointer-events-none" />

                    {/* Rank Badge */}
                    <div className="absolute top-3 left-3 flex items-center gap-1.5 px-3 py-1 bg-amber-400 text-[#2D2D2A] font-extrabold text-xs rounded-full shadow-md">
                      <Medal className="w-3.5 h-3.5" />
                      <span>{item.rank}</span>
                    </div>

                    {/* Level Badge */}
                    <div className="absolute top-3 right-3 px-2.5 py-1 bg-black/60 backdrop-blur-md text-white font-bold text-[10px] uppercase tracking-wider rounded-full">
                      Tingkat {item.level}
                    </div>

                    {/* Category & Year at Bottom of Image */}
                    <div className="absolute bottom-3 left-3 right-3 text-white">
                      <span className="text-[10px] font-bold uppercase tracking-wider text-amber-300">
                        {item.category} • Tahun {item.year}
                      </span>
                      <h4 className="font-serif text-sm font-bold text-white line-clamp-1">
                        {item.competitionName}
                      </h4>
                    </div>
                  </div>

                  {/* Body Content */}
                  <div className="p-5 space-y-3">
                    <div>
                      <h3 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA] line-clamp-2">
                        {item.title}
                      </h3>
                      <div className="flex items-center gap-1.5 text-xs text-[#8C4A32] dark:text-[#E07A5F] font-bold mt-1.5">
                        <Users className="w-3.5 h-3.5 shrink-0" />
                        <span>Peraih Juara: {item.winnerName}</span>
                      </div>
                    </div>

                    {item.description && (
                      <p className="text-xs text-[#5A5A40] dark:text-[#A4A485] leading-relaxed line-clamp-3">
                        {item.description}
                      </p>
                    )}
                  </div>
                </div>

                {/* Admin Control Footer */}
                {canManage && (
                  <div className="p-3 bg-[#F5F5F0] dark:bg-[#282825] border-t border-[#E0E0D8] dark:border-[#383832] flex items-center justify-end gap-2">
                    <button
                      onClick={() => handleOpenEdit(item)}
                      className="p-1.5 text-xs text-[#5A5A40] dark:text-[#D1D1BE] hover:bg-[#E0E0D8] dark:hover:bg-[#383832] rounded-lg transition-colors cursor-pointer flex items-center gap-1 font-semibold"
                    >
                      <Edit3 className="w-3.5 h-3.5" />
                      <span>Edit</span>
                    </button>
                    <button
                      onClick={() => handleDelete(item)}
                      className="p-1.5 text-xs text-rose-600 hover:bg-rose-50 dark:hover:bg-rose-950/30 rounded-lg transition-colors cursor-pointer flex items-center gap-1 font-semibold"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                      <span>Hapus</span>
                    </button>
                  </div>
                )}
              </div>
            );
          })}
        </div>
      )}

      {/* Add / Edit Modal */}
      {isModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-xs">
          <div className="bg-white dark:bg-[#1E1E1C] w-full max-w-lg rounded-[28px] border border-[#E0E0D8] dark:border-[#383832] shadow-2xl overflow-hidden animate-scale-in">
            <div className="p-5 border-b border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between">
              <div className="flex items-center gap-2">
                <div className="w-9 h-9 rounded-xl bg-[#8C4A32] text-white flex items-center justify-center font-bold">
                  <Trophy className="w-5 h-5" />
                </div>
                <h3 className="font-serif text-lg font-bold text-[#2D2D2A] dark:text-[#EAEAE2]">
                  {editingItem ? 'Edit Data Prestasi' : 'Tambah Catatan Prestasi'}
                </h3>
              </div>
              <button
                onClick={() => setIsModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:text-[#2D2D2A] dark:hover:text-white rounded-lg cursor-pointer"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleSubmit} className="p-5 space-y-4 max-h-[75vh] overflow-y-auto">
              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Judul Pencapaian Prestasi *
                </label>
                <input
                  type="text"
                  required
                  value={form.title}
                  onChange={e => setForm({ ...form, title: e.target.value })}
                  placeholder="Contoh: Juara 1 Olimpiade Sains Nasional (OSN) Matematika"
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Nama Kejuaraan / Lomba *
                  </label>
                  <input
                    type="text"
                    required
                    value={form.competitionName}
                    onChange={e => setForm({ ...form, competitionName: e.target.value })}
                    placeholder="Contoh: OSN Tingkat Nasional"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Peraih Juara (Siswa/Tim) *
                  </label>
                  <input
                    type="text"
                    required
                    value={form.winnerName}
                    onChange={e => setForm({ ...form, winnerName: e.target.value })}
                    placeholder="Contoh: Ahmad Fauzan (XII MIPA 1)"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <div className="grid grid-cols-3 gap-3">
                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Juara Diraih *
                  </label>
                  <input
                    type="text"
                    required
                    value={form.rank}
                    onChange={e => setForm({ ...form, rank: e.target.value })}
                    placeholder="Juara 1 / Medali Emas"
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Tingkat *
                  </label>
                  <select
                    value={form.level}
                    onChange={e => setForm({ ...form, level: e.target.value as any })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs font-semibold text-[#2D2D2A] dark:text-[#EAEAE2]"
                  >
                    <option value="Kabupaten">Kabupaten</option>
                    <option value="Provinsi">Provinsi</option>
                    <option value="Nasional">Nasional</option>
                    <option value="Internasional">Internasional</option>
                  </select>
                </div>

                <div>
                  <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                    Tahun *
                  </label>
                  <input
                    type="number"
                    required
                    value={form.year}
                    onChange={e => setForm({ ...form, year: Number(e.target.value) })}
                    className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                  />
                </div>
              </div>

              <MediaUploadField
                label="Foto Penyerahan Piala / Piagam (URL / Upload)"
                value={form.photoUrl}
                onChange={url => setForm({ ...form, photoUrl: url })}
                placeholder="https://..."
              />

              <div>
                <label className="block text-xs font-bold uppercase text-[#5A5A40] dark:text-[#A4A485] mb-1">
                  Deskripsi Singkat / Cerita Kejuaraan
                </label>
                <textarea
                  rows={3}
                  value={form.description}
                  onChange={e => setForm({ ...form, description: e.target.value })}
                  placeholder="Keterangan singkat mengenai proses perlombaan..."
                  className="w-full px-3.5 py-2.5 bg-[#F5F5F0] dark:bg-[#282825] border border-[#E0E0D8] dark:border-[#42423B] rounded-xl text-xs sm:text-sm text-[#2D2D2A] dark:text-[#EAEAE2]"
                />
              </div>

              <div className="pt-2 flex items-center justify-end gap-3">
                <button
                  type="button"
                  onClick={() => setIsModalOpen(false)}
                  className="px-4 py-2.5 text-xs font-bold text-[#8E8E85] hover:bg-[#F5F5F0] dark:hover:bg-[#282825] rounded-xl cursor-pointer"
                >
                  Batal
                </button>
                <button
                  type="submit"
                  className="px-5 py-2.5 bg-[#8C4A32] hover:bg-[#723B28] text-white text-xs font-bold rounded-xl shadow-md cursor-pointer"
                >
                  {editingItem ? 'Simpan Perubahan' : 'Tambah Prestasi'}
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
