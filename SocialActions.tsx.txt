import React, { useState } from 'react';
import { Heart, MessageCircle, Share2, Send, Check, User, Maximize2, Minimize2, X } from 'lucide-react';
import { SocialComment } from '../types';
import { useApp } from '../context/AppContext';

interface SocialActionsProps {
  itemId: string;
  itemTitle?: string;
  itemCategory?: string;
  likes?: number;
  likedBy?: string[];
  comments?: SocialComment[];
  shares?: number;
  onToggleLike: () => void;
  onAddComment: (content: string, customAuthor?: string) => void;
  onShare?: () => void;
  compact?: boolean;
}

export const SocialActions: React.FC<SocialActionsProps> = ({
  itemId,
  itemTitle = 'Dokumentasi Sekolah',
  itemCategory,
  likes = 0,
  likedBy = [],
  comments = [],
  shares = 0,
  onToggleLike,
  onAddComment,
  onShare,
  compact = false
}) => {
  const { authSession, deviceId, showToast } = useApp();
  const [isCommentsOpen, setIsCommentsOpen] = useState<boolean>(false);
  const [isExpandedModalOpen, setIsExpandedModalOpen] = useState<boolean>(false);
  const [commentText, setCommentText] = useState<string>('');
  const [visitorName, setVisitorName] = useState<string>('');
  const [copied, setCopied] = useState<boolean>(false);

  // Each user can like only once per photo/video (toggleable)
  const currentUserId = authSession.studentId || authSession.username || deviceId;
  const isLiked = (likedBy || []).includes(currentUserId);

  const handleSubmitComment = (e: React.FormEvent) => {
    e.preventDefault();
    if (!commentText.trim()) return;

    const author = authSession.role !== 'visitor'
      ? authSession.name || 'Pengguna Terdaftar'
      : (visitorName.trim() || 'Pengunjung Website');

    onAddComment(commentText.trim(), author);
    setCommentText('');
  };

  const handleShare = async () => {
    // Notify parent to increment share count in persistent database
    if (onShare) {
      onShare();
    }

    const shareUrl = window.location.href;
    const shareData = {
      title: itemTitle,
      text: `Lihat dokumentasi "${itemTitle}" di Galeri Kenangan Sekolah!`,
      url: shareUrl
    };

    if (navigator.share) {
      try {
        await navigator.share(shareData);
        showToast('✅ Berhasil dibagikan!', 'success');
        return;
      } catch (err) {
        // Fallback to clipboard if share dialog dismissed or error
      }
    }

    // Fallback: Copy link
    try {
      await navigator.clipboard.writeText(shareUrl);
      setCopied(true);
      showToast('🔗 Tautan berhasil disalin ke papan klip!', 'success');
      setTimeout(() => setCopied(false), 2500);
    } catch {
      showToast('Gagal menyalin tautan.', 'error');
    }
  };

  const commentsCount = comments?.length || 0;

  return (
    <div className="w-full border-t border-[#E0E0D8] pt-3 mt-3 space-y-3">
      {/* Action Buttons Bar */}
      <div className="flex items-center justify-between gap-2 text-xs">
        {/* Like Button (1 like per user per photo/video) */}
        <button
          type="button"
          onClick={onToggleLike}
          className={`flex items-center gap-1.5 px-3 py-1.5 rounded-xl font-semibold transition-all cursor-pointer select-none active:scale-90 ${
            isLiked
              ? 'bg-rose-50 text-rose-600 border border-rose-200'
              : 'bg-[#F5F5F0] text-[#5A5A40] hover:bg-[#EBEBE5] border border-[#E0E0D8]'
          }`}
          title={isLiked ? 'Batal Suka' : 'Sukai Foto/Video Ini'}
        >
          <Heart className={`w-4 h-4 ${isLiked ? 'fill-rose-500 text-rose-500 scale-110' : 'text-[#5A5A40]'}`} />
          <span>{likes > 0 ? `${likes} Suka` : 'Suka'}</span>
        </button>

        {/* Comment Toggle Button */}
        <button
          type="button"
          onClick={() => setIsCommentsOpen(!isCommentsOpen)}
          className={`flex items-center gap-1.5 px-3 py-1.5 rounded-xl font-semibold transition-all cursor-pointer select-none ${
            isCommentsOpen
              ? 'bg-[#5A5A40] text-white shadow-sm'
              : 'bg-[#F5F5F0] text-[#5A5A40] hover:bg-[#EBEBE5] border border-[#E0E0D8]'
          }`}
          title="Buka / Tutup Kolom Komentar"
        >
          <MessageCircle className="w-4 h-4" />
          <span>{commentsCount > 0 ? `${commentsCount} Komentar` : 'Komentar'}</span>
        </button>

        {/* Share Button */}
        <button
          type="button"
          onClick={handleShare}
          className="flex items-center gap-1.5 px-3 py-1.5 rounded-xl font-semibold bg-[#F5F5F0] text-[#5A5A40] hover:bg-[#EBEBE5] border border-[#E0E0D8] transition-all cursor-pointer select-none active:scale-95 ml-auto"
          title="Bagikan Foto/Video Ini"
        >
          {copied ? <Check className="w-4 h-4 text-emerald-600" /> : <Share2 className="w-4 h-4" />}
          <span>{copied ? 'Tersalin' : (shares > 0 ? `${shares} Bagikan` : 'Bagikan')}</span>
        </button>
      </div>

      {/* Expandable Side-to-Side Comment Container ("Kotak dari samping kanan sampai samping kiri") */}
      {isCommentsOpen && (
        <div className="w-full bg-[#F8F8F5] border border-[#E0E0D8] rounded-2xl p-4 space-y-3.5 shadow-inner animate-fade-in transition-all">
          {/* Header of Comment Section */}
          <div className="flex items-center justify-between border-b border-[#E0E0D8] pb-2.5">
            <div className="flex items-center gap-2">
              <MessageCircle className="w-4 h-4 text-[#5A5A40]" />
              <span className="text-xs font-bold text-[#2D2D2A]">
                Ruang Komentar ({commentsCount})
              </span>
            </div>
            <div className="flex items-center gap-1.5">
              <button
                type="button"
                onClick={() => setIsExpandedModalOpen(true)}
                className="p-1 text-[#8E8E85] hover:text-[#2D2D2A] hover:bg-[#EBEBE5] rounded-lg transition-colors cursor-pointer"
                title="Perbesar Layar Penuh"
              >
                <Maximize2 className="w-3.5 h-3.5" />
              </button>
              <button
                type="button"
                onClick={() => setIsCommentsOpen(false)}
                className="p-1 text-[#8E8E85] hover:text-rose-600 hover:bg-rose-50 rounded-lg transition-colors cursor-pointer"
                title="Tutup Komentar"
              >
                <X className="w-3.5 h-3.5" />
              </button>
            </div>
          </div>

          {/* Comment List Container */}
          <div className="w-full space-y-2.5 max-h-56 overflow-y-auto pr-1">
            {commentsCount === 0 ? (
              <div className="text-center py-6 px-4 bg-white/70 rounded-xl border border-dashed border-[#E0E0D8]">
                <p className="text-xs font-medium text-[#5A5A40]">Belum ada komentar</p>
                <p className="text-[11px] text-[#8E8E85] mt-0.5">Jadilah yang pertama memberikan apresiasi dan tanggapan!</p>
              </div>
            ) : (
              (comments || []).map(c => (
                <div key={c.id} className="w-full bg-white p-3 rounded-xl border border-[#E0E0D8] shadow-xs space-y-1.5">
                  <div className="flex items-center justify-between text-[11px]">
                    <div className="flex items-center gap-1.5 font-bold text-[#2D2D2A]">
                      <span className="truncate max-w-[160px] sm:max-w-none">{c.authorName}</span>
                      {c.authorRole && c.authorRole !== 'visitor' && (
                        <span className="px-1.5 py-0.5 text-[9px] bg-[#F5F5F0] text-[#5A5A40] rounded font-semibold border border-[#E0E0D8]">
                          {c.authorRole === 'super_admin' ? 'Admin Utama' : c.authorRole === 'class_admin' ? 'Admin Kelas' : 'Siswa'}
                        </span>
                      )}
                    </div>
                    <span className="text-[10px] text-[#8E8E85] whitespace-nowrap">
                      {new Date(c.createdAt).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' })}
                    </span>
                  </div>
                  <p className="text-xs text-[#2D2D2A] leading-relaxed break-words whitespace-pre-wrap">{c.content}</p>
                </div>
              ))
            )}
          </div>

          {/* Comment Input Form */}
          <form onSubmit={handleSubmitComment} className="w-full space-y-2 pt-2 border-t border-[#E0E0D8]">
            {authSession.role === 'visitor' && (
              <input
                type="text"
                placeholder="Nama Anda (opsional, cth: Budi Pratama / Alumni)..."
                value={visitorName}
                onChange={e => setVisitorName(e.target.value)}
                className="w-full px-3 py-2 text-xs bg-white border border-[#E0E0D8] rounded-xl text-[#2D2D2A] focus:outline-none focus:border-[#5A5A40]"
              />
            )}
            <div className="flex items-center gap-2">
              <input
                type="text"
                required
                placeholder="Tulis komentar atau tanggapan Anda..."
                value={commentText}
                onChange={e => setCommentText(e.target.value)}
                className="flex-1 px-3.5 py-2.5 text-xs bg-white border border-[#E0E0D8] rounded-xl text-[#2D2D2A] placeholder:text-[#8E8E85] focus:outline-none focus:border-[#5A5A40] focus:ring-1 focus:ring-[#5A5A40]"
              />
              <button
                type="submit"
                className="flex items-center gap-1 px-4 py-2.5 rounded-xl bg-[#5A5A40] hover:bg-[#484833] text-white font-semibold text-xs cursor-pointer active:scale-95 shadow-sm transition-all whitespace-nowrap"
                title="Kirim Komentar"
              >
                <span>Kirim</span>
                <Send className="w-3.5 h-3.5" />
              </button>
            </div>
          </form>
        </div>
      )}

      {/* Full-Width Modal Overlay for Extended Reading ("Ukuran kotak layar penuh dari samping kanan sampai samping kiri") */}
      {isExpandedModalOpen && (
        <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-sm flex items-center justify-center p-3 sm:p-6 animate-fade-in">
          <div className="w-full max-w-2xl bg-white border border-[#E0E0D8] rounded-3xl p-5 sm:p-6 shadow-2xl flex flex-col max-h-[85vh] space-y-4">
            {/* Modal Header */}
            <div className="flex items-center justify-between border-b border-[#E0E0D8] pb-3">
              <div>
                <h3 className="text-base font-bold text-[#2D2D2A] flex items-center gap-2">
                  <MessageCircle className="w-5 h-5 text-[#5A5A40]" />
                  Semua Komentar ({commentsCount})
                </h3>
                <p className="text-xs text-[#8E8E85] truncate max-w-sm sm:max-w-md mt-0.5">
                  {itemTitle}
                </p>
              </div>
              <button
                type="button"
                onClick={() => setIsExpandedModalOpen(false)}
                className="p-2 text-[#8E8E85] hover:text-[#2D2D2A] hover:bg-[#F5F5F0] rounded-xl transition-colors cursor-pointer"
                title="Tutup Modal"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            {/* Modal Comment List */}
            <div className="flex-1 overflow-y-auto pr-2 space-y-3">
              {commentsCount === 0 ? (
                <div className="text-center py-12 px-4 bg-[#F8F8F5] rounded-2xl border border-dashed border-[#E0E0D8]">
                  <MessageCircle className="w-10 h-10 text-[#8E8E85] mx-auto mb-2 opacity-50" />
                  <p className="text-sm font-semibold text-[#5A5A40]">Belum ada komentar</p>
                  <p className="text-xs text-[#8E8E85] mt-1">Jadilah yang pertama menuliskan komentar dan kesan!</p>
                </div>
              ) : (
                (comments || []).map(c => (
                  <div key={c.id} className="w-full bg-[#F8F8F5] p-3.5 rounded-2xl border border-[#E0E0D8] space-y-1.5">
                    <div className="flex items-center justify-between text-xs">
                      <div className="flex items-center gap-2 font-bold text-[#2D2D2A]">
                        <span>{c.authorName}</span>
                        {c.authorRole && c.authorRole !== 'visitor' && (
                          <span className="px-2 py-0.5 text-[10px] bg-white text-[#5A5A40] rounded font-semibold border border-[#E0E0D8]">
                            {c.authorRole === 'super_admin' ? 'Admin Utama' : c.authorRole === 'class_admin' ? 'Admin Kelas' : 'Siswa'}
                          </span>
                        )}
                      </div>
                      <span className="text-[11px] text-[#8E8E85]">
                        {new Date(c.createdAt).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })}
                      </span>
                    </div>
                    <p className="text-xs sm:text-sm text-[#2D2D2A] leading-relaxed break-words whitespace-pre-wrap">{c.content}</p>
                  </div>
                ))
              )}
            </div>

            {/* Modal Input Form */}
            <form onSubmit={handleSubmitComment} className="border-t border-[#E0E0D8] pt-3 space-y-2">
              {authSession.role === 'visitor' && (
                <input
                  type="text"
                  placeholder="Nama Anda (opsional)..."
                  value={visitorName}
                  onChange={e => setVisitorName(e.target.value)}
                  className="w-full px-3.5 py-2 text-xs bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] focus:outline-none focus:border-[#5A5A40]"
                />
              )}
              <div className="flex items-center gap-2">
                <input
                  type="text"
                  required
                  placeholder="Tulis tanggapan atau komentar Anda..."
                  value={commentText}
                  onChange={e => setCommentText(e.target.value)}
                  className="flex-1 px-4 py-2.5 text-xs sm:text-sm bg-[#F8F8F5] border border-[#E0E0D8] rounded-xl text-[#2D2D2A] placeholder:text-[#8E8E85] focus:outline-none focus:border-[#5A5A40]"
                />
                <button
                  type="submit"
                  className="flex items-center gap-1.5 px-5 py-2.5 rounded-xl bg-[#5A5A40] hover:bg-[#484833] text-white font-semibold text-xs sm:text-sm cursor-pointer active:scale-95 shadow-sm transition-all"
                >
                  <span>Kirim</span>
                  <Send className="w-4 h-4" />
                </button>
              </div>
            </form>
          </div>
        </div>
      )}
    </div>
  );
};
