import React from 'react';
import { useApp } from '../context/AppContext';
import { AlertTriangle, Trash2, X } from 'lucide-react';

export const ConfirmModal: React.FC = () => {
  const { deleteConfirmState, closeDeleteConfirm } = useApp();

  if (!deleteConfirmState.isOpen) return null;

  return (
    <div
      id="confirm-delete-modal"
      className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-950/80 backdrop-blur-sm animate-in fade-in duration-150"
    >
      <div className="relative w-full max-w-md bg-slate-900 border border-slate-800 rounded-2xl shadow-2xl overflow-hidden p-6 sm:p-7 text-center">
        {/* Warning Icon Badge */}
        <div className="mx-auto w-14 h-14 rounded-2xl bg-rose-500/15 border border-rose-500/30 flex items-center justify-center text-rose-500 mb-4 shadow-inner">
          <AlertTriangle className="w-7 h-7" />
        </div>

        {/* Modal Title */}
        <h3 className="text-lg sm:text-xl font-bold text-white mb-2">
          {deleteConfirmState.title || 'Konfirmasi Penghapusan'}
        </h3>

        {/* Target Item Name Highlight if any */}
        {deleteConfirmState.itemName && (
          <div className="inline-block max-w-full px-3 py-1 bg-slate-800 border border-slate-700 text-amber-300 text-xs font-semibold rounded-lg mb-3 truncate">
            {deleteConfirmState.itemName}
          </div>
        )}

        {/* Required Precise Message */}
        <p className="text-sm sm:text-base text-slate-300 mb-6 leading-relaxed">
          {deleteConfirmState.message || 'Yakin ingin menghapus? Tindakan ini tidak bisa dibatalkan.'}
        </p>

        {/* Actions Grid */}
        <div className="grid grid-cols-2 gap-3">
          <button
            id="btn-confirm-delete-cancel"
            type="button"
            onClick={closeDeleteConfirm}
            className="w-full py-2.5 px-4 rounded-xl text-sm font-semibold bg-slate-800 hover:bg-slate-700 text-slate-200 border border-slate-700 transition-colors cursor-pointer"
          >
            Batal
          </button>
          <button
            id="btn-confirm-delete-execute"
            type="button"
            onClick={deleteConfirmState.onConfirm}
            className="w-full py-2.5 px-4 rounded-xl text-sm font-semibold bg-rose-600 hover:bg-rose-500 text-white shadow-lg shadow-rose-600/30 flex items-center justify-center gap-2 transition-all cursor-pointer active:scale-95"
          >
            <Trash2 className="w-4 h-4" />
            <span>Hapus</span>
          </button>
        </div>

        {/* Close cross button */}
        <button
          onClick={closeDeleteConfirm}
          className="absolute top-4 right-4 text-slate-400 hover:text-white p-1 rounded-lg hover:bg-slate-800 transition-colors cursor-pointer"
          aria-label="Tutup modal"
        >
          <X className="w-5 h-5" />
        </button>
      </div>
    </div>
  );
};
