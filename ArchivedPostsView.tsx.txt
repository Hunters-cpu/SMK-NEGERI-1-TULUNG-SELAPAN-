import React, { useState } from 'react';
import { useApp } from '../context/AppContext';
import { ScrollEmboss } from '../components/ScrollEmboss';
import { 
  Archive, 
  Unlock, 
  Lock, 
  Eye, 
  ArrowLeft, 
  Calendar, 
  Clock, 
  Trash2, 
  Sparkles, 
  Layers,
  Image as ImageIcon,
  MessageSquare
} from 'lucide-react';

export const ArchivedPostsView: React.FC = () => {
  const {
    sliderItems,
    toggleSliderItemPrivacy,
    deleteSliderItem,
    announcements,
    toggleAnnouncementPrivacy,
    deleteAnnouncement,
    activities,
    toggleActivityPrivacy,
    deleteActivity,
    classes,
    toggleClassPhotoPrivacy,
    deleteClassPhoto,
    eskulList,
    toggleEskulPhotoPrivacy,
    deleteEskulPhoto,
    authSession,
    setCurrentView,
    requestDeleteConfirm,
    showToast
  } = useApp();

  const [activeTab, setActiveTab] = useState<'all' | 'slider' | 'activities' | 'announcements' | 'classes' | 'eskul'>('all');

  // Collect all private items
  const privateSliders = sliderItems.filter(s => s.isPrivate).map(s => ({
    type: 'slider' as const,
    id: s.id,
    title: s.title,
    subtitle: s.description || 'Sorotan Beranda',
    mediaUrl: s.url,
    mediaType: s.type,
    archivedAt: s.archivedAt,
    origin: 'Sorotan Beranda',
    onUnprivate: () => toggleSliderItemPrivacy(s.id),
    onDelete: () => requestDeleteConfirm({
      itemName: `Sorotan "${s.title}"`,
      onConfirm: () => deleteSliderItem(s.id)
    })
  }));

  const privateActivities = activities.filter(a => a.isPrivate).map(a => ({
    type: 'activity' as const,
    id: a.id,
    title: a.title,
    subtitle: a.category,
    mediaUrl: a.mediaUrl,
    mediaType: a.mediaType || 'image',
    archivedAt: a.archivedAt,
    origin: 'Kegiatan Sekolah',
    onUnprivate: () => toggleActivityPrivacy(a.id),
    onDelete: () => requestDeleteConfirm({
      itemName: `Kegiatan "${a.title}"`,
      onConfirm: () => deleteActivity(a.id)
    })
  }));

  const privateAnnouncements = announcements.filter(a => a.isPrivate).map(a => ({
    type: 'announcement' as const,
    id: a.id,
    title: a.title,
    subtitle: a.category,
    content: a.content,
    archivedAt: a.archivedAt,
    origin: 'Pengumuman Resmi',
    onUnprivate: () => toggleAnnouncementPrivacy(a.id),
    onDelete: () => requestDeleteConfirm({
      itemName: `Pengumuman "${a.title}"`,
      onConfirm: () => deleteAnnouncement(a.id)
    })
  }));

  // Class Photos Private
  const privateClassPhotos: any[] = [];
  classes.forEach(c => {
    (c.gallery || []).forEach(p => {
      if (p.isPrivate) {
        privateClassPhotos.push({
          type: 'class_photo' as const,
          id: p.id,
          classId: c.id,
          title: p.title || (p as any).caption || `Kenangan Kelas ${c.name}`,
          subtitle: `Kelas ${c.name} • TA ${c.academicYear}`,
          mediaUrl: p.imageUrl || (p as any).photoUrl,
          mediaType: p.mediaType || 'image',
          archivedAt: p.archivedAt,
          origin: `Galeri Kelas ${c.name}`,
          onUnprivate: () => toggleClassPhotoPrivacy(c.id, p.id),
          onDelete: () => requestDeleteConfirm({
            itemName: `Foto Kelas "${p.title || c.name}"`,
            onConfirm: () => deleteClassPhoto(c.id, p.id)
          })
        });
      }
    });
  });

  // Eskul Photos Private
  const privateEskulPhotos: any[] = [];
  eskulList.forEach(e => {
    (e.gallery || []).forEach(p => {
      if (p.isPrivate) {
        privateEskulPhotos.push({
          type: 'eskul_photo' as const,
          id: p.id,
          eskulId: e.id,
          title: p.title || p.caption || `Dokumentasi ${e.name}`,
          subtitle: `Eskul ${e.name}`,
          mediaUrl: p.imageUrl || p.photoUrl,
          mediaType: p.mediaType || 'image',
          archivedAt: p.archivedAt,
          origin: `Dokumentasi Eskul ${e.name}`,
          onUnprivate: () => toggleEskulPhotoPrivacy(e.id, p.id),
          onDelete: () => requestDeleteConfirm({
            itemName: `Foto Eskul "${p.title || p.caption || e.name}"`,
            onConfirm: () => deleteEskulPhoto(e.id, p.id)
          })
        });
      }
    });
  });

  const allArchived = [
    ...privateSliders,
    ...privateActivities,
    ...privateAnnouncements,
    ...privateClassPhotos,
    ...privateEskulPhotos
  ];

  const filteredItems = allArchived.filter(item => {
    if (activeTab === 'all') return true;
    if (activeTab === 'slider') return item.type === 'slider';
    if (activeTab === 'activities') return item.type === 'activity';
    if (activeTab === 'announcements') return item.type === 'announcement';
    if (activeTab === 'classes') return item.type === 'class_photo';
    if (activeTab === 'eskul') return item.type === 'eskul_photo';
    return true;
  });

  return (
    <div className="space-y-8 pb-24 text-[#2D2D2A] dark:text-[#EAEAE2]">
      {/* Header */}
      <div className="bg-white dark:bg-[#242422] border border-[#E0E0D8] dark:border-[#383832] p-6 sm:p-8 rounded-[32px] shadow-sm space-y-3">
        <button
          onClick={() => setCurrentView('home')}
          className="inline-flex items-center gap-1.5 text-xs font-semibold text-[#5A5A40] dark:text-[#C5C5B0] hover:underline cursor-pointer"
        >
          <ArrowLeft className="w-3.5 h-3.5" />
          <span>Kembali ke Beranda</span>
        </button>

        <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
          <div>
            <div className="inline-flex items-center gap-2 px-3 py-1 rounded-full bg-amber-50 dark:bg-amber-950/40 text-amber-900 dark:text-amber-200 text-xs font-bold uppercase tracking-wider border border-amber-200 dark:border-amber-800">
              <Archive className="w-3.5 h-3.5 text-amber-700 dark:text-amber-400" />
              <span>Arsip Konten & Media Terprivat</span>
            </div>
            <h2 className="font-serif text-2xl sm:text-3xl lg:text-4xl font-bold text-[#2D2D2A] dark:text-[#F0F0EA] mt-1">
              Arsipan Postingan Sekolah
            </h2>
            <p className="text-xs sm:text-sm text-[#8E8E85] dark:text-[#A4A49B] max-w-2xl leading-relaxed mt-1">
              Semua foto, video, berita kegiatan, dan pengumuman yang di-private oleh Admin Utama atau Admin Kelas tersimpan aman di sini dan dapat dipulihkan kembali menjadi publik kapan saja.
            </p>
          </div>

          <div className="px-5 py-3 bg-[#F5F5F0] dark:bg-[#1E1E1C] border border-[#E0E0D8] dark:border-[#383832] rounded-2xl text-center shrink-0">
            <span className="text-[11px] font-bold uppercase tracking-wider text-[#8E8E85] dark:text-[#A4A49B] block">Total Arsip</span>
            <span className="font-serif text-2xl font-bold text-[#8C4A32] dark:text-[#E07A5F]">{allArchived.length} Item</span>
          </div>
        </div>
      </div>

      {/* Tabs Filter */}
      <div className="flex items-center gap-2 overflow-x-auto no-scrollbar py-1">
        {[
          { id: 'all', label: 'Semua Arsip', count: allArchived.length },
          { id: 'slider', label: 'Sorotan Beranda', count: privateSliders.length },
          { id: 'activities', label: 'Kegiatan', count: privateActivities.length },
          { id: 'announcements', label: 'Pengumuman', count: privateAnnouncements.length },
          { id: 'classes', label: 'Foto Kelas', count: privateClassPhotos.length },
          { id: 'eskul', label: 'Foto Eskul', count: privateEskulPhotos.length }
        ].map(tab => (
          <button
            key={tab.id}
            onClick={() => setActiveTab(tab.id as any)}
            className={`px-4 py-2 rounded-xl text-xs font-semibold whitespace-nowrap transition-all cursor-pointer flex items-center gap-2 ${
              activeTab === tab.id
                ? 'bg-[#8C4A32] text-white font-bold shadow-xs'
                : 'bg-white dark:bg-[#242422] border border-[#E0E0D8] dark:border-[#383832] text-[#5A5A40] dark:text-[#C5C5B0] hover:bg-[#F5F5F0]'
            }`}
          >
            <span>{tab.label}</span>
            <span className="px-1.5 py-0.5 text-[10px] rounded-full bg-black/10 dark:bg-white/10">
              {tab.count}
            </span>
          </button>
        ))}
      </div>

      {/* Archived Grid */}
      {filteredItems.length === 0 ? (
        <div className="bg-white dark:bg-[#242422] border border-[#E0E0D8] dark:border-[#383832] rounded-[32px] p-12 text-center space-y-3">
          <div className="w-16 h-16 rounded-full bg-amber-50 dark:bg-amber-950/40 text-amber-800 dark:text-amber-300 flex items-center justify-center mx-auto">
            <Archive className="w-8 h-8" />
          </div>
          <h3 className="font-serif text-xl font-bold text-[#2D2D2A] dark:text-[#F0F0EA]">
            Tidak Ada Postingan yang Di-Private
          </h3>
          <p className="text-xs sm:text-sm text-[#8E8E85] dark:text-[#A4A49B] max-w-md mx-auto">
            Semua konten saat ini berstatus publik dan dapat dilihat oleh pengunjung situs. Untuk memprivate postingan, klik tombol gembok pada kartu postingan.
          </p>
        </div>
      ) : (
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-6">
          {filteredItems.map((item, idx) => (
            <ScrollEmboss key={`${item.type}-${item.id}`} delay={idx * 40} interactiveHover className="h-full">
              <div className="h-full bg-white dark:bg-[#242422] border border-amber-200 dark:border-amber-800/60 rounded-[28px] overflow-hidden shadow-sm hover:shadow-md transition-all flex flex-col justify-between group">
                {/* Media Preview or Announcement Banner */}
                {item.mediaUrl ? (
                  <div className="aspect-[16/10] w-full bg-[#1E1E1C] overflow-hidden relative">
                    {item.mediaType === 'video' ? (
                      <video
                        src={item.mediaUrl}
                        className="w-full h-full object-cover"
                        controls
                        playsInline
                      />
                    ) : (
                      <img
                        src={item.mediaUrl}
                        alt={item.title}
                        className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
                        loading="lazy"
                      />
                    )}
                    <span className="absolute top-3 left-3 bg-neutral-900/80 backdrop-blur-sm text-amber-300 px-2.5 py-1 rounded-lg text-[10px] font-bold border border-amber-500/30 flex items-center gap-1">
                      <Lock className="w-3 h-3" />
                      <span>{item.origin}</span>
                    </span>
                  </div>
                ) : (
                  <div className="p-5 bg-amber-50/50 dark:bg-amber-950/20 border-b border-amber-200 dark:border-amber-800/40">
                    <span className="inline-flex items-center gap-1 text-[10px] font-bold uppercase tracking-wider text-amber-800 dark:text-amber-300 mb-2">
                      <Lock className="w-3 h-3" />
                      <span>{item.origin}</span>
                    </span>
                    <p className="text-xs text-[#8E8E85] dark:text-[#A4A49B] line-clamp-3">
                      {item.content || item.subtitle}
                    </p>
                  </div>
                )}

                {/* Content */}
                <div className="p-5 space-y-3 flex-1 flex flex-col justify-between">
                  <div className="space-y-1">
                    <span className="text-[10px] font-bold text-[#8E8E85] dark:text-[#A4A49B]">
                      {item.subtitle}
                    </span>
                    <h4 className="font-serif text-base font-bold text-[#2D2D2A] dark:text-[#F0F0EA] line-clamp-2">
                      {item.title}
                    </h4>
                    {item.archivedAt && (
                      <p className="text-[11px] text-[#8E8E85] dark:text-[#A4A49B] flex items-center gap-1 mt-1">
                        <Clock className="w-3 h-3" />
                        <span>Diarsipkan: {new Date(item.archivedAt).toLocaleDateString('id-ID', { day: 'numeric', month: 'short', year: 'numeric' })}</span>
                      </p>
                    )}
                  </div>

                  {/* Actions: Pulihkan / Jadikan Publik & Hapus Permanen */}
                  <div className="pt-3 border-t border-[#E0E0D8] dark:border-[#383832] flex items-center justify-between gap-2">
                    <button
                      onClick={item.onUnprivate}
                      className="px-3.5 py-1.5 bg-emerald-600 hover:bg-emerald-700 text-white rounded-xl text-xs font-bold flex items-center gap-1.5 transition-all cursor-pointer active:scale-95 shadow-xs"
                      title="Pulihkan dan tampilkan kembali di website publik"
                    >
                      <Unlock className="w-3.5 h-3.5" />
                      <span>Pulihkan / Publik</span>
                    </button>

                    <button
                      onClick={item.onDelete}
                      className="p-1.5 text-xs text-red-600 hover:bg-red-50 dark:hover:bg-red-950/30 rounded-xl transition-colors cursor-pointer"
                      title="Hapus Permanen"
                    >
                      <Trash2 className="w-4 h-4" />
                    </button>
                  </div>
                </div>
              </div>
            </ScrollEmboss>
          ))}
        </div>
      )}
    </div>
  );
};
